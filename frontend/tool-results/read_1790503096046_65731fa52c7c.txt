   108→   108→/* ============================== Placeholder resolution ============================== */
   109→   109→
   110→   110→/** Replaces {company} / {skill} / {role} with data from the active resume. */
   111→   111→export function resolvePlaceholders(text: string, resume: ResumeData): string {
   112→   112→  const company = resume.experience[0]?.company || "your current company";
   113→   113→  const skill = resume.skills[0]?.name || "your core skill";
   114→   114→  const role = resume.personal.jobTitle || resume.title || "this role";
   115→   115→  return text
   116→   116→    .replace(/\{company\}/g, company)
   117→   117→    .replace(/\{skill\}/g, skill)
   118→   118→    .replace(/\{role\}/g, role);
   119→   119→}
   120→   120→
   121→   121→function resolveStar(star: StarAnswer, resume: ResumeData): StarAnswer {
   122→   122→  return {
   123→   123→    situation: resolvePlaceholders(star.situation, resume),
   124→   124→    task: resolvePlaceholders(star.task, resume),
   125→   125→    action: resolvePlaceholders(star.action, resume),
   126→   126→    result: resolvePlaceholders(star.result, resume),
   127→   127→  };
   128→   128→}

   946→   946→function shuffle<T>(input: readonly T[]): T[] {
   947→   947→  const arr = [...input];
   948→   948→  for (let i = arr.length - 1; i > 0; i--) {
   949→   949→    const j = Math.floor(Math.random() * (i + 1));
   950→   950→    const a = arr[i];
   951→   951→    const b = arr[j];
   952→   952→    if (a !== undefined && b !== undefined) {
   953→   953→      arr[i] = b;
   954→   954→      arr[j] = a;
   955→   955→    }
   956→   956→  }
   957→   957→  return arr;
   958→   958→}

   974→   974→/**
   975→   975→ * Builds a practice set: filters the bank by category + difficulty, shuffles,
   976→   976→ * prefers resume-tailored variants, and round-robins across categories so
   977→   977→ * every selected category is represented when possible.
   978→   978→ */
   979→   979→export function generateQuestions(
   980→   980→  resume: ResumeData,
   981→   981→  { categories, difficulty, count = 8 }: GenerateOptions
   982→   982→): PracticeQuestion[] {
   983→   983→  const allowed = categories.length > 0 ? categories : CATEGORY_ORDER;
   984→   984→  const pool = QUESTION_BANK.filter(
   985→   985→    (q) =>
   986→   986→      allowed.includes(q.category) &&
   987→   987→      (difficulty === "all" || q.difficulty === difficulty)
   988→   988→  );
   989→   989→
   990→   990→  // One shuffled, tailored-first bucket per selected category.
   991→   991→  const buckets = new Map<QuestionCategory, Question[]>();
   992→   992→  for (const cat of allowed) {
   993→   993→    const inCat = shuffle(pool.filter((q) => q.category === cat)).sort(
   994→   994→      (a, b) => Number(Boolean(b.tailored)) - Number(Boolean(a.tailored))
   995→   995→    );
   996→   996→    if (inCat.length > 0) buckets.set(cat, inCat);
   997→   997→  }
   998→   998→
   999→   999→  const picked: Question[] = [];
  1000→  1000→  while (picked.length < count && buckets.size > 0) {
  1001→  1001→    for (const [cat, queue] of buckets) {
  1002→  1002→      const next = queue.shift();
  1003→  1003→      if (next) picked.push(next);
  1004→  1004→      if (queue.length === 0) buckets.delete(cat);
  1005→  1005→      if (picked.length >= count) break;
  1006→  1006→    }
  1007→  1007→  }
  1008→  1008→
  1009→  1009→  return picked.map((q) => instantiate(q, resume));
  1010→  1010→}
