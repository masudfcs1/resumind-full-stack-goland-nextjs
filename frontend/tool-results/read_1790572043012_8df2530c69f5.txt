   430→   430→/* "+12 pts" pill that counts up when scrolled into view */
   431→   431→function DeltaPill() {
   432→   432→  const ref = React.useRef<HTMLSpanElement>(null);
   433→   433→  const inView = useInView(ref, { once: true, margin: "-60px" });
   434→   434→  const mv = useMotionValue(0);
   435→   435→  const text = useTransform(mv, (v: number) => `+${Math.round(v)} pts`);
   436→   436→
   437→   437→  React.useEffect(() => {
   438→   438→    if (!inView) return;
   439→   439→    const controls = animate(mv, 12, { duration: 1.4, ease: "easeOut", delay: 0.5 });
   440→   440→    return () => controls.stop();
   441→   441→  }, [inView, mv]);
   442→   442→
   443→   443→  return (
