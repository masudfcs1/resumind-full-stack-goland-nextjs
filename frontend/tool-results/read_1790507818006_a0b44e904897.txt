   331→   331→
   332→   332→export default function MyResumesPage() {
   333→   333→  const mounted = useMounted();
   334→   334→  const resumes = useResumeStore((s) => s.resumes);
   335→   335→  const activeResumeId = useResumeStore((s) => s.activeResumeId);
   336→   336→  const active = useActiveResume();
   337→   337→  const duplicateResume = useResumeStore((s) => s.duplicateResume);
   338→   338→  const deleteResume = useResumeStore((s) => s.deleteResume);
   339→   339→  const scoreHistory = useResumeStore((s) => s.scoreHistory);
   340→   340→
   341→   341→  const [query, setQuery] = React.useState("");
   342→   342→  const [sort, setSort] = React.useState<SortKey>("recent");
   343→   343→  const [view, setView] = React.useState<ViewMode>("grid");
   344→   344→  const [createOpen, setCreateOpen] = React.useState(false);
   345→   345→  const [selectMode, setSelectMode] = React.useState(false);
   346→   346→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   347→   347→  const [confirmDelete, setConfirmDelete] = React.useState(false);
   348→   348→
   349→   349→  const scoreMap = React.useMemo(() => {
   350→   350→    const map = new Map<string, number>();
   351→   351→    for (const r of resumes) map.set(r.id, atsAnalyze(r).score);
   352→   352→    return map;
   353→   353→  }, [resumes]);
   354→   354→
   355→   355→  /* Health summary: scan RECENCY per resume (fresh < 24h, aging 1–7d,
   356→   356→     stale > 7d) with never-scanned counted separately. */
   357→   357→  const healthCounts = React.useMemo(() => {
   358→   358→    const now = Date.now();
   359→   359→    let fresh = 0;
   360→   360→    let aging = 0;
   361→   361→    let stale = 0;
   362→   362→    let never = 0;
   363→   363→    for (const r of resumes) {
   364→   364→      const h = resumeHealth(scoreHistory, r.id, now);
   365→   365→      if (h.bucket === "fresh") fresh += 1;
   366→   366→      else if (h.bucket === "aging") aging += 1;
   367→   367→      else if (h.scanned) stale += 1;
   368→   368→      else never += 1;
   369→   369→    }
   370→   370→    return { fresh, aging, stale, never };
   371→   371→  }, [resumes, scoreHistory]);
   372→   372→
   373→   373→  const visible = React.useMemo(() => {
   374→   374→    const q = query.trim().toLowerCase();
   375→   375→    const filtered = resumes.filter(
   376→   376→      (r) =>
   377→   377→        !q ||
   378→   378→        r.title.toLowerCase().includes(q) ||
   379→   379→        r.personal.fullName.toLowerCase().includes(q) ||
   380→   380→        r.personal.jobTitle.toLowerCase().includes(q)
   381→   381→    );
   382→   382→    const scoreOf = (id: string) => scoreMap.get(id) ?? 0;
   383→   383→    switch (sort) {
   384→   384→      case "name":
   385→   385→        return [...filtered].sort((a, b) => a.title.localeCompare(b.title));
   386→   386→      case "ats":
   387→   387→        return [...filtered].sort((a, b) => scoreOf(b.id) - scoreOf(a.id));
   388→   388→      default:
   389→   389→        return [...filtered].sort((a, b) => b.updatedAt - a.updatedAt);
   390→   390→    }
   391→   391→  }, [resumes, query, sort, scoreMap]);
   392→   392→
   393→   393→  /* Keep the selection honest when resumes disappear (per-card deletes, resets) */
   394→   394→  React.useEffect(() => {
   395→   395→    setSelected((prev) => {
   396→   396→      if (prev.size === 0) return prev;
   397→   397→      const ids = new Set(resumes.map((r) => r.id));
   398→   398→      const next = new Set([...prev].filter((id) => ids.has(id)));
   399→   399→      return next.size === prev.size ? prev : next;
   400→   400→    });
   401→   401→  }, [resumes]);
   402→   402→
   403→   403→  const exitSelectMode = React.useCallback(() => {
   404→   404→    setSelectMode(false);
   405→   405→    setSelected(new Set());
   406→   406→  }, []);
   407→   407→
   408→   408→  const toggleSelect = React.useCallback((id: string) => {
   409→   409→    setSelected((prev) => {
   410→   410→      const next = new Set(prev);
   411→   411→      if (next.has(id)) next.delete(id);
   412→   412→      else next.add(id);
   413→   413→      return next;
   414→   414→    });
   415→   415→  }, []);
   416→   416→
   417→   417→  /* ESC exits selection mode — but never while an AlertDialog (bulk delete or
   418→   418→     per-card delete) is open. The listener runs in the CAPTURE phase so it sees
   419→   419→     the dialog before Radix's document-level handler closes it: Radix calls
   420→   420→     onOpenChange(false) synchronously during the Escape dispatch (React 19
   421→   421→     flush), so a bubble-phase DOM/state check would already observe the dialog
   422→   422→     as gone and wrongly exit selection mode. */
   423→   423→  React.useEffect(() => {
   424→   424→    if (!selectMode) return;
   425→   425→    const onKey = (e: KeyboardEvent) => {
   426→   426→      if (e.key !== "Escape") return;
   427→   427→      if (document.querySelector("[role=alertdialog]")) return;
   428→   428→      exitSelectMode();
   429→   429→    };
   430→   430→    window.addEventListener("keydown", onKey, true);
   431→   431→    return () => window.removeEventListener("keydown", onKey, true);
   432→   432→  }, [selectMode, exitSelectMode]);
   433→   433→
   434→   434→  const handleBulkDuplicate = () => {
   435→   435→    let ok = 0;
   436→   436→    for (const id of selected) {
   437→   437→      if (duplicateResume(id)) ok += 1;
   438→   438→    }
   439→   439→    if (ok > 0) {
   440→   440→      toast.success(`Duplicated ${ok} ${ok === 1 ? "resume" : "resumes"}`, {
   441→   441→        description: "Copies were added to the top of your list.",
   442→   442→      });
   443→   443→    }
   444→   444→    /* Selection mode stays ON and the same set stays selected so users can continue. */
   445→   445→  };
   446→   446→
   447→   447→  const handleBulkDelete = () => {
   448→   448→    const ids = [...selected];
   449→   449→    for (const id of ids) deleteResume(id);
   450→   450→    if (ids.length > 0) {
