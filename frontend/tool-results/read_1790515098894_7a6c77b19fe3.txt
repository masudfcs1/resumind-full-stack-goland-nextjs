   164→export default function AtsPage() {
   165→  // useSearchParams needs a Suspense boundary for static prerendering.

   172→
   173→function AtsScanner() {
   174→  const mounted = useMounted();
   175→  const searchParams = useSearchParams();
   176→  const router = useRouter();
   177→  const resumes = useResumeStore((s) => s.resumes);
   178→  const scoreHistory = useResumeStore((s) => s.scoreHistory);
   179→  const activeResumeId = useResumeStore((s) => s.activeResumeId);
   180→  const setActive = useResumeStore((s) => s.setActive);
   181→  const logScore = useResumeStore((s) => s.logScore);
   182→
   183→  const resume =
   184→    resumes.find((r) => r.id === activeResumeId) ?? resumes[0];
   185→
   186→  const [jd, setJd] = React.useState("");
   187→  const [jdOpen, setJdOpen] = React.useState(false);
   188→  const [scanning, setScanning] = React.useState(false);
   189→  const [result, setResult] = React.useState<AtsResult | null>(null);
   190→  const [scanId, setScanId] = React.useState(0);
   191→  /** Last history entry that existed BEFORE the live scan (the "old" side). */
   192→  const [baseline, setBaseline] = React.useState<ScoreEntry | null>(null);
   193→
   194→  const usedJobDescription = jd.trim().length > 40;
   195→
   196→  /* --- Shareable ?resume=<id> param --- */
   197→
   198→  const paramResumeId = searchParams.get("resume");
   199→
   200→  // URL → store: a valid ?resume= preselects that resume (falls back to the
   201→  // store's active resume, then the first resume, when missing/invalid).
   202→  React.useEffect(() => {
   203→    if (!mounted) return;
   204→    // window.location fallback: on a fresh load the router's searchParams
   205→    // (hook) settle a beat later than the address bar — without this, a
   206→    // deep link would be missed entirely.
   207→    const urlId =
   208→      paramResumeId ?? new URLSearchParams(window.location.search).get("resume");
   209→    if (!urlId) return;
   210→    if (!resumes.some((r) => r.id === urlId)) return;
   211→    if (urlId !== activeResumeId) setActive(urlId);
   212→  }, [mounted, paramResumeId, activeResumeId, resumes, setActive]);
   213→
   214→  // Store → URL: keep ?resume= in sync with the selection so any scan state is
   215→  // shareable (replace — the back button stays clean). Also heals an invalid
   216→  // or missing param to the resume actually shown.
   217→  React.useEffect(() => {
   218→    if (!mounted || !activeResumeId) return;
   219→    const locParams = new URLSearchParams(window.location.search);
   220→    // Router state not ready yet (hook still empty while the address bar
   221→    // carries a ?resume=) — never "heal" a deep link before the hook has had
   222→    // the chance to see it.
   223→    if (searchParams.get("resume") === null && locParams.get("resume")) return;
   224→    if (locParams.get("resume") === activeResumeId) return;
   225→    locParams.set("resume", activeResumeId);
   226→    router.replace(`/dashboard/ats?${locParams.toString()}`, { scroll: false });
   227→  }, [mounted, activeResumeId, router, searchParams]);
   228→
   229→  /* --- Scan state --- */
   230→
   231→  const resumeId = resume?.id ?? "";
   232→
   233→  // A finished scan always describes the currently selected resume: switching
   234→  // resumes clears the previous result so the delta panel starts fresh.
   235→  React.useEffect(() => {
   236→    if (!mounted) return;
   237→    setResult(null);
   238→    setBaseline(null);
   239→  }, [mounted, resumeId]);
   240→
   241→  const handleAnalyze = async () => {
   242→    if (!resume || scanning) return;
   243→    const target = resume;
   244→    setScanning(true);
   245→    try {
   246→      await sleep(1200);
   247→      // Ignore stale completions if the user switched resumes mid-scan.
   248→      if (useResumeStore.getState().activeResumeId !== target.id) return;
   249→      const res = atsAnalyze(target, jd.trim() || undefined);
   250→      // Capture the pre-scan "old" side before this scan is logged.
   251→      setBaseline(
   252→        lastScoreEntryFor(
   253→          useResumeStore.getState().scoreHistory,
   254→          target.id
   255→        )
   256→      );
   257→      setResult(res);
   258→      setScanId((n) => n + 1);
   259→      logScore(target.id, target.title, res.score);
   260→      toast.success(`Scan complete — ATS score ${res.score}/100`, {
   261→        description: "Score logged to your dashboard history.",
   262→      });
   263→    } catch {
   264→      toast.error("Something went wrong while scanning. Try again.");
   265→    } finally {
   266→      setScanning(false);
   267→    }
   268→  };
   269→
   270→  /* --- vs last scan (delta) --- */
   271→
   272→  // Before any scan: the resume's latest history entry, if one exists.
   273→  const loadBaseline = React.useMemo(
   274→    () =>
   275→      mounted && resume && !result
   276→        ? lastScoreEntryFor(scoreHistory, resume.id)
   277→        : null,
   278→    [mounted, resume, result, scoreHistory]
   279→  );
   280→
   281→  const panelEntry = result ? baseline : loadBaseline;
   282→  const showDeltaPanel = Boolean(mounted && resume && panelEntry);
   283→  const deltaDiff = result && panelEntry ? result.score - panelEntry.score : 0;
   284→  const deltaKindValue = deltaKind(deltaDiff);
   285→
   286→  return (
   287→    <motion.div
   288→      variants={staggerContainer}
   289→      initial="hidden"
   290→      animate="show"
   291→      className="mx-auto w-full max-w-6xl space-y-6"
   292→    >
   293→      <ToolPageHeader
   294→        icon={Target}
   295→        title="ATS Score Scanner"
   296→        description="Run your resume through our simulated applicant tracking system. Get an instant score, a category breakdown, and concrete fixes."
   297→        actions={
   298→          <>
   299→            <ResumeSelect
   300→              value={resume?.id ?? ""}
   301→              onValueChange={setActive}
   302→              ariaLabel="Resume to analyze"
   303→              className="w-full sm:w-64"
   304→            />
   305→            <Button
   306→              onClick={handleAnalyze}
   307→              disabled={scanning || !resume || !mounted}
   308→              className="bg-gradient-to-r from-emerald-500 to-teal-600 text-white shadow-lg shadow-emerald-500/25 hover:from-emerald-600 hover:to-teal-700"
   309→              aria-label="Analyze resume with ATS scanner"
   310→            >
   311→              {scanning ? (
   312→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   313→              ) : (
   314→                <ScanSearch className="size-4" aria-hidden="true" />
   315→              )}
   316→              {scanning ? "Scanning…" : "Analyze"}
   317→            </Button>
   318→          </>
   319→        }
   320→      />
   321→
   322→      {/* Job description (optional) */}
   323→      <motion.section variants={staggerItem} aria-label="Job description input">
   324→        <Card>
   325→          <CardContent className="p-4 sm:p-5">
   326→            <Collapsible open={jdOpen} onOpenChange={setJdOpen}>
   327→              <div className="flex flex-wrap items-center justify-between gap-3">
   328→                <CollapsibleTrigger asChild>
   329→                  <Button
   330→                    variant="ghost"
   331→                    size="sm"
   332→                    className="text-muted-foreground"
   333→                    aria-expanded={jdOpen}
   334→                  >
   335→                    <FileText className="size-4" aria-hidden="true" />
   336→                    Job description{" "}
   337→                    <span className="font-normal">
   338→                      {jd.trim() ? "(added)" : "(optional)"}
   339→                    </span>
   340→                    <ChevronDown
   341→                      className={cn(
   342→                        "size-4 transition-transform duration-200",
   343→                        jdOpen && "rotate-180"
   344→                      )}
   345→                      aria-hidden="true"
   346→                    />
   347→                  </Button>
   348→                </CollapsibleTrigger>
   349→                <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   350→                  <Sparkles className="size-3.5 text-emerald-500" aria-hidden="true" />
   351→                  Matching against a real job description boosts accuracy
   352→                </p>
   353→              </div>
   354→              <CollapsibleContent className="pt-4">
   355→                <div className="space-y-2">
   356→                  <Textarea
   357→                    value={jd}
   358→                    onChange={(e) => setJd(e.target.value)}
   359→                    placeholder="Paste the full job description here — responsibilities, requirements, the works…"
   360→                    aria-label="Job description"
   361→                    className="min-h-[140px] resize-y"
   362→                  />
   363→                  <div className="flex items-center gap-3">
   364→                    <button
   365→                      type="button"
   366→                      onClick={() => setJd(SAMPLE_JD)}
   367→                      className="text-xs font-medium text-emerald-600 underline-offset-4 hover:underline dark:text-emerald-400"
   368→                    >
   369→                      Paste sample JD
   370→                    </button>
   371→                    {jd ? (
   372→                      <button
   373→                        type="button"
   374→                        onClick={() => setJd("")}
   375→                        className="text-xs font-medium text-muted-foreground underline-offset-4 hover:underline"
   376→                      >
   377→                        Clear
   378→                      </button>
   379→                    ) : null}
   380→                  </div>
   381→                </div>
   382→              </CollapsibleContent>
   383→            </Collapsible>
   384→          </CardContent>
   385→        </Card>
   386→      </motion.section>
   387→
   388→      {/* vs last scan (delta panel) */}
   389→      {showDeltaPanel && panelEntry ? (
   390→        <motion.section
   391→          key={result ? `delta-full-${scanId}` : "delta-load"}
   392→          variants={staggerItem}
   393→          aria-label="Comparison with last scan"
   394→          aria-live="polite"
   395→          data-delta={result ? deltaKindValue : "none"}
   396→        >
   397→          <div className="rounded-xl border p-4 sm:p-5">
   398→            <div className="flex flex-wrap items-center gap-x-3 gap-y-2">
   399→              <span
   400→                aria-hidden="true"
   401→                className="flex size-8 shrink-0 items-center justify-center rounded-lg bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   402→              >
   403→                <History className="size-4" />
   404→              </span>
   405→              <div className="min-w-0 flex-1">
   406→                <p className="text-sm font-medium">vs last scan</p>
   407→                <p className="mt-0.5 text-sm tabular-nums text-muted-foreground">
   408→                  Last scan{" "}
   409→                  {formatDistanceToNow(new Date(panelEntry.at), {
   410→                    addSuffix: true,
   411→                  })}
   412→                  :{" "}
   413→                  <span className="font-semibold text-foreground">
   414→                    {panelEntry.score}
   415→                  </span>
   416→                  {result ? (
   417→                    <>
   418→                      {" \u2192 "}
   419→                      Now:{" "}
   420→                      <span className="font-semibold text-foreground">
   421→                        {result.score}
   422→                      </span>
   423→                    </>
   424→                  ) : (
   425→                    <span className="font-normal">
   426→                      {" \u00b7 "} run a scan to compare
   427→                    </span>
   428→                  )}
   429→                </p>
   430→              </div>
   431→              {result ? (
   432→                <span
   433→                  data-delta={deltaKindValue}
   434→                  data-delta-value={deltaDiff}
   435→                  aria-label={deltaAriaLabel(deltaDiff)}
   436→                  className={cn(
   437→                    "inline-flex shrink-0 items-center gap-1 rounded-full px-2.5 py-1 text-xs font-semibold tabular-nums ring-1",
   438→                    DELTA_CHIP_CLASS[deltaKindValue]
   439→                  )}
   440→                >
   441→                  <DeltaChipIcon kind={deltaKindValue} />
   442→                  {deltaKindValue === "flat" ? "No change" : formatDelta(deltaDiff)}
   443→                </span>
   444→              ) : null}
   445→            </div>
   446→            {result && deltaInsight(result.breakdown) ? (
   447→              <p className="mt-3 flex items-center gap-1.5 border-t pt-3 text-xs text-muted-foreground">
   448→                <Sparkles
   449→                  className="size-3.5 shrink-0 text-emerald-500"
   450→                  aria-hidden="true"
   451→                />
   452→                {deltaInsight(result.breakdown)}
   453→              </p>
   454→            ) : null}
   455→          </div>
   456→        </motion.section>
   457→      ) : null}
   458→
   459→      {/* Scanning animation */}
   460→      {scanning ? (
   461→        <motion.section
   462→          variants={staggerItem}
   463→          aria-busy="true"
   464→          aria-label="Scanning resume"
   465→        >
   466→          <Card className="overflow-hidden">
   467→            <CardContent className="p-6">
   468→              <p className="flex items-center justify-center gap-2 text-sm font-medium" role="status">
   469→                <Loader2 className="size-4 animate-spin text-emerald-600 dark:text-emerald-400" aria-hidden="true" />
   470→                Scanning resume
   471→                <span className="animate-pulse" aria-hidden="true">…</span>
   472→              </p>
   473→              <div className="relative mx-auto mt-6 max-w-sm" aria-hidden="true">
   474→                <div className="rounded-lg border bg-background p-5 shadow-sm">
   475→                  <div className="mx-auto mb-4 h-3 w-28 rounded bg-muted" />
   476→                  <div className="space-y-2">
   477→                    {[92, 100, 78, 96, 64, 88, 72, 40].map((w, i) => (
   478→                      <div
   479→                        key={i}
   480→                        className="h-2 rounded bg-muted"
   481→                        style={{ width: `${w}%` }}
   482→                      />
   483→                    ))}
   484→                  </div>
   485→                  <div className="mt-5 h-2 w-20 rounded bg-emerald-500/40" />
   486→                </div>
   487→                <motion.div
   488→                  className="pointer-events-none absolute inset-x-0 h-14 bg-gradient-to-b from-transparent via-emerald-400/30 to-transparent"
   489→                  initial={{ top: "-18%" }}
   490→                  animate={{ top: "112%" }}
   491→                  transition={{
   492→                    duration: 1.05,
   493→                    repeat: Infinity,
   494→                    ease: "linear",
   495→                  }}
   496→                />
   497→              </div>
   498→            </CardContent>
   499→          </Card>
   500→        </motion.section>
   501→      ) : !result ? (
   502→        /* Empty state before first scan */
   503→        <motion.section variants={staggerItem} aria-label="Start a scan">
   504→          <div className="flex flex-col items-center justify-center rounded-xl border border-dashed p-10 text-center">
   505→            <div
   506→              aria-hidden="true"
   507→              className="flex size-14 items-center justify-center rounded-2xl bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   508→            >
   509→              <ScanSearch className="size-7" />
   510→            </div>
   511→            <h2 className="mt-4 font-display text-lg font-semibold">
   512→              Run your first scan
   513→            </h2>
   514→            <p className="mt-1 max-w-sm text-sm text-muted-foreground">
   515→              Pick a resume above and hit Analyze — optionally paste a job
   516→              description first for keyword-level matching.
   517→            </p>
   518→          </div>
   519→        </motion.section>
   520→      ) : null}
   521→
   522→      {/* Results */}
   523→      {result ? (
   524→        <motion.section
   525→          key={scanId}
   526→          variants={staggerContainer}
   527→          initial="hidden"
   528→          animate="show"
   529→          aria-label="ATS scan results"
   530→          aria-live="polite"
   531→          className="space-y-6"
   532→        >
   533→          {/* Score + breakdown */}
   534→          <div className="grid gap-6 lg:grid-cols-5">
   535→            <motion.div variants={staggerItem} className="lg:col-span-2">
   536→              <Card className="h-full">
   537→                <CardHeader>
   538→                  <CardTitle className="font-display">Your ATS score</CardTitle>
   539→                  <CardDescription>
   540→                    {usedJobDescription
   541→                      ? "Matched against the pasted job description"
   542→                      : "Matched against typical role keywords"}
   543→                  </CardDescription>
   544→                </CardHeader>
   545→                <CardContent className="flex flex-col items-center gap-4">
   546→                  <ScoreGauge score={result.score} grade={result.grade} />
   547→                  <Badge variant="secondary" className="gap-1.5">
   548→                    <FileText className="size-3" aria-hidden="true" />
   549→                    {result.wordCount} words scanned
   550→                  </Badge>
   551→                  <Button
   552→                    variant="ghost"
   553→                    size="sm"
   554→                    onClick={handleAnalyze}
   555→                    disabled={scanning}
