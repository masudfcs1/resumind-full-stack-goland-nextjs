    92→
    93→export default function CoverLetterPage() {
    94→  const mounted = useMounted();

    99→  const incrementCoverLetters = useResumeStore((s) => s.incrementCoverLetters);
   100→
   101→  const resume = resumes.find((r) => r.id === activeResumeId) ?? activeResume;
   102→
   103→  const [company, setCompany] = React.useState("");
   104→  const [role, setRole] = React.useState("");
   105→  const [roleTouched, setRoleTouched] = React.useState(false);
   106→  const [tone, setTone] = React.useState<CoverTone>("professional");

   108→  const [generating, setGenerating] = React.useState(false);
   109→  const [letter, setLetter] = React.useState<string | null>(null);
   110→  const [typedLen, setTypedLen] = React.useState(0);
   111→  const [history, setHistory] = React.useState<MatchHistoryEntry[]>([]);
   112→  const [selectedJdId, setSelectedJdId] = React.useState<string | null>(null);
   113→  const [injected, setInjected] = React.useState<string[]>([]);
   114→  const [genKey, setGenKey] = React.useState(0);
   115→
   116→  /* Prefill role from the active resume once hydrated (unless user edited). */

   130→    () => history.filter((h) => h.resumeId === activeResumeId),

   134→
   135→  /* A selected scan belongs to one resume — drop the pick when it changes. */

   137→    setSelectedJdId(null);

   145→    }

   147→    const id = window.setInterval(() => {
   148→      setTypedLen((n) => {
   149→        if (n >= letter.length) {
   150→          window.clearInterval(id);
   151→          return n;
   152→        }
   153→        return Math.min(n + 2, letter.length);
   154→      });
   155→    }, 12);
   156→    return () => window.clearInterval(id);
   157→  }, [letter]);
   158→
   159→  const typing = letter !== null && typedLen < letter.length;
   160→  const typed = letter === null ? "" : letter.slice(0, typedLen);
   161→
   162→  const buildLetter = React.useCallback(
   163→    (): { text: string; injected: string[] } | null => {
   164→      if (!resume) return null;
   165→      const baseOpts = {
   166→        resume,
   167→        company: company.trim() || "your team",
   168→        role: role.trim() || resume.personal.jobTitle || "the open role",
   169→        tone,
   170→      };
   171→      if (selectedJd) {
   172→        const tailored = generateTailoredCoverLetter({
   173→          ...baseOpts,
   174→          jd: selectedJd.jd,
   175→        });
   176→        return { text: tailored.text, injected: tailored.injectedKeywords };
   177→      }
   178→      return { text: generateCoverLetter(baseOpts), injected: [] };
   179→    },
   180→    [resume, company, role, tone, selectedJd]

   188→      await sleep(1400);

   191→        setLetter(result.text);
   192→        setInjected(result.injected);
   193→        setGenKey((k) => k + 1);

   195→          result.injected.length > 0
   196→            ? `Tailored with ${result.injected.length} keyword${result.injected.length === 1 ? "" : "s"}`
   197→            : "Cover letter drafted"
   198→        );
   199→      }
   200→    } catch {
   201→      toast.error("Generation failed — please try again.");
   202→    } finally {
   203→      setGenerating(false);
   204→    }
   205→  };
   206→
   207→  const handleRegenerate = async () => {
   208→    if (generating || !resume) return;
   209→    setGenerating(true);
   210→    try {
   211→      await sleep(600);
   212→      const result = buildLetter();
   213→      if (result) {
   214→        setLetter(result.text);
   215→        setInjected(result.injected);
   216→        setGenKey((k) => k + 1);
   217→      }
   218→    } finally {
   219→      setGenerating(false);
   220→    }

   223→  const handleClearJd = () => setSelectedJdId(null);
   224→

   228→      await navigator.clipboard.writeText(letter);

   233→  };

   237→    const blob = new Blob([letter], { type: "text/plain;charset=utf-8" });

   239→    const a = document.createElement("a");
   240→    a.href = url;
   241→    const slug =
   242→      company
   243→        .trim()
   244→        .toLowerCase()
   245→        .replace(/[^a-z0-9]+/g, "-")
   246→        .replace(/(^-|-$)/g, "") || "draft";
   247→    a.download = `cover-letter-${slug}.txt`;
   248→    document.body.appendChild(a);
   249→    a.click();
   250→    a.remove();
   251→    URL.revokeObjectURL(url);
   252→    toast.success("Cover letter downloaded");
   253→  };
   254→
   255→  const handleSave = () => {
   256→    incrementCoverLetters();
   257→    toast.success("Saved to dashboard");
   258→  };
   259→
   260→  const handleSkip = () => {
   261→    if (letter !== null) setTypedLen(letter.length);
   262→  };
   263→
   264→  /* Split the letter into letterhead + body for styled rendering. */
   265→  const lines = typed.split("\n");
   266→  const headerLines = lines.slice(0, 2);
   267→  const bodyText = lines.slice(2).join("\n");
   268→
   269→  return (
   270→    <motion.div
   271→      variants={staggerContainer}
   272→      initial="hidden"
   273→      animate="show"
   274→      className="mx-auto w-full max-w-6xl space-y-6"

   289→                <Building2
   290→                  className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   291→                  aria-hidden="true"
   292→                />
   293→                <Input
   294→                  id="cl-company"
   295→                  value={company}
   296→                  onChange={(e) => setCompany(e.target.value)}
   297→                  placeholder="e.g. Lumen Labs"
   298→                  className="pl-9"
   299→                  autoComplete="off"
   300→                />
   301→              </div>
   302→            </div>
   303→            <div className="space-y-1.5">
   304→              <Label htmlFor="cl-role">Role</Label>

   306→                id="cl-role"
   307→                value={role}
   308→                onChange={(e) => {
   309→                  setRole(e.target.value);
   310→                  setRoleTouched(true);
   311→                }}
   312→                placeholder="e.g. Senior Frontend Engineer"
   313→                autoComplete="off"
   314→              />
   315→            </div>
   316→            <div className="space-y-1.5">
   317→              <Label htmlFor="cl-tone">Tone</Label>
   318→              <Select value={tone} onValueChange={(v) => setTone(v as CoverTone)}>

   320→                  <SelectValue />
   321→                </SelectTrigger>
   322→                <SelectContent>
   323→                  {TONES.map((t) => (
   324→                    <SelectItem key={t.value} value={t.value}>
   325→                      <span className="flex items-center gap-2">
   326→                        <t.icon className="size-4 text-emerald-600 dark:text-emerald-400" aria-hidden="true" />
   327→                        {t.label}
   328→                        <span className="text-xs text-muted-foreground">— {t.hint}</span>
   329→                      </span>
   330→                    </SelectItem>
   331→                  ))}
   332→                </SelectContent>
   333→              </Select>
   334→            </div>
   335→            <div className="space-y-1.5">
   336→              <Label>Resume</Label>
   337→              <ResumeSelect
   338→                value={resume?.id ?? ""}

   340→                ariaLabel="Resume to base the letter on"

   350→                  return (
   351→                    <button
   352→                      key={s.value}

   360→                          : "border-border"
   361→                      )}
   362→                    >
   363→                      <span className="flex h-4 items-end" aria-hidden="true">
   364→                        {s.value === "classic" ? (
   365→                          <span className="mx-auto h-1.5 w-10 rounded-full bg-zinc-400 dark:bg-zinc-500" />
   366→                        ) : s.value === "modern" ? (
   367→                          <span className="flex items-center gap-1">
   368→                            <span className="h-3.5 w-1 rounded bg-emerald-500" />
   369→                            <span className="h-1.5 w-9 rounded-full bg-zinc-400 dark:bg-zinc-500" />

   371→                        ) : (
   372→                          <span className="h-1 w-7 rounded-full bg-zinc-400 dark:bg-zinc-500" />
   373→                        )}
   374→                      </span>
   375→                      <span className="mt-2 block text-xs font-semibold">{s.label}</span>
   376→                      <span className="block text-[11px] text-muted-foreground">{s.blurb}</span>
   377→                    </button>
   378→                  );
   379→                })}
   380→              </div>
   381→            </div>
   382→
   383→            {/* Tailor to a job description (Job Match scans) */}

   385→              className="min-w-0 space-y-2.5 rounded-xl border border-emerald-500/20 bg-emerald-500/[0.04] p-4 sm:col-span-2 dark:border-emerald-500/25 dark:bg-emerald-500/[0.06]"

   396→                    variant="ghost"

   402→                    <X className="size-3.5" aria-hidden="true" />
   403→                    Clear
   404→                  </Button>
   405→                ) : null}
   406→              </div>
   407→
   408→              {!mounted ? (

   410→                  className="h-11 animate-pulse rounded-lg bg-emerald-500/5"
   411→                  aria-hidden="true"

   415→                  <p className="text-sm font-medium">
   416→                    No Job Match scans for this resume yet

   419→                    Scan a job description to see what it needs — then weave it in here.

   423→                    className="mt-3 inline-flex items-center gap-1.5 rounded-full border border-emerald-500/30 bg-emerald-500/10 px-3 py-1.5 text-xs font-semibold text-emerald-700 transition-colors hover:bg-emerald-500/20 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring dark:text-emerald-300"

   439→                        <li key={entry.id} className="relative list-none">
   440→                          {active ? (
   441→                            <span

   444→                            />

   448→                            onClick={() => setSelectedJdId(active ? null : entry.id)}
   449→                            aria-pressed={active}
   450→                            aria-label={`Tailor the letter to ${entry.jdBrief} — match score ${entry.score} of 100, scanned ${formatDistanceToNow(entry.at, { addSuffix: true })}`}
   451→                            className={cn(
   452→                              "flex w-full items-center gap-2.5 rounded-lg border text-left transition-all focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring",
   453→                              active
   454→                                ? "border-emerald-500/60 bg-emerald-500/10 py-2.5 pl-4 pr-10 ring-2 ring-emerald-500/20"
   455→                                : "border-border/70 py-2.5 pl-3 pr-3 hover:border-emerald-500/40 hover:bg-emerald-500/[0.03]"

   458→                            <span className="min-w-0 flex-1 truncate text-sm">
   459→                              {entry.jdBrief}
   460→                            </span>

   464→                                gradeBadgeClass(entry.score)

   466→                            >
   467→                              {entry.score}

   469→                            <span className="hidden w-20 shrink-0 text-right text-[11px] text-muted-foreground sm:inline">
   470→                              {formatDistanceToNow(entry.at, { addSuffix: true })}
   471→                            </span>
   472→                          </button>

   475→                              type="button"
   476→                              onClick={handleClearJd}

   479→                            >
   480→                              <X className="size-3.5" aria-hidden="true" />
   481→                            </button>
   482→                          ) : null}
   483→                        </li>
   484→                      );
   485→                    })}
   486→                  </ul>
   487→                  <p
   488→                    className="flex items-center gap-1.5 text-xs text-emerald-700 dark:text-emerald-300"
   489→                    role="status"
   490→                  >
   491→                    {selectedJd ? (

   493→                        <Sparkles className="size-3.5 shrink-0" aria-hidden="true" />
   494→                        The letter will weave in up to 5 keywords this JD needs.
   495→                      </>

   504→            </div>

   506→            <div className="sm:col-span-2">
