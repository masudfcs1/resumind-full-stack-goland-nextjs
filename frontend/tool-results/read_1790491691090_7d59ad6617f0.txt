    95→    95→  let t = text.trim().replace(/^(i\s+|we\s+)/i, "");
    96→    96→  const weak = WEAK_STARTERS.find((w) => t.toLowerCase().startsWith(w.toLowerCase()));

    99→    99→  const verb = pick(ACTION_VERBS);
   100→   100→  const metric = pick(METRICS);
   101→   101→  const capitalized = t.charAt(0).toUpperCase() + t.slice(1);
   102→   102→  const endsWithPunct = /[.!?]$/.test(t);
   103→   103→  return `${verb} ${t}${endsWithPunct ? "" : ","} ${metric}`;
   104→   104→}

   195→   195→function extractJobKeywords(jd: string): string[] {
   196→   196→  const stop = new Set(["the", "and", "for", "with", "you", "our", "are", "will", "that", "this", "have", "from", "your", "work", "team", "role", "job", "who", "all", "not", "but", "can", "has", "may", "per", "using", "use", "experience", "years", "including", "ability", "strong", "plus", "etc", "new", "other", "well", "across", "within", "about", "into", "more", "than", "also", "such", "their", "them", "they", "what", "when", "where", "which", "while", "would", "should", "could", "been", "being", "were", "does", "each", "some", "most", "must", "need", "required", "requirements", "preferred", "qualifications", "responsibilities", "company", "candidate", "candidates", "looking", "join", "help", "build", "help us"]);
   197→   197→  const words = jd.toLowerCase().match(/[a-z][a-z+#.\-]{2,}/g) ?? [];
   198→   198→  const freq = new Map<string, number>();
   199→   199→  for (const w of words) {
   200→   200→    if (stop.has(w)) continue;
   201→   201→    freq.set(w, (freq.get(w) ?? 0) + 1);
   202→   202→  }
   203→   203→  return [...freq.entries()]
   204→   204→    .filter(([, c]) => c >= 2)
   205→   205→    .sort((a, b) => b[1] - a[1])
   206→   206→    .slice(0, 18)
   207→   207→    .map(([w]) => w);
   208→   208→}

   210→   210→export function atsAnalyze(resume: ResumeData, jobDescription?: string): AtsResult {
   211→   211→  const text = resumeToText(resume);
   212→   212→  const lower = text.toLowerCase();
   213→   213→  const wordCount = text.split(/\s+/).filter(Boolean).length;
   214→   214→  const issues: AtsIssue[] = [];
   215→   215→
   216→   216→  /* --- Keywords (30) --- */
   217→   217→  const targetKeywords = jobDescription && jobDescription.trim().length > 40
   218→   218→    ? extractJobKeywords(jobDescription)
   219→   219→    : roleKeywordSet(resume);
   220→   220→  const matched: string[] = [];
   221→   221→  const missing: string[] = [];
   222→   222→  for (const kw of targetKeywords) {
   223→   223→    if (lower.includes(kw.toLowerCase())) matched.push(kw);
   224→   224→    else missing.push(kw);
   225→   225→  }
   226→   226→  const keywordScore = targetKeywords.length
   227→   227→    ? Math.round((matched.length / targetKeywords.length) * 30)
   228→   228→    : 18;
   229→   229→  if (missing.length > 6) {
   230→   230→    issues.push({
   231→   231→      id: "kw-missing",
   232→   232→      severity: "warning",
   233→   233→      category: "Keywords",
   234→   234→      title: `${missing.length} target keywords missing`,
   235→   235→      detail: `The job description mentions terms like ${missing.slice(0, 6).join(", ")} that don't appear in your resume.`,
   236→   236→      fix: "Weave the most relevant missing keywords naturally into your summary and experience bullets.",
   237→   237→    });
   238→   238→  } else {
   239→   239→    issues.push({
   240→   240→      id: "kw-good",
   241→   241→      severity: "good",
   242→   242→      category: "Keywords",
   243→   243→      title: "Strong keyword coverage",
   244→   244→      detail: `${matched.length} of ${targetKeywords.length} target keywords found in your resume.`,
   245→   245→    });
   246→   246→  }
   247→   247→
   248→   248→  /* --- Sections (20) --- */
   249→   249→  const sectionChecks: Array<[boolean, string, string]> = [
   250→   250→    [resume.summary.trim().length > 40, "Professional summary", "Add a 2–3 sentence summary at the top."],
   251→   251→    [resume.experience.length > 0, "Work experience", "Add at least one work experience entry."],
   252→   252→    [resume.education.length > 0, "Education", "Add your education history."],
   253→   253→    [resume.skills.length >= 5, "Skills (5+)", "List at least 5 relevant skills."],
   254→   254→  ];
   255→   255→  let sectionCount = 0;
   256→   256→  for (const [ok, label, fix] of sectionChecks) {
   257→   257→    if (ok) sectionCount++;
   258→   258→    else
   259→   259→      issues.push({
   260→   260→        id: `section-${label}`,
   261→   261→        severity: "critical",
   262→   262→        category: "Sections",
   263→   263→        title: `Missing section: ${label}`,
   264→   264→        detail: `ATS parsers expect a dedicated ${label} section.`,
   265→   265→        fix,
   266→   266→      });
   267→   267→  }
   268→   268→  const bonus = (resume.projects.length > 0 ? 1 : 0) + (resume.certifications.length > 0 ? 1 : 0);
   269→   269→  const sectionScore = Math.min(20, sectionCount * 5 + bonus);
   270→   270→  if (resume.projects.length === 0 && resume.certifications.length === 0) {
   271→   271→    issues.push({
   272→   272→      id: "section-bonus",
   273→   273→      severity: "info",
   274→   274→      category: "Sections",
   275→   275→      title: "Consider adding Projects or Certifications",

   277→   277→    });

   293→   293→      fix: 'Rewrite bullets like "Reduced page load time by 43%, cutting bounce rate 18%".',

   317→   317→    });
   318→   318→  }
   319→   319→  const hasBullets = resume.experience.some((e) => e.bullets.length > 0);
   320→   320→  if (!hasBullets) {
   321→   321→    formatScore -= 5;
   322→   322→    issues.push({
   323→   323→      id: "fmt-bullets",
   324→   324→      severity: "warning",
   325→   325→      category: "Formatting",
   326→   326→      title: "No bullet points detected",
   327→   327→      detail: "Bulleted achievements are easier for ATS to parse than paragraphs.",
   328→   328→      fix: "Break each role into 3–5 achievement bullets.",
   329→   329→    });
   330→   330→  }
   331→   331→  if (resume.personal.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(resume.personal.email)) {
   332→   332→    formatScore -= 3;
   333→   333→    issues.push({

   339→   339→      fix: "Use the format name@domain.com.",
   340→   340→    });
   341→   341→  }
   342→   342→  if (!resume.personal.phone && !resume.personal.email) {
   343→   343→    formatScore -= 4;
   344→   344→    issues.push({
   345→   345→      id: "fmt-contact",
   346→   346→      severity: "critical",
   347→   347→      category: "Formatting",
   348→   348→      title: "No contact information",
   349→   349→      detail: "Add at least an email address and phone number so recruiters can reach you.",
   350→   350→    });
   351→   351→  }
   352→   352→  if (formatScore === 15) {
   353→   353→    issues.push({
   354→   354→      id: "fmt-good",
   355→   355→      severity: "good",
   356→   356→      category: "Formatting",
   357→   357→      title: "Clean, parseable formatting",
   358→   358→      detail: "Dates, bullets, and contact details are all present and correctly formatted.",
   359→   359→    });
   360→   360→  }
   361→   361→
   362→   362→  /* --- Length (10) --- */
   363→   363→  let lengthScore: number;
   364→   364→  if (wordCount < 250) {
   365→   365→    lengthScore = 4;
   366→   366→    issues.push({
   367→   367→      id: "len-short",
   368→   368→      severity: "warning",
   369→   369→      category: "Length",
   370→   370→      title: "Resume is too short",
   371→   371→      detail: `${wordCount} words. Aim for 400–800 words to give ATS enough context.`,
   372→   372→    });
   373→   373→  } else if (wordCount > 900) {
   374→   374→    lengthScore = 6;
   375→   375→    issues.push({
   376→   376→      id: "len-long",
   377→   377→      severity: "info",
   378→   378→      category: "Length",
   379→   379→      title: "Resume is quite long",
   380→   380→      detail: `${wordCount} words. Trim older or less relevant roles to stay under ~800 words.`,
   381→   381→    });
   382→   382→  } else {
   383→   383→    lengthScore = 10;
   384→   384→    issues.push({
   385→   385→      id: "len-good",
   386→   386→      severity: "good",
   387→   387→      category: "Length",
   388→   388→      title: "Ideal resume length",
   389→   389→      detail: `${wordCount} words hits the 400–800 sweet spot for ATS readability.`,
   390→   390→    });
   391→   391→  }
   392→   392→
   393→   393→  const score = Math.max(0, Math.min(100, keywordScore + sectionScore + impactScore + actionScore + Math.max(0, formatScore) + lengthScore));
   394→   394→  const grade = score >= 85 ? "Excellent" : score >= 70 ? "Good" : score >= 50 ? "Fair" : "Needs Work";
   395→   395→
   396→   396→  return {
   397→   397→    score,
   398→   398→    grade,
   399→   399→    breakdown: [
   400→   400→      { label: "Keywords", score: keywordScore, max: 30, hint: "Match against target role keywords" },
   401→   401→      { label: "Sections", score: sectionScore, max: 20, hint: "Standard sections present" },
   402→   402→      { label: "Impact", score: impactScore + actionScore, max: 25, hint: "Quantified, action-driven content" },
   403→   403→      { label: "Formatting", score: Math.max(0, formatScore), max: 15, hint: "Dates, bullets, contact info" },
   404→   404→      { label: "Length", score: lengthScore, max: 10, hint: "Word count in the sweet spot" },
   405→   405→    ],
   406→   406→    issues: issues.sort((a, b) => {
   407→   407→      const order = { critical: 0, warning: 1, info: 2, good: 3 };
   408→   408→      return order[a.severity] - order[b.severity];
   409→   409→    }),
   410→   410→    matchedKeywords: matched.slice(0, 12),
   411→   411→    missingKeywords: missing.slice(0, 12),
   412→   412→    wordCount,
   413→   413→  };
   414→   414→}

   416→   416→/* ============================== Grammar Checker ============================== */
   417→   417→
   418→   418→export interface GrammarIssue {
   419→   419→  id: string;
   420→   420→  type: "spelling" | "grammar" | "style" | "punctuation" | "weak-phrase";
   421→   421→  original: string;
   422→   422→  suggestion: string;
   423→   423→  message: string;
   424→   424→}
   425→   425→
   426→   426→export interface GrammarResult {
   427→   427→  score: number;
   428→   428→  wordCount: number;
   429→   429→  sentenceCount: number;
   430→   430→  issues: GrammarIssue[];
   431→   431→  stats: { label: string; value: string }[];
   432→   432→}
   433→   433→
   434→   434→const MISSPELLINGS: Record<string, string> = {
   435→   435→  recieve: "receive", seperate: "separate", teh: "the", definately: "definitely",
   436→   436→  occured: "occurred", acheive: "achieve", responsable: "responsible",
   437→   437→  managment: "management", sucessful: "successful", adviced: "advised",
   438→   438→  independant: "independent", enviroment: "environment", developement: "development",
   439→   439→  commited: "committed", excellant: "excellent", experiance: "experience",
   440→   440→  knowlege: "knowledge", maintance: "maintenance", neccessary: "necessary",
   441→   441→  occassion: "occasion", posession: "possession", proffesional: "professional",
   442→   442→  recomend: "recommend", succesful: "successful", truely: "truly",
   443→   443→  untill: "until", wich: "which", wierd: "weird", leadeship: "leadership",
   444→   444→  buisness: "business", colaboration: "collaboration", acheivment: "achievement",
   445→   445→};
   446→   446→
   447→   447→const WEAK_PHRASES: Array<{ phrase: RegExp; suggestion: string; why: string }> = [
   448→   448→  { phrase: /\bresponsible for\b/gi, suggestion: "Owned / Led", why: "Weak opener — lead with a strong action verb instead" },
   449→   449→  { phrase: /\bworked on\b/gi, suggestion: "Developed / Built / Delivered", why: "Vague verb — describe what you actually did" },
   450→   450→  { phrase: /\bhelped (with|to)?\b/gi, suggestion: "Supported / Contributed to", why: "Understates your contribution" },
   451→   451→  { phrase: /\bduties included\b/gi, suggestion: "Drove / Managed", why: "Task-focused; reframe as achievements" },
   452→   452→  { phrase: /\bteam player\b/gi, suggestion: "Collaborated cross-functionally", why: "Cliché — show collaboration through results" },
   453→   453→  { phrase: /\bhard working\b|\bhard-working\b/gi, suggestion: "Results-driven", why: "Cliché — quantify your output instead" },
   454→   454→  { phrase: /\bvery\s+(\w+)/gi, suggestion: "stronger adjective", why: "'Very' weakens the word that follows" },
   455→   455→  { phrase: /\breally\s+(\w+)/gi, suggestion: "stronger adjective", why: "'Really' is filler — cut it" },
   456→   456→  { phrase: /\bjust\b/gi, suggestion: "(remove)", why: "Filler word that minimizes your impact" },
   457→   457→  { phrase: /\bin order to\b/gi, suggestion: "to", why: "Wordy — 'to' means the same thing" },
   458→   458→];
   459→   459→
   460→   460→const PASSIVE_RE = /\b(was|were|been|being)\s+\w+(ed|en)\b/gi;
   461→   461→
   462→   462→export function grammarCheckText(text: string): GrammarResult {
   463→   463→  const issues: GrammarIssue[] = [];
   464→   464→  const wordCount = text.split(/\s+/).filter(Boolean).length;
   465→   465→  const sentenceCount = (text.match(/[.!?]+(\s|$)/g) ?? []).length || 1;
   466→   466→
   467→   467→  /* spelling */
   468→   468→  const words = text.match(/[A-Za-z][A-Za-z']*/g) ?? [];
   469→   469→  const seen = new Set<string>();
   470→   470→  for (const w of words) {
   471→   471→    const lw = w.toLowerCase();
   472→   472→    if (MISSPELLINGS[lw] && !seen.has(lw)) {
   473→   473→      seen.add(lw);
   474→   474→      issues.push({
   475→   475→        id: `sp-${lw}`,
   476→   476→        type: "spelling",
   477→   477→        original: w,
   478→   478→        suggestion: MISSPELLINGS[lw],
   479→   479→        message: `Possible misspelling — did you mean "${MISSPELLINGS[lw]}"?`,
   480→   480→      });
   481→   481→    }
   482→   482→  }
   483→   483→
   484→   484→  /* lowercase standalone i */
   485→   485→  const loneI = text.match(/(^|\s)i(\s|,|'|\.)/g);
   486→   486→  if (loneI) {
   487→   487→    issues.push({
   488→   488→      id: "gr-lower-i",
   489→   489→      type: "grammar",
   490→   490→      original: "i",
   491→   491→      suggestion: "I",
   492→   492→      message: 'The pronoun "I" should always be capitalized.',
   493→   493→    });
   494→   494→  }
   495→   495→
   496→   496→  /* double spaces */
   497→   497→  if (/ {2,}/.test(text)) {
   498→   498→    issues.push({
   499→   499→      id: "pu-double-space",
   500→   500→      type: "punctuation",
   501→   501→      original: "double spaces",
   502→   502→      suggestion: "single space",
   503→   503→      message: "Multiple consecutive spaces found — use a single space.",
   504→   504→    });
   505→   505→  }
   506→   506→
   507→   507→  /* repeated words */
   508→   508→  const repeated = text.match(/\b(\w+)\s+\1\b/gi);
   509→   509→  if (repeated) {
   510→   510→    for (const r of repeated.slice(0, 3)) {
   511→   511→      const w = r.trim().split(/\s+/)[0];
   512→   512→      if (w.length > 1) {
   513→   513→        issues.push({
   514→   514→          id: `gr-repeat-${w}`,
   515→   515→          type: "grammar",
   516→   516→          original: r.trim(),
   517→   517→          suggestion: w,
   518→   518→          message: `"${w}" is repeated — remove the duplicate.`,
   519→   519→        });
   520→   520→      }
   521→   521→    }
   522→   522→  }
   523→   523→
   524→   524→  /* passive voice */
   525→   525→  const passives = text.match(PASSIVE_RE) ?? [];
   526→   526→  if (passives.length > 0) {
   527→   527→    issues.push({
   528→   528→      id: "st-passive",
   529→   529→      type: "style",
   530→   530→      original: passives[0],
   531→   531→      suggestion: "active voice",
   532→   532→      message: `${passives.length} passive construction${passives.length > 1 ? "s" : ""} found. Active voice is stronger: "Led the migration" vs "The migration was led".`,
   533→   533→    });
   534→   534→  }
   535→   535→
   536→   536→  /* weak phrases */
   537→   537→  for (const { phrase, suggestion, why } of WEAK_PHRASES) {
   538→   538→    const m = text.match(phrase);
   539→   539→    if (m) {
   540→   540→      issues.push({
   541→   541→        id: `wk-${suggestion.replace(/\W/g, "")}`,
   542→   542→        type: "weak-phrase",
   543→   543→        original: m[0],
   544→   544→        suggestion,
   545→   545→        message: why,
   546→   546→      });
   547→   547→    }
   548→   548→  }
   549→   549→
   550→   550→  /* missing capitalization at sentence start */
   551→   551→  const sentences = text.split(/[.!?]\s+/).filter((s) => s.trim().length > 3);
   552→   552→  const lowerStarts = sentences.filter((s) => /^[a-z]/.test(s.trim())).length;
   553→   553→  if (lowerStarts > 0) {
   554→   554→    issues.push({
   555→   555→      id: "pu-sentence-case",
   556→   556→      type: "punctuation",
   557→   557→      original: "sentence start",
   558→   558→      suggestion: "Capital letter",
   559→   559→      message: `${lowerStarts} sentence${lowerStarts > 1 ? "s start" : " starts"} with a lowercase letter.`,
   560→   560→    });
   561→   561→  }
   562→   562→
   563→   563→  const score = Math.max(40, Math.min(100, 100 - issues.length * 7 - (passives.length > 2 ? 5 : 0)));
   564→   564→
   565→   565→  const avgWords = Math.round(wordCount / sentenceCount);
   566→   566→  return {
   567→   567→    score,
   568→   568→    wordCount,
   569→   569→    sentenceCount,
   570→   570→    issues,
   571→   571→    stats: [
   572→   572→      { label: "Words", value: String(wordCount) },
   573→   573→      { label: "Sentences", value: String(sentenceCount) },
   574→   574→      { label: "Avg. words / sentence", value: String(avgWords) },
   575→   575→      { label: "Readability", value: avgWords > 25 ? "Complex" : avgWords > 18 ? "Moderate" : "Easy to read" },
   576→   576→    ],
   577→   577→  };
   578→   578→}

   657→   657→export interface LinkedInParsed {
   658→   658→  name: string;
   659→   659→  headline: string;
   660→   660→  location: string;
   661→   661→  summary: string;
   662→   662→  experience: Array<{ company: string; role: string; period: string; bullets: string[] }>;
   663→   663→  education: Array<{ school: string; degree: string; period: string }>;
   664→   664→  skills: string[];
   665→   665→  confidence: number;
   666→   666→}
   667→   667→
   668→   668→export function parseLinkedInProfile(text: string): LinkedInParsed {
   669→   669→  const lines = text
   670→   670→    .split("\n")
   671→   671→    .map((l) => l.trim())
   672→   672→    .filter(Boolean);
   673→   673→
   674→   674→  const result: LinkedInParsed = {
   675→   675→    name: "", headline: "", location: "", summary: "",
   676→   676→    experience: [], education: [], skills: [], confidence: 0,
   677→   677→  };
   678→   678→
   679→   679→  let section = "header";
   680→   680→  let currentExp: LinkedInParsed["experience"][number] | null = null;
   681→   681→  const lowerSections = new Set(["experience", "education", "skills", "about", "summary", "licenses & certifications"]);
   682→   682→
   683→   683→  for (const line of lines) {
   684→   684→    const l = line.toLowerCase().replace(/:$/, "").trim();
   685→   685→    if (lowerSections.has(l)) {
   686→   686→      section = l === "summary" ? "about" : l;
   687→   687→      continue;
   688→   688→    }
   689→   689→    if (section === "header") {
   690→   690→      if (!result.name) result.name = line;
   691→   691→      else if (!result.headline) result.headline = line;
   692→   692→      else if (!result.location && /^(greater |)[A-Z]/.test(line) && line.split(",").length >= 2) result.location = line;
   693→   693→    } else if (section === "about") {
   694→   694→      result.summary += (result.summary ? " " : "") + line;
   695→   695→    } else if (section === "experience") {
   696→   696→      const atMatch = line.match(/^(.{2,60}?)\s+at\s+(.+)$/i);
   697→   697→      const dateMatch = /^(Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec|[A-Za-z]+ \d{4}).*\d{4}/i.test(line) || /^\d{4}\s*[-–]\s*/.test(line);
   698→   698→      if (atMatch) {
   699→   699→        if (currentExp) result.experience.push(currentExp);
   700→   700→        currentExp = { role: atMatch[1], company: atMatch[2], period: "", bullets: [] };
   701→   701→      } else if (dateMatch && currentExp) {
   702→   702→        currentExp.period = line;
   703→   703→      } else if (currentExp && (line.startsWith("•") || line.startsWith("-") || line.startsWith("· ") || line.length > 40)) {
   704→   704→        currentExp.bullets.push(line.replace(/^[•\-·]\s*/, ""));
   705→   705→      }
   706→   706→    } else if (section === "education") {
   707→   707→      const m = line.match(/^(.+?),\s*(B\.?S\.?|B\.?A\.?|M\.?S\.?|M\.?B\.?A\.?|Ph\.?D\.?|Bachelor|Master|Doctor)/i);
   708→   708→      if (m) result.education.push({ school: m[1], degree: m[2], period: "" });
   709→   709→      else if (result.education.length && !result.education[result.education.length - 1].period && /\d{4}/.test(line)) {
   710→   710→        result.education[result.education.length - 1].period = line;
   711→   711→      }
   712→   712→    } else if (section === "skills") {
   713→   713→      for (const s of line.split(/[,•·|]/)) {
   714→   714→        const t = s.trim();
   715→   715→        if (t && t.length < 40 && !result.skills.includes(t)) result.skills.push(t);
   716→   716→      }
   717→   717→    }
   718→   718→  }
   719→   719→  if (currentExp) result.experience.push(currentExp);
   720→   720→
   721→   721→  let filled = 0;
   722→   722→  const checks = [result.name, result.headline, result.summary, result.experience.length, result.education.length, result.skills.length];
   723→   723→  for (const c of checks) if (c) filled++;
   724→   724→  result.confidence = Math.round((filled / checks.length) * 100);
   725→   725→  return result;
   726→   726→}
