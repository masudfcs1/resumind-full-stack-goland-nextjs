   422→/* "+12 pts" pill that counts up when scrolled into view */
   423→function DeltaPill() {
   424→  const ref = React.useRef<HTMLSpanElement>(null);
   425→  const inView = useInView(ref, { once: true, margin: "-60px" });
   426→  const mv = useMotionValue(0);
   427→  const text = useTransform(mv, (v: number) => `+${Math.round(v)} pts`);
   428→
   429→  React.useEffect(() => {
   430→    if (!inView) return;
   431→    const controls = animate(mv, 12, { duration: 1.4, ease: "easeOut", delay: 0.5 });
   432→    return () => controls.stop();
   433→  }, [inView, mv]);
   434→
   435→  return (
   436→    <span
   437→      ref={ref}
   438→      className="absolute -bottom-3 left-1/2 z-10 inline-flex -translate-x-1/2 items-center gap-1 rounded-full border border-emerald-500/30 bg-background/90 px-2.5 py-1 text-[10px] font-extrabold text-emerald-600 shadow-sm backdrop-blur-sm dark:border-emerald-400/25 dark:text-emerald-400"
   439→    >
   440→      <TrendingUp className="size-3" aria-hidden />
   441→      <motion.span className="tabular-nums">{text}</motion.span>
   442→    </span>
   443→  );
   444→}
