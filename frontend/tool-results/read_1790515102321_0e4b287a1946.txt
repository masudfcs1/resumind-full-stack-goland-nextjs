   122→   122→function parseBackupText(text: string): { ok: true; data: ParsedBackup } | { ok: false; reason: string } {
   123→   123→  let parsed: unknown;
   124→   124→  try {
   125→   125→    parsed = JSON.parse(text);
   126→   126→  } catch {
   127→   127→    return { ok: false, reason: "That file isn't valid JSON. Export a fresh backup from Settings and try again." };
   128→   128→  }
   129→   129→  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
   130→   130→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — expected a JSON object." };
   131→   131→  }
   132→   132→  const obj = parsed as Record<string, unknown>;
   133→   133→  const rawResumes = Array.isArray(obj.resumes) ? obj.resumes : null;
   134→   134→  if (obj.app !== "ResumeForge AI" && !rawResumes) {
   135→   135→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — the \"app\" marker and resumes array are missing." };
   136→   136→  }
   137→   137→  if (!rawResumes || rawResumes.length === 0) {
   138→   138→    return { ok: false, reason: "The backup contains no resumes, so there is nothing to restore." };
   139→   139→  }
   140→   140→  for (let i = 0; i < rawResumes.length; i++) {
   141→   141→    const entry: unknown = rawResumes[i];
   142→   142→    if (!entry || typeof entry !== "object" || Array.isArray(entry)) {
   143→   143→      return { ok: false, reason: `Resume #${i + 1} in the backup is malformed (expected an object).` };
   144→   144→    }
   145→   145→    const rec = entry as Record<string, unknown>;
   146→   146→    const titleOk = typeof rec.title === "string";
   147→   147→    if (
   148→   148→      typeof rec.id !== "string" ||
   149→   149→      rec.id.trim() === "" ||
   150→   150→      !titleOk ||
   151→   151→      !rec.personal ||
   152→   152→      typeof rec.personal !== "object" ||
   153→   153→      Array.isArray(rec.personal)
   154→   154→    ) {
   155→   155→      const label = titleOk ? `"${rec.title}"` : `#${i + 1}`;
   156→   156→      return {
   157→   157→        ok: false,
   158→   158→        reason: `Resume ${label} is missing an id, a title or personal details — the backup looks corrupted.`,
   159→   159→      };
   160→   160→    }
   161→   161→  }
   162→   162→  /* Format detection (v1 vs v2): a 2.x version string or a meta object marks a v2 backup.
   163→   163→     Everything else — including versionless legacy exports — is treated as v1. */
   164→   164→  const rawMeta: unknown = obj.meta;
   165→   165→  const hasMeta = !!rawMeta && typeof rawMeta === "object" && !Array.isArray(rawMeta);
   166→   166→  const versionIs2 = typeof obj.version === "string" && obj.version.trim().startsWith("2");
   167→   167→  const formatVersion: "v1" | "v2" = versionIs2 || hasMeta ? "v2" : "v1";
   168→   168→
   169→   169→  /* v2 extras: meta.favorites.interview is an optional array of company keys.
   170→   170→     Normalized the same way readFavorites() stores them (trim/lowercase/dedupe). */
   171→   171→  let interviewFavorites: string[] | null = null;
   172→   172→  if (formatVersion === "v2" && hasMeta) {
   173→   173→    const favorites: unknown = (rawMeta as Record<string, unknown>).favorites;
   174→   174→    if (favorites && typeof favorites === "object" && !Array.isArray(favorites)) {
   175→   175→      const interview: unknown = (favorites as Record<string, unknown>).interview;
   176→   176→      if (Array.isArray(interview)) {
   177→   177→        interviewFavorites = interview
   178→   178→          .filter((v): v is string => typeof v === "string")
   179→   179→          .map((v) => v.trim().toLowerCase())
   180→   180→          .filter((v) => v.length > 0)
   181→   181→          .filter((v, i, arr) => arr.indexOf(v) === i); // dedupe, keep first
   182→   182→      }
   183→   183→    }
   184→   184→  }
   185→   185→
   186→   186→  return {
   187→   187→    ok: true,
   188→   188→    data: {
   189→   189→      payload: obj,
   190→   190→      resumeCount: rawResumes.length,
   191→   191→      applicationCount: Array.isArray(obj.applications) ? obj.applications.length : null,
   192→   192→      scoreHistoryCount: Array.isArray(obj.scoreHistory) ? obj.scoreHistory.length : null,
   193→   193→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : null,
   194→   194→      formatVersion,
   195→   195→      interviewFavorites,
   196→   196→    },
   197→   197→  };
   198→   198→}
