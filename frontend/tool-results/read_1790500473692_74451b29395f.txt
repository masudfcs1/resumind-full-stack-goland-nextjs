   134→function buildNotifications(
   135→  scoreHistory: ScoreEntry[],
   136→  applications: ReturnType<typeof useResumeStore.getState>["applications"],
   137→  resumes: ReturnType<typeof useResumeStore.getState>["resumes"]
   138→): NotificationItem[] {
   139→  const items: NotificationItem[] = [];
   140→  const now = Date.now();
   141→  const day = 1000 * 60 * 60 * 24;
   142→
   143→  /* Latest ATS scan per resume, with delta vs the previous scan */
   144→  const byResume = new Map<string, ScoreEntry[]>();
   145→  for (const e of scoreHistory) {
   146→    const list = byResume.get(e.resumeId) ?? [];
   147→    list.push(e);
   148→    byResume.set(e.resumeId, list);
   149→  }
   150→  for (const list of byResume.values()) {
   151→    const sorted = [...list].sort((a, b) => a.at - b.at);
   152→    const latest = sorted[sorted.length - 1];
   153→    const prev = sorted[sorted.length - 2];
   154→    const delta = prev ? latest.score - prev.score : 0;
   155→    items.push({
   156→      id: `score-${latest.id}`,
   157→      kind: delta > 0 ? "score-up" : delta < 0 ? "score-down" : "resume",
   158→      title: `ATS scan — ${latest.score}/100`,
   159→      body:
   160→        delta === 0
   161→          ? `${latest.resumeTitle} · no change since your last scan.`
   162→          : `${latest.resumeTitle} ${delta > 0 ? "improved" : "dropped"} ${Math.abs(delta)} pts since your last scan.`,
   163→      at: latest.at,
   164→      href: "/dashboard/ats",
   165→    });
   166→  }
   167→
   168→  /* Pipeline events updated in the last 7 days */
   169→  for (const app of applications) {
   170→    if (now - app.updatedAt > day * 7) continue;
   171→    if (app.stage === "offer") {
   172→      items.push({
   173→        id: `app-${app.id}`,
   174→        kind: "offer",
   175→        title: `Offer from ${app.company}`,
   176→        body: `${app.role} — time to celebrate (and negotiate).`,
   177→        at: app.updatedAt,
   178→        href: "/dashboard/tracker",
   179→      });
   180→    } else if (app.stage === "interview") {
   181→      items.push({
   182→        id: `app-${app.id}`,
   183→        kind: "interview",
   184→        title: `Interview at ${app.company}`,
   185→        body: `${app.role} — drill the question bank before the call.`,
   186→        at: app.updatedAt,
   187→        href: "/dashboard/interview",
   188→      });
   189→    } else if (app.stage === "rejected") {
   190→      items.push({
   191→        id: `app-${app.id}`,
   192→        kind: "rejected",
   193→        title: `${app.company} passed on ${app.role}`,
   194→        body: "Closed — keep the pipeline moving, momentum matters.",
   195→        at: app.updatedAt,
   196→        href: "/dashboard/tracker",
   197→      });
   198→    }
   199→  }
   200→
   201→  /* Resume edits from the last 3 days */
   202→  for (const r of resumes) {
   203→    if (now - r.updatedAt > day * 3) continue;
   204→    items.push({
   205→      id: `resume-${r.id}-${r.updatedAt}`,
   206→      kind: "resume",
   207→      title: "Resume saved",
   208→      body: `${r.title} — all changes are stored safely.`,
   209→      at: r.updatedAt,
   210→      href: "/dashboard/builder",
   211→    });
   212→  }
   213→
   214→  return items.sort((a, b) => b.at - a.at).slice(0, 8);
   215→}
