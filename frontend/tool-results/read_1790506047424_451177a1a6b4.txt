   110→            onClick={() => onChange(option.value)}
   111→            className={cn(
   112→              "rounded-md px-3 py-1.5 text-xs font-medium transition-all",
   113→              active
   114→                ? "bg-background text-foreground shadow-sm ring-1 ring-border"

   122→    </div>

   126→function CategoryChips({
   127→  selected,
   128→  onToggle,
   129→}: {
   130→  selected: QuestionCategory[];

   132→}) {

   134→    <div className="flex flex-wrap gap-2" role="group" aria-label="Question categories">

   140→          <button
   141→            key={cat}

   146→              "inline-flex items-center gap-2 rounded-full border px-3.5 py-2 text-sm font-medium transition-all",
   147→              active
   148→                ? cn(meta.tint, "shadow-sm")

   151→          >
   152→            <Icon className="size-4" aria-hidden="true" />
   153→            {meta.label}
   154→            {active ? (
   155→              <Check className="size-3.5" aria-hidden="true" />
   156→            ) : null}
   157→          </button>
   158→        );
   159→      })}
   160→    </div>
   161→  );

   230→export default function InterviewPage() {
   231→  const mounted = useMounted();
   232→  const resumes = useResumeStore((s) => s.resumes);
   233→  const activeResumeId = useResumeStore((s) => s.activeResumeId);
   234→  const setActive = useResumeStore((s) => s.setActive);
   235→  const applications = useResumeStore((s) => s.applications);
   236→
   237→  const resume = resumes.find((r) => r.id === activeResumeId) ?? resumes[0];
   238→
   239→  const [selectedCategories, setSelectedCategories] =
   240→    React.useState<QuestionCategory[]>(CATEGORY_ORDER);
   241→  const [difficulty, setDifficulty] = React.useState<DifficultyFilter>("all");
   242→  const [generating, setGenerating] = React.useState(false);
   243→  const [session, setSession] = React.useState<PracticeQuestion[] | null>(null);
   244→  const [generation, setGeneration] = React.useState(0);
   245→  const [practicedIds, setPracticedIds] = React.useState<Set<string>>(new Set());
   246→  const [focusOpen, setFocusOpen] = React.useState(false);
   247→  const [focusIndex, setFocusIndex] = React.useState(0);
   248→  const [starNotes, setStarNotes] = React.useState<Record<string, StarAnswer>>({});
   249→  const [sessionOrigin, setSessionOrigin] = React.useState<SessionOrigin | null>(null);
   250→  const [companyVariant, setCompanyVariant] = React.useState(0);
   251→  const timerRef = React.useRef<number | null>(null);
   252→
   253→  React.useEffect(() => {
   254→    return () => {
   255→      if (timerRef.current !== null) window.clearTimeout(timerRef.current);
   256→    };
   257→  }, []);
   258→
   259→  const generate = React.useCallback(() => {
   260→    if (!resume || generating) return;
   261→    setGenerating(true);
   262→    const next = generateQuestions(resume, {
   263→      categories: selectedCategories,
   264→      difficulty,
   265→      count: QUESTIONS_PER_SET,
   266→    });
   267→    timerRef.current = window.setTimeout(() => {
   268→      if (next.length === 0) {
   269→        toast.warning("No questions match those filters", {
   270→          description: "Try selecting more categories or a different difficulty.",
   271→        });
   272→        setGenerating(false);
   273→        return;
   274→      }
   275→      setSession(next);
   276→      setPracticedIds(new Set());
   277→      setFocusIndex(0);
   278→      setGeneration((g) => g + 1);
   279→      setSessionOrigin({ kind: "bank" });
   280→      setCompanyVariant(0);
   281→      setGenerating(false);
   282→      toast.success(`Question set ready — ${next.length} questions`, {
   283→        description: `Tailored from "${resume.title}". Flip a card to reveal the STAR framework.`,
   284→      });
   285→    }, GENERATE_DELAY_MS);
   286→  }, [resume, generating, selectedCategories, difficulty]);
   287→
   288→  /** Starts a company-specific session from a tracked application. */
   289→  const startCompanySession = React.useCallback(
   290→    (entry: CompanyEntry) => {
   291→      if (generating) return;
   292→      setGenerating(true);
   293→      setCompanyVariant(0);
   294→      const next = buildCompanyQuestions(entry.company, entry.role, entry.stage, {
   295→        resume,
   296→        variant: 0,
   297→      });
   298→      timerRef.current = window.setTimeout(() => {
   299→        setSession(next);
   300→        setPracticedIds(new Set());
   301→        setFocusIndex(0);
   302→        setGeneration((g) => g + 1);
   303→        setSessionOrigin({
   304→          kind: "company",
   305→          company: entry.company,
   306→          role: entry.role,
   307→          stage: entry.stage,
   308→        });
   309→        setGenerating(false);
   310→        toast.success(`${entry.company} set ready — ${next.length} questions`, {
   311→          description: `Built from your tracker · ${STAGE_META[entry.stage].label} stage.`,
   312→        });
   313→      }, COMPANY_GENERATE_DELAY_MS);
   314→    },
   315→    [generating, resume]
   316→  );
   317→
   318→  /** Re-rolls the pulled questions of the active company set (6 fixed ones stay). */
   319→  const reshuffleCompany = React.useCallback(() => {
   320→    if (!sessionOrigin || sessionOrigin.kind !== "company" || generating) return;
   321→    const variant = companyVariant + 1;
   322→    setGenerating(true);
   323→    setCompanyVariant(variant);
   324→    const next = buildCompanyQuestions(
   325→      sessionOrigin.company,
   326→      sessionOrigin.role,
   327→      sessionOrigin.stage,
   328→      { resume, variant }
   329→    );
   330→    timerRef.current = window.setTimeout(() => {
   331→      setSession(next);
   332→      setPracticedIds(new Set());
   333→      setFocusIndex(0);
   334→      setGeneration((g) => g + 1);
   335→      setGenerating(false);
   336→      toast.success(`${sessionOrigin.company} set reshuffled — ${next.length} questions`);
   337→    }, COMPANY_GENERATE_DELAY_MS);
   338→  }, [sessionOrigin, generating, companyVariant, resume]);
   339→
   340→  /** Unique companies from the tracker — latest application defines stage/role. */
   341→  const companySets = React.useMemo<CompanyEntry[]>(() => {
   342→    const byKey = new Map<string, { entry: CompanyEntry; latest: number }>();
   343→    for (const app of applications) {
   344→      const key = app.company.trim().toLowerCase();
   345→      if (!key) continue;
   346→      const existing = byKey.get(key);
   347→      if (existing) {
   348→        existing.entry.count += 1;
   349→        if (app.updatedAt > existing.latest) {
   350→          existing.latest = app.updatedAt;
   351→          existing.entry.company = app.company;
   352→          existing.entry.role = app.role;
   353→          existing.entry.stage = app.stage;
   354→        }
   355→      } else {
   356→        byKey.set(key, {
   357→          entry: { key, company: app.company, role: app.role, stage: app.stage, count: 1 },
   358→          latest: app.updatedAt,
   359→        });
   360→      }
   361→    }
   362→    return [...byKey.values()]
   363→      .sort((a, b) => b.latest - a.latest)
   364→      .map((v) => v.entry);
   365→  }, [applications]);
   366→
   367→  const toggleCategory = React.useCallback((cat: QuestionCategory) => {
   368→    setSelectedCategories((prev) =>
   369→      prev.includes(cat) ? prev.filter((c) => c !== cat) : [...prev, cat]
   370→    );
   371→  }, []);
   372→
   373→  /** Marks a card practiced — each unique card counts once. */
   374→  const markPracticed = React.useCallback(
   375→    (id: string) => {
   376→      setPracticedIds((prev) => {
   377→        if (prev.has(id)) return prev;
   378→        const next = new Set(prev);
   379→        next.add(id);
   380→        if (session && next.size === session.length) {
   381→          toast.success("Session complete — every question practiced. Nice work.");
   382→        }
   383→        return next;
   384→      });
   385→    },
   386→    [session]
   387→  );
   388→
   389→  const handleNotesChange = React.useCallback((id: string, note: StarAnswer) => {
   390→    setStarNotes((prev) => ({ ...prev, [id]: note }));
   391→  }, []);
   392→
   393→  const openFocus = (index: number) => {
   394→    setFocusIndex(index);
   395→    setFocusOpen(true);
   396→  };
   397→
   398→  const practicedCount = practicedIds.size;
   399→  const totalCount = session?.length ?? 0;
   400→  const pct = totalCount > 0 ? Math.round((practicedCount / totalCount) * 100) : 0;
   401→  const tailoredCount = session?.filter((q) => q.isTailored).length ?? 0;
   402→
   403→  /* ============================== Not hydrated yet ============================== */
   404→
   405→  if (!mounted) {
   406→    return (
   407→      <div className="mx-auto flex w-full max-w-6xl flex-col gap-6" aria-busy="true">
   408→        <div className="flex items-start gap-4">
   409→          <Skeleton className="size-12 rounded-xl" />
   410→          <div className="space-y-2">
   411→            <Skeleton className="h-7 w-56" />
   412→            <Skeleton className="h-4 w-96 max-w-full" />
   413→          </div>
   414→        </div>
   415→        <Skeleton className="h-44 w-full rounded-xl" />
   416→        <div className="grid grid-cols-1 gap-5 sm:grid-cols-2 xl:grid-cols-3">
   417→          {Array.from({ length: 6 }).map((_, i) => (
   418→            <Skeleton key={i} className="h-[350px] rounded-2xl" />
   419→          ))}
   420→        </div>
   421→      </div>
   422→    );
   423→  }
   424→
   425→  /* ============================== Render ============================== */
   426→
   427→  return (
   428→    <motion.div
   429→      variants={staggerContainer}
   430→      initial="hidden"
   431→      animate="show"
   432→      className="mx-auto flex w-full max-w-6xl flex-col gap-6"
   433→    >
   434→      {/* ============ Header ============ */}
   435→      <ToolPageHeader
   436→        icon={MessagesSquare}
   437→        title="Interview Prep"
   438→        description="AI-simulated interview training tailored to your resume. Practice with flashcards, study the STAR model answers, and walk in ready."
   439→        actions={
   440→          <>
   441→            <ResumeSelect
   442→              value={resume?.id ?? ""}
   443→              onValueChange={setActive}
   444→              ariaLabel="Resume to tailor questions from"
   445→              className="w-full sm:w-60"
   446→            />
   447→            <DifficultySegmentedControl value={difficulty} onChange={setDifficulty} />
   448→          </>
   449→        }
   450→      />
   451→
   452→      {/* ============ Setup card ============ */}
   453→      <motion.section variants={staggerItem} aria-label="Session setup">
   454→        <Card>
   455→          <CardHeader className="pb-4">
   456→            <CardTitle className="font-display text-base">
   457→              Build your question set
   458→            </CardTitle>
   459→            <CardDescription>
   460→              Pick categories — the generator mixes difficulties and prefers
   461→              questions customized from your resume.
   462→            </CardDescription>
   463→          </CardHeader>
   464→          <CardContent className="space-y-4">
   465→            <CategoryChips
   466→              selected={selectedCategories}
   467→              onToggle={toggleCategory}
   468→            />
   469→
   470→            <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   471→              <Sparkles className="size-3.5 text-violet-500" aria-hidden="true" />
   472→              Sparkle-marked cards reference your most recent role, top skills, and
   473→              projects.
   474→            </p>
   475→
   476→            <div className="flex flex-wrap items-center justify-between gap-3 border-t pt-4">
   477→              <p className="text-sm text-muted-foreground">
   478→                <span className="font-semibold text-foreground">
   479→                  {selectedCategories.length}/5
   480→                </span>{" "}
   481→                categories ·{" "}
   482→                <span className="font-semibold text-foreground">
   483→                  {DIFFICULTY_META[difficulty]?.label ?? "All"}
   484→                </span>{" "}
   485→                difficulty ·{" "}
   486→                <span className="font-semibold text-foreground">
   487→                  {QUESTIONS_PER_SET} questions
   488→                </span>{" "}
   489→                per set
   490→              </p>
   491→              <Button
   492→                onClick={generate}
   493→                disabled={generating || !resume || selectedCategories.length === 0}
   494→                className="bg-gradient-to-r from-emerald-500 to-teal-600 text-white shadow-lg shadow-emerald-500/25 hover:from-emerald-600 hover:to-teal-700"
   495→                aria-label="Generate question set"
   496→              >
   497→                {generating ? (
   498→                  <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   499→                ) : (
   500→                  <Wand2 className="size-4" aria-hidden="true" />
   501→                )}
   502→                {generating ? "Generating…" : "Generate question set"}
   503→              </Button>
   504→            </div>
   505→          </CardContent>
   506→        </Card>
   507→      </motion.section>
   508→
   509→      {/* ============ From your tracker ============ */}
   510→      <motion.section variants={staggerItem} aria-labelledby="tracker-sets-heading">
   511→        <Card>
   512→          <CardHeader className="pb-4">
   513→            <div className="flex items-start gap-3">
   514→              <span
   515→                aria-hidden="true"
   516→                className="flex size-9 shrink-0 items-center justify-center rounded-lg bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   517→              >
   518→                <Building2 className="size-4" />
   519→              </span>
   520→              <div className="space-y-1">
   521→                <p className="text-[10px] font-semibold uppercase tracking-[0.14em] text-emerald-600 dark:text-emerald-400">
   522→                  From your tracker
   523→                </p>
   524→                <CardTitle className="font-display text-base" id="tracker-sets-heading">
   525→                  Company-specific question sets
   526→                </CardTitle>
   527→              </div>
   528→            </div>
   529→            <CardDescription className="mt-1">
   530→              Sets built from the companies and roles you are actually pursuing — pick one to
   531→              start practicing.
   532→            </CardDescription>
   533→          </CardHeader>
   534→          <CardContent>
   535→            {companySets.length === 0 ? (
   536→              <div className="flex flex-col items-start gap-4 rounded-xl border border-dashed bg-muted/30 p-4 sm:flex-row sm:items-center sm:justify-between">
   537→                <div className="flex items-center gap-3">
   538→                  <span
   539→                    aria-hidden="true"
   540→                    className="flex size-10 shrink-0 items-center justify-center rounded-full bg-gradient-to-br from-emerald-500 to-teal-600 text-white shadow-sm"
   541→                  >
   542→                    <Briefcase className="size-5" />
   543→                  </span>
   544→                  <div>
   545→                    <p className="text-sm font-medium">No tracked applications yet</p>
   546→                    <p className="text-xs text-muted-foreground">
   547→                      Track your first application to unlock company sets.
   548→                    </p>
   549→                  </div>
   550→                </div>
   551→                <Button asChild variant="outline" size="sm">
   552→                  <Link href="/dashboard/tracker">
   553→                    Go to Job Tracker
   554→                    <ArrowRight className="size-3.5" aria-hidden="true" />
   555→                  </Link>
   556→                </Button>
   557→              </div>
   558→            ) : (
   559→              <motion.div
   560→                variants={staggerContainer}
   561→                initial="hidden"
   562→                animate="show"
   563→                className="flex flex-wrap gap-2.5"
   564→                role="group"
   565→                aria-label="Companies from your tracker"
   566→              >
   567→                {companySets.map((c) => {
   568→                  const active =
   569→                    sessionOrigin?.kind === "company" &&
   570→                    sessionOrigin.company.trim().toLowerCase() === c.key;
   571→                  return (
   572→                    <motion.div key={c.key} variants={staggerItem}>
   573→                      <button
   574→                        type="button"
   575→                        aria-pressed={active}
   576→                        onClick={() => startCompanySession(c)}
   577→                        className={cn(
   578→                          "flex items-center gap-2.5 rounded-full border py-1.5 pl-1.5 pr-3.5 text-left transition-all duration-200",
