   457→export default function MyResumesPage() {
   458→  const mounted = useMounted();
   459→  const resumes = useResumeStore((s) => s.resumes);
   460→  const activeResumeId = useResumeStore((s) => s.activeResumeId);
   461→  const active = useActiveResume();
   462→  const duplicateResume = useResumeStore((s) => s.duplicateResume);
   463→  const deleteResume = useResumeStore((s) => s.deleteResume);
   464→  const logScore = useResumeStore((s) => s.logScore);
   465→  const scoreHistory = useResumeStore((s) => s.scoreHistory);
   466→
   467→  const [query, setQuery] = React.useState("");
   468→  const [sort, setSort] = React.useState<SortKey>("recent");
   469→  const [tab, setTab] = React.useState<ResumeTab>("active");
   470→  const [healthFilter, setHealthFilter] = React.useState<HealthFilterKey>("all");
   471→  const [scanningStale, setScanningStale] = React.useState(false);
   472→  const [view, setView] = React.useState<ViewMode>("grid");
   473→  const [createOpen, setCreateOpen] = React.useState(false);
   474→  const [selectMode, setSelectMode] = React.useState(false);
   475→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   476→  const [confirmDelete, setConfirmDelete] = React.useState(false);
   477→
   478→  const scoreMap = React.useMemo(() => {
   479→    const map = new Map<string, number>();
   480→    for (const r of resumes) map.set(r.id, atsAnalyze(r).score);
   481→    return map;
   482→  }, [resumes]);
   483→
   484→  /* View split. !!r.archived (never a truthy check on the key itself) so old
   485→     persisted state without the optional flag behaves as NOT archived. */
   486→  const activeResumes = React.useMemo(() => resumes.filter((r) => !r.archived), [resumes]);
   487→  const archivedResumes = React.useMemo(() => resumes.filter((r) => !!r.archived), [resumes]);
   488→
   489→  /* Health per resume, derived once and shared by the strip, the filter chips,
   490→     the filtering pipeline and the "Health first" sort. */
   491→  const healthById = React.useMemo(() => {
   492→    const now = Date.now();
   493→    const map = new Map<string, ResumeHealth>();
   494→    for (const r of resumes) map.set(r.id, resumeHealth(scoreHistory, r.id, now));
   495→    return map;
   496→  }, [resumes, scoreHistory]);
   497→
   498→  /* Health summary: scan RECENCY per resume (fresh < 24h, aging 1–7d,
   499→     stale > 7d) with never-scanned counted separately.
   500→     ACTIVE VIEW ONLY — archived resumes are excluded from the health strip,
   501→     the filter chips and the bulk stale re-scan by design: the health
   502→     workflow is about keeping the resumes you actually use fresh. */
   503→  const healthCounts = React.useMemo(() => {
   504→    let fresh = 0;
   505→    let aging = 0;
   506→    let stale = 0;
   507→    let never = 0;
   508→    for (const r of activeResumes) {
   509→      const h = healthById.get(r.id) ?? NEVER_HEALTH;
   510→      if (h.bucket === "fresh") fresh += 1;
   511→      else if (h.bucket === "aging") aging += 1;
   512→      else if (h.scanned) stale += 1;
   513→      else never += 1;
   514→    }
   515→    return { fresh, aging, stale, never };
   516→  }, [activeResumes, healthById]);
   517→
   518→  /* Bucket-true stale count: scanned-stale PLUS never-scanned (both are the
   519→     "stale" bucket), so chips, filter and bulk scan always agree. */
   520→  const staleScanCount = healthCounts.stale + healthCounts.never;
   521→
   522→  const visible = React.useMemo(() => {
   523→    const q = query.trim().toLowerCase();
   524→    /* The list always renders the CURRENT tab's resumes; health chips drive
