    99→function initialsOf(name: string): string {
   100→  const parts = name.trim().split(/\s+/).filter(Boolean);
   101→  if (parts.length === 0) return "?";
   102→  return parts
   103→    .slice(0, 2)
   104→    .map((p) => p[0]!.toUpperCase())
   105→    .join("");
   106→}

   110→interface ParsedBackup {
   111→  payload: Record<string, unknown>;
   112→  resumeCount: number;
   113→  /** Resumes carrying a truthy archived flag. Old exports (v1 and pre-archive v2)
   114→      never have the key, so they read as 0 archived — presentation only, the
   115→      payload schema is untouched. */
   116→  archivedResumeCount: number;
   117→  applicationCount: number | null; // null = absent, store will fall back to seed applications
   118→  scoreHistoryCount: number | null; // null = absent, store will fall back to seed score history
   119→  exportedAt: string | null;
   120→  /** "v2" when the backup carries a 2.x version string or a meta object, "v1" otherwise. */
   121→  formatVersion: "v1" | "v2";
   122→  /** Interview favorites from meta.favorites.interview — null when absent or malformed (v1 backups, bare v2). */
   123→  interviewFavorites: string[] | null;
   124→}

   126→function parseBackupText(text: string): { ok: true; data: ParsedBackup } | { ok: false; reason: string } {
   127→  let parsed: unknown;
   128→  try {
   129→    parsed = JSON.parse(text);
   130→  } catch {
   131→    return { ok: false, reason: "That file isn't valid JSON. Export a fresh backup from Settings and try again." };
   132→  }
   133→  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
   134→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — expected a JSON object." };
   135→  }
   136→  const obj = parsed as Record<string, unknown>;
   137→  const rawResumes = Array.isArray(obj.resumes) ? obj.resumes : null;
   138→  if (obj.app !== "ResumeForge AI" && !rawResumes) {
   139→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — the \"app\" marker and resumes array are missing." };
   140→  }
   141→  if (!rawResumes || rawResumes.length === 0) {
   142→    return { ok: false, reason: "The backup contains no resumes, so there is nothing to restore." };
   143→  }
   144→  for (let i = 0; i < rawResumes.length; i++) {
   145→    const entry: unknown = rawResumes[i];
   146→    if (!entry || typeof entry !== "object" || Array.isArray(entry)) {
   147→      return { ok: false, reason: `Resume #${i + 1} in the backup is malformed (expected an object).` };
   148→    }
   149→    const rec = entry as Record<string, unknown>;
   150→    const titleOk = typeof rec.title === "string";
   151→    if (
   152→      typeof rec.id !== "string" ||
   153→      rec.id.trim() === "" ||
   154→      !titleOk ||
   155→      !rec.personal ||
   156→      typeof rec.personal !== "object" ||
   157→      Array.isArray(rec.personal)
   158→    ) {
   159→      const label = titleOk ? `"${rec.title}"` : `#${i + 1}`;
   160→      return {
   161→        ok: false,
   162→        reason: `Resume ${label} is missing an id, a title or personal details — the backup looks corrupted.`,
   163→      };
   164→    }
   165→  }
   166→  /* Format detection (v1 vs v2): a 2.x version string or a meta object marks a v2 backup.
   167→     Everything else — including versionless legacy exports — is treated as v1. */
   168→  const rawMeta: unknown = obj.meta;
   169→  const hasMeta = !!rawMeta && typeof rawMeta === "object" && !Array.isArray(rawMeta);
   170→  const versionIs2 = typeof obj.version === "string" && obj.version.trim().startsWith("2");
   171→  const formatVersion: "v1" | "v2" = versionIs2 || hasMeta ? "v2" : "v1";
   172→
   173→  /* v2 extras: meta.favorites.interview is an optional array of company keys.
   174→     Normalized the same way readFavorites() stores them (trim/lowercase/dedupe). */
   175→  let interviewFavorites: string[] | null = null;
   176→  if (formatVersion === "v2" && hasMeta) {
   177→    const favorites: unknown = (rawMeta as Record<string, unknown>).favorites;
   178→    if (favorites && typeof favorites === "object" && !Array.isArray(favorites)) {
   179→      const interview: unknown = (favorites as Record<string, unknown>).interview;
   180→      if (Array.isArray(interview)) {
   181→        interviewFavorites = interview
   182→          .filter((v): v is string => typeof v === "string")
   183→          .map((v) => v.trim().toLowerCase())
   184→          .filter((v) => v.length > 0)
   185→          .filter((v, i, arr) => arr.indexOf(v) === i); // dedupe, keep first
   186→      }
   187→    }
   188→  }
   189→
   190→  return {
   191→    ok: true,
   192→    data: {
   193→      payload: obj,
   194→      resumeCount: rawResumes.length,
   195→      archivedResumeCount: rawResumes.reduce(
   196→        (n, entry) =>
   197→          n +
   198→          (entry && typeof entry === "object" && !Array.isArray(entry) && !!(entry as Record<string, unknown>).archived
   199→            ? 1
   200→            : 0),
   201→        0
   202→      ),
   203→      applicationCount: Array.isArray(obj.applications) ? obj.applications.length : null,
   204→      scoreHistoryCount: Array.isArray(obj.scoreHistory) ? obj.scoreHistory.length : null,
   205→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : null,
   206→      formatVersion,
   207→      interviewFavorites,
   208→    },
   209→  };
   210→}

   212→function formatBackupDate(iso: string | null): string {
   213→  if (!iso) return "Not recorded";
   214→  const date = new Date(iso);
   215→  if (!isValidDate(date)) return "Not recorded";
   216→  return format(date, "MMM d, yyyy 'at' h:mm a");
   217→}

   235→/** Number of notification ids marked as read in the bell's read map. */
   236→function countReadNotificationIds(): number {
   237→  if (typeof window === "undefined") return 0;
   238→  try {
   239→    const raw = window.localStorage.getItem(NOTIFS_READ_KEY);
   240→    if (!raw) return 0;
   241→    const parsed: unknown = JSON.parse(raw);
   242→    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return 0;
   243→    return Object.keys(parsed as Record<string, unknown>).length;
   244→  } catch {
   245→    return 0;
   246→  }
