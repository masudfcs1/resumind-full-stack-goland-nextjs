   210→/**
   211→ * Amber star toggle for favoriting a company. Filled amber-400 when active,
   212→ * pops on press, and stops propagation so it can live beside a select button.
   213→ */
   214→function StarToggle({
   215→  active,
   216→  companyName,
   217→  onToggle,
   218→  size = "size-7",
   219→  iconSize = "size-3.5",
   220→  className,
   221→}: {
   222→  active: boolean;
   223→  companyName: string;
   224→  onToggle: () => void;
   225→  size?: string;
   226→  iconSize?: string;
   227→  className?: string;
   228→}) {
   229→  const label = active
   230→    ? `Remove ${companyName} from favorites`
   231→    : `Add ${companyName} to favorites`;
   232→  return (
   233→    <motion.button
   234→      type="button"
   235→      whileTap={{ scale: 0.85 }}
   236→      aria-pressed={active}
   237→      aria-label={label}
   238→      title={label}
   239→      onClick={(e) => {
   240→        e.stopPropagation();
   241→        onToggle();
   242→      }}
   243→      className={cn(
   244→        "flex shrink-0 items-center justify-center rounded-full transition-colors",
   245→        "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500 focus-visible:ring-offset-2 focus-visible:ring-offset-background",
   246→        active
   247→          ? "text-amber-400 hover:bg-amber-500/10"
   248→          : "text-muted-foreground/60 hover:bg-amber-500/10 hover:text-amber-500",
   249→        size,
   250→        className
   251→      )}
   252→    >
   253→      <Star
   254→        aria-hidden="true"
   255→        className={cn(iconSize, active && "fill-amber-400")}
   256→      />
   257→    </motion.button>
   258→  );
   259→}

   261→/** Tabs-like All/Starred two-button filter for the company chip list. */
   262→function FavFilterToggle({
   263→  value,
   264→  onChange,
   265→  starredCount,
   266→}: {
   267→  value: "all" | "starred";
   268→  onChange: (value: "all" | "starred") => void;
   269→  starredCount: number;
   270→}) {
   271→  const options = [
   272→    { value: "all" as const, label: "All" },
   273→    { value: "starred" as const, label: "Starred", count: starredCount },
   274→  ];
   275→  return (
   276→    <div
   277→      role="group"
   278→      aria-label="Filter companies by favorites"
   279→      className="flex items-center gap-1 rounded-lg border bg-muted/50 p-1"
   280→    >
   281→      {options.map((option) => {
   282→        const active = value === option.value;
   283→        return (
   284→          <button
   285→            key={option.value}
   286→            type="button"
   287→            aria-pressed={active}
   288→            onClick={() => onChange(option.value)}
   289→            className={cn(
   290→              "inline-flex items-center gap-1.5 rounded-md px-2.5 py-1 text-xs font-medium transition-all",
   291→              "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-500 focus-visible:ring-offset-1 focus-visible:ring-offset-background",
   292→              active
   293→                ? "bg-emerald-500/10 text-emerald-700 ring-1 ring-emerald-500/40 dark:text-emerald-300"
   294→                : "text-muted-foreground hover:text-foreground"
   295→            )}
   296→          >
   297→            {option.value === "starred" ? (
   298→              <Star
   299→                aria-hidden="true"
   300→                className={cn("size-3", active && "fill-amber-400 text-amber-400")}
   301→              />
   302→            ) : null}
   303→            {option.label}
   304→            {option.count !== undefined ? (
   305→              <span className="tabular-nums opacity-70">{option.count}</span>
   306→            ) : null}
   307→          </button>
   308→        );
   309→      })}
   310→    </div>
   311→  );
   312→}

   340→export default function InterviewPage() {
   341→  const mounted = useMounted();
   342→  const resumes = useResumeStore((s) => s.resumes);
   343→  const activeResumeId = useResumeStore((s) => s.activeResumeId);
   344→  const setActive = useResumeStore((s) => s.setActive);
   345→  const applications = useResumeStore((s) => s.applications);
   346→
   347→  const resume = resumes.find((r) => r.id === activeResumeId) ?? resumes[0];
   348→
   349→  const [selectedCategories, setSelectedCategories] =
   350→    React.useState<QuestionCategory[]>(CATEGORY_ORDER);
   351→  const [difficulty, setDifficulty] = React.useState<DifficultyFilter>("all");
   352→  const [generating, setGenerating] = React.useState(false);
   353→  const [session, setSession] = React.useState<PracticeQuestion[] | null>(null);
   354→  const [generation, setGeneration] = React.useState(0);
   355→  const [practicedIds, setPracticedIds] = React.useState<Set<string>>(new Set());
   356→  const [focusOpen, setFocusOpen] = React.useState(false);
   357→  const [focusIndex, setFocusIndex] = React.useState(0);
   358→  const [starNotes, setStarNotes] = React.useState<Record<string, StarAnswer>>({});
   359→  const [sessionOrigin, setSessionOrigin] = React.useState<SessionOrigin | null>(null);
   360→  const [companyVariant, setCompanyVariant] = React.useState(0);
   361→  /** Starred companies (normalized keys) — lazy-read after mount, SSR-safe. */
   362→  const [favorites, setFavorites] = React.useState<string[]>(() => readFavorites());
   363→  const [favFilter, setFavFilter] = React.useState<"all" | "starred">("all");
   364→  const timerRef = React.useRef<number | null>(null);
   365→
   366→  /** Adds/removes a company from favorites and persists best-effort. */
   367→  const toggleFavorite = React.useCallback(
   368→    (companyName: string) => {
   369→      const key = favoriteKey(companyName);
   370→      const wasFavorite = favorites.includes(key);
   371→      const next = wasFavorite
   372→        ? favorites.filter((f) => f !== key)
   373→        : [...favorites, key];
   374→      setFavorites(next);
   375→      if (!writeFavorites(next)) {
   376→        toast.error("Couldn't save favorites", {
   377→          description: "Browser storage is unavailable in this session.",
   378→        });
   379→      }
   380→      if (wasFavorite) {
   381→        toast(`${companyName} removed from favorites`);
   382→      } else {
   383→        toast.success(`${companyName} added to favorites`);
   384→      }
   385→    },
   386→    [favorites]
   387→  );
   388→
   389→  React.useEffect(() => {
   390→    // If the last favorite is un-starred while the Starred filter is active,
   391→    // fall back to All so the list never renders as an empty mystery.
   392→    if (favFilter === "starred" && favorites.length === 0) setFavFilter("all");
   393→  }, [favFilter, favorites.length]);
   394→
   395→  React.useEffect(() => {
   396→    return () => {
   397→      if (timerRef.current !== null) window.clearTimeout(timerRef.current);
   398→    };
   399→  }, []);
   400→
   401→  const generate = React.useCallback(() => {
   402→    if (!resume || generating) return;
   403→    setGenerating(true);
   404→    const next = generateQuestions(resume, {
   405→      categories: selectedCategories,
   406→      difficulty,
   407→      count: QUESTIONS_PER_SET,
   408→    });
   409→    timerRef.current = window.setTimeout(() => {
   410→      if (next.length === 0) {
   411→        toast.warning("No questions match those filters", {
   412→          description: "Try selecting more categories or a different difficulty.",
   413→        });
   414→        setGenerating(false);
   415→        return;
   416→      }
   417→      setSession(next);
   418→      setPracticedIds(new Set());
   419→      setFocusIndex(0);
   420→      setGeneration((g) => g + 1);
   421→      setSessionOrigin({ kind: "bank" });
   422→      setCompanyVariant(0);
   423→      setGenerating(false);
   424→      toast.success(`Question set ready — ${next.length} questions`, {
   425→        description: `Tailored from "${resume.title}". Flip a card to reveal the STAR framework.`,
   426→      });
   427→    }, GENERATE_DELAY_MS);
   428→  }, [resume, generating, selectedCategories, difficulty]);
   429→
   430→  /** Starts a company-specific session from a tracked application. */
   431→  const startCompanySession = React.useCallback(
   432→    (entry: CompanyEntry) => {
   433→      if (generating) return;
   434→      setGenerating(true);
   435→      setCompanyVariant(0);
   436→      const next = buildCompanyQuestions(entry.company, entry.role, entry.stage, {
   437→        resume,
   438→        variant: 0,
   439→      });
   440→      timerRef.current = window.setTimeout(() => {
   441→        setSession(next);
   442→        setPracticedIds(new Set());
   443→        setFocusIndex(0);
   444→        setGeneration((g) => g + 1);
   445→        setSessionOrigin({
   446→          kind: "company",
   447→          company: entry.company,
   448→          role: entry.role,
   449→          stage: entry.stage,
   450→        });
   451→        setGenerating(false);
   452→        toast.success(`${entry.company} set ready — ${next.length} questions`, {
   453→          description: `Built from your tracker · ${STAGE_META[entry.stage].label} stage.`,
   454→        });
   455→      }, COMPANY_GENERATE_DELAY_MS);
   456→    },
   457→    [generating, resume]
   458→  );
   459→
   460→  /** Re-rolls the pulled questions of the active company set (6 fixed ones stay). */
   461→  const reshuffleCompany = React.useCallback(() => {
   462→    if (!sessionOrigin || sessionOrigin.kind !== "company" || generating) return;
   463→    const variant = companyVariant + 1;
   464→    setGenerating(true);
   465→    setCompanyVariant(variant);
   466→    const next = buildCompanyQuestions(
   467→      sessionOrigin.company,
   468→      sessionOrigin.role,
   469→      sessionOrigin.stage,
   470→      { resume, variant }
   471→    );
   472→    timerRef.current = window.setTimeout(() => {
   473→      setSession(next);
   474→      setPracticedIds(new Set());
   475→      setFocusIndex(0);
   476→      setGeneration((g) => g + 1);
   477→      setGenerating(false);
   478→      toast.success(`${sessionOrigin.company} set reshuffled — ${next.length} questions`);
   479→    }, COMPANY_GENERATE_DELAY_MS);
   480→  }, [sessionOrigin, generating, companyVariant, resume]);
   481→
   482→  /** Unique companies from the tracker — latest application defines stage/role. */
   483→  const companySets = React.useMemo<CompanyEntry[]>(() => {
   484→    const byKey = new Map<string, { entry: CompanyEntry; latest: number }>();
   485→    for (const app of applications) {
   486→      const key = app.company.trim().toLowerCase();
   487→      if (!key) continue;
   488→      const existing = byKey.get(key);
   489→      if (existing) {
   490→        existing.entry.count += 1;
   491→        if (app.updatedAt > existing.latest) {
   492→          existing.latest = app.updatedAt;
   493→          existing.entry.company = app.company;
   494→          existing.entry.role = app.role;
   495→          existing.entry.stage = app.stage;
   496→        }
   497→      } else {
   498→        byKey.set(key, {
   499→          entry: { key, company: app.company, role: app.role, stage: app.stage, count: 1 },
   500→          latest: app.updatedAt,
   501→        });
   502→      }
   503→    }
   504→    const all = [...byKey.values()].map((v) => v.entry);
   505→    // Deterministic ordering: favorites first, each group alphabetical.
   506→    const byName = (a: CompanyEntry, b: CompanyEntry) =>
   507→      a.company.localeCompare(b.company, "en", { sensitivity: "base" }) ||
   508→      a.key.localeCompare(b.key);
   509→    const starred = all
   510→      .filter((e) => favorites.includes(e.key))
   511→      .sort(byName);
   512→    const rest = all
   513→      .filter((e) => !favorites.includes(e.key))
   514→      .sort(byName);
   515→    return [...starred, ...rest];
   516→  }, [applications, favorites]);
   517→
   518→  /** Chips currently visible — applies the All/Starred filter when set. */
   519→  const visibleCompanySets = React.useMemo<CompanyEntry[]>(() => {
   520→    if (favFilter !== "starred" || favorites.length === 0) return companySets;
   521→    return companySets.filter((c) => favorites.includes(c.key));
   522→  }, [companySets, favorites, favFilter]);
   523→
   524→  const toggleCategory = React.useCallback((cat: QuestionCategory) => {
   525→    setSelectedCategories((prev) =>
   526→      prev.includes(cat) ? prev.filter((c) => c !== cat) : [...prev, cat]
   527→    );
   528→  }, []);
   529→
   530→  /** Marks a card practiced — each unique card counts once. */
   531→  const markPracticed = React.useCallback(
   532→    (id: string) => {
   533→      setPracticedIds((prev) => {
   534→        if (prev.has(id)) return prev;
   535→        const next = new Set(prev);
   536→        next.add(id);
   537→        if (session && next.size === session.length) {
   538→          toast.success("Session complete — every question practiced. Nice work.");
   539→        }
   540→        return next;
   541→      });
   542→    },
   543→    [session]
   544→  );
   545→
   546→  const handleNotesChange = React.useCallback((id: string, note: StarAnswer) => {
   547→    setStarNotes((prev) => ({ ...prev, [id]: note }));
   548→  }, []);
   549→
   550→  const openFocus = (index: number) => {
   551→    setFocusIndex(index);
   552→    setFocusOpen(true);
   553→  };
   554→
   555→  const practicedCount = practicedIds.size;
   556→  const totalCount = session?.length ?? 0;
   557→  const pct = totalCount > 0 ? Math.round((practicedCount / totalCount) * 100) : 0;
   558→  const tailoredCount = session?.filter((q) => q.isTailored).length ?? 0;
   559→
   560→  /* ============================== Not hydrated yet ============================== */
   561→
   562→  if (!mounted) {
   563→    return (
   564→      <div className="mx-auto flex w-full max-w-6xl flex-col gap-6" aria-busy="true">
   565→        <div className="flex items-start gap-4">
   566→          <Skeleton className="size-12 rounded-xl" />
   567→          <div className="space-y-2">
   568→            <Skeleton className="h-7 w-56" />
   569→            <Skeleton className="h-4 w-96 max-w-full" />
   570→          </div>
   571→        </div>
   572→        <Skeleton className="h-44 w-full rounded-xl" />
   573→        <div className="grid grid-cols-1 gap-5 sm:grid-cols-2 xl:grid-cols-3">
   574→          {Array.from({ length: 6 }).map((_, i) => (
   575→            <Skeleton key={i} className="h-[350px] rounded-2xl" />
   576→          ))}
   577→        </div>
   578→      </div>
   579→    );
   580→  }
   581→
   582→  /* ============================== Render ============================== */
   583→
   584→  return (
   585→    <motion.div
   586→      variants={staggerContainer}
