   192→export function formatMonth(value: string): string {
   193→  if (!value) return "";
   194→  const [y, m] = value.split("-");
   195→  if (!y) return value;
   196→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   197→  if (!m) return y;
   198→  const idx = parseInt(m, 10) - 1;
   199→  return `${months[idx] ?? ""} ${y}`;
   200→}

   660→/* ============================== Undo buffer (module-local, NOT persisted) ============================== */
   661→
   662→/* Captures the most recent delete action so the Undo button on the delete
   663→   toast can put the resumes back exactly where they were. Deliberately kept
   664→   OUTSIDE the persisted store: undo data is transient and must never reach
   665→   localStorage or widen the persisted schema. */
   666→let undoBuffer: DeletedResumeEntry[] = [];
   667→let undoBatchIds: string[] = [];
   668→let undoBatchSealed = false;
   669→
   670→/* deleteResume may fire several times for ONE user action (multi-select
   671→   delete loops over the selected ids). Calls made within the same synchronous
   672→   task share one batch: the first call snapshots the pre-delete id order so
   673→   every captured entry keeps its ORIGINAL position even though the array
   674→   shrinks between calls; the next task starts a fresh batch. */
   675→function openUndoBatch(): void {
   676→  if (undoBatchSealed) return;
   677→  undoBuffer = [];
   678→  undoBatchIds = useResumeStore.getState().resumes.map((r) => r.id);
   679→  undoBatchSealed = true;
   680→  queueMicrotask(() => {
   681→    undoBatchSealed = false;
   682→  });
   683→}
   684→
   685→/** Returns (and clears) the snapshots captured for the most recent delete
   686→    action. Feed the result straight into showDeleteUndoToast. */
   687→export function takeUndoEntries(): DeletedResumeEntry[] {
   688→  const entries = undoBuffer;
   689→  undoBuffer = [];
   690→  return entries;
   691→}
   692→
   693→/** Standard "deleted" toast with a 7s Undo action that restores the given
   694→    entries at their original positions (restoreResumes). */
   695→export function showDeleteUndoToast(entries: DeletedResumeEntry[]): void {
   696→  const n = entries.length;
   697→  if (n === 0) return;
   698→  toast.success(n === 1 ? "Resume deleted" : `Deleted ${n} resumes`, {
   699→    description:
   700→      n === 1
   701→        ? `“${entries[0]!.resume.title}” was removed from your workspace.`
   702→        : `${n} resumes were removed from your workspace.`,
   703→    duration: 7000,
   704→    action: {
   705→      label: "Undo",
   706→      onClick: () => {
   707→        useResumeStore.getState().restoreResumes(entries);
   708→        toast.success(n === 1 ? "Resume restored" : `Restored ${n} resumes`, {
   709→          description: n === 1 ? "Back at its original position." : "Back at their original positions.",
   710→        });
   711→      },
   712→    },
   713→  });
   714→}

   717→
   718→interface ResumeState {
   719→  resumes: ResumeData[];
   720→  activeResumeId: string;
   721→  coverLetterCount: number;
   722→  applications: JobApplication[];
   723→  scoreHistory: ScoreEntry[];
   724→  createResume: (title?: string, template?: TemplateId, accent?: string) => string;
   725→  duplicateResume: (id: string) => string | undefined;
   726→  deleteResume: (id: string) => void;
   727→  /** Soft-hide: keeps the resume in the store but flags it archived. */
   728→  archiveResume: (id: string) => void;
   729→  /** Reverse of archiveResume — clears the archived flag. */
   730→  unarchiveResume: (id: string) => void;
   731→  /** Batch archive/unarchive for the resume select-mode bulk toolbar.
   732→      Idempotent per resume: flags already in the target state are left so
   733→      their updatedAt is untouched. */
   734→  bulkSetArchived: (ids: string[], archived: boolean) => void;
   735→  /** Splices previously deleted resumes back at their original positions
   736→      (used by the Undo action on delete toasts). Idempotent: entries whose id
   737→      is already present are skipped. */
   738→  restoreResumes: (entries: DeletedResumeEntry[]) => void;
   739→  setActive: (id: string) => void;
   740→  updateResume: (id: string, patch: Partial<ResumeData>) => void;
   741→  resetAll: () => void;
   742→  /** Validates a parsed backup payload and replaces the workspace with it. Returns the number of resumes imported, or -1 if the payload is invalid. */
   743→  importData: (payload: unknown) => number;
   744→  incrementCoverLetters: () => void;
   745→  addApplication: (app: Omit<JobApplication, "id" | "appliedAt" | "updatedAt">) => string;
   746→  updateApplication: (id: string, patch: Partial<JobApplication>) => void;
   747→  moveApplication: (id: string, stage: ApplicationStage) => void;
   748→  deleteApplication: (id: string) => void;
   749→  logScore: (
   750→    resumeId: string,
   751→    resumeTitle: string,
   752→    score: number,
   753→    categories?: Record<string, number>
   754→  ) => void;
   755→  /* --- Per-tool data purge (Settings → Data & privacy). Each action clears
   756→         exactly ONE domain to its empty state and deliberately leaves
   757→         `resumes` (and `activeResumeId`) untouched. Additive: older persisted
   758→         state simply never had these keys, and clearing only writes slices the
   759→         persist config already serializes. --- */
   760→  /** Empties the ATS score-history log (all entries). Resumes are untouched,
   761→      but anything derived from this log resets with it: resume-health
   762→      freshness falls back to "never scanned" and Compare loses its baseline
   763→      entries. Not undoable — callers confirm first. */
   764→  clearScoreHistory: () => void;
   765→  /** Empties the job-application tracker. Resumes are untouched; the dangling
   766→      `resumeId` references simply have no applications left to point at. */
   767→  clearApplications: () => void;
   768→  /** Resets the generated-cover-letter counter to 0. Cover letters are not
   769→      stored as documents anywhere — this counter is the only persisted trace
   770→      of them — so resetting it IS the full cover-letter purge. Resumes are
   771→      untouched. */
   772→  clearCoverLetters: () => void;
   773→}

   775→export const useResumeStore = create<ResumeState>()(
   776→  persist(
   777→    (set, get) => ({
   778→      resumes: [seedResume1(), seedResume2(), seedResume3()],
   779→      activeResumeId: "seed-alex",
   780→      coverLetterCount: 2,
   781→      applications: seedApplications(),
   782→      scoreHistory: seedScoreHistory(),
   783→      createResume: (title, template, accent) => {
   784→        const id = uid();
   785→        const base = emptyResume(title ?? "Untitled Resume", template ?? "modern", accent ?? "#10b981");
   786→        const resume: ResumeData = { ...base, id, updatedAt: Date.now() };
   787→        set((s) => ({ resumes: [resume, ...s.resumes], activeResumeId: id }));
   788→        return id;
   789→      },
   790→      duplicateResume: (id) => {
   791→        const src = get().resumes.find((r) => r.id === id);
   792→        if (!src) return undefined;
   793→        const newId = uid();
   794→        const copy: ResumeData = {
   795→          ...structuredClone(src),
   796→          id: newId,
   797→          title: `${src.title} (Copy)`,
   798→          updatedAt: Date.now(),
   799→          /* Copies always start life in the active list, even if the source was archived. */
   800→          archived: false,
   801→        };
   802→        set((s) => ({ resumes: [copy, ...s.resumes] }));
   803→        return newId;
   804→      },
   805→      deleteResume: (id) => {
   806→        const s = get();
   807→        const index = s.resumes.findIndex((r) => r.id === id);
   808→        if (index === -1) return;
   809→        /* Capture BEFORE removal (deep copy + original position, anchored to
   810→           the batch's pre-delete order) so the Undo toast can restore it. */
   811→        openUndoBatch();
   812→        const batchIndex = undoBatchIds.indexOf(id);
   813→        undoBuffer.push({
   814→          resume: structuredClone(s.resumes[index]!),
   815→          index: batchIndex !== -1 ? batchIndex : index,
   816→        });
   817→        const resumes = s.resumes.filter((r) => r.id !== id);
   818→        set({
   819→          resumes,
   820→          activeResumeId:
   821→            s.activeResumeId === id ? (resumes[0]?.id ?? "") : s.activeResumeId,
   822→        });
   823→      },
   824→      /* Archive bumps updatedAt like every other mutation (store convention).
   825→         The active pointer is intentionally left untouched when archiving the
   826→         active resume — it still resolves (the resume stays in the store), so
   827→         the builder/ATS binding never breaks. */
   828→      archiveResume: (id) =>
   829→        set((s) => ({
   830→          resumes: s.resumes.map((r) =>
   831→            r.id === id ? { ...r, archived: true, updatedAt: Date.now() } : r
   832→          ),
   833→        })),
   834→      unarchiveResume: (id) =>
   835→        set((s) => ({
   836→          resumes: s.resumes.map((r) =>
   837→            r.id === id ? { ...r, archived: false, updatedAt: Date.now() } : r
   838→          ),
   839→        })),
   840→      bulkSetArchived: (ids, archived) =>
   841→        set((s) => {
   842→          const wanted = new Set(ids);
   843→          return {
   844→            resumes: s.resumes.map((r) =>
   845→              wanted.has(r.id) && r.archived !== archived
   846→                ? { ...r, archived, updatedAt: Date.now() }
   847→                : r
   848→            ),
   849→          };
   850→        }),
   851→      restoreResumes: (entries) =>
   852→        set((s) => {
   853→          const resumes = [...s.resumes];
   854→          const present = new Set(resumes.map((r) => r.id));
   855→          /* Splice in descending index order so every entry lands exactly
   856→             where it came from, regardless of how many were removed. */
   857→          const ordered = [...entries].sort((a, b) => b.index - a.index);
   858→          for (const entry of ordered) {
   859→            if (present.has(entry.resume.id)) continue;
   860→            const at = Math.min(Math.max(entry.index, 0), resumes.length);
   861→            resumes.splice(at, 0, structuredClone(entry.resume));
   862→            present.add(entry.resume.id);
   863→          }
   864→          return { resumes };
   865→        }),
   866→      setActive: (id) => set({ activeResumeId: id }),
   867→      updateResume: (id, patch) =>
   868→        set((s) => ({
   869→          resumes: s.resumes.map((r) =>
   870→            r.id === id ? { ...r, ...patch, updatedAt: Date.now() } : r
   871→          ),
   872→        })),
   873→      resetAll: () =>
   874→        set({
   875→          resumes: [seedResume1(), seedResume2(), seedResume3()],
   876→          activeResumeId: "seed-alex",
   877→          coverLetterCount: 2,
   878→          applications: seedApplications(),
   879→          scoreHistory: seedScoreHistory(),
   880→        }),
   881→      importData: (payload) => {
   882→        if (!payload || typeof payload !== "object" || Array.isArray(payload)) return -1;
   883→        const raw = payload as Record<string, unknown>;
   884→
   885→        /* Resumes are required — every entry needs an id, a title and a personal object. */
   886→        const rawResumes = raw.resumes;
   887→        if (!Array.isArray(rawResumes) || rawResumes.length === 0) return -1;
   888→        const blank = emptyResume();
   889→        const resumes: ResumeData[] = [];
   890→        for (const entry of rawResumes) {
   891→          if (!entry || typeof entry !== "object" || Array.isArray(entry)) return -1;
   892→          const r = entry as Record<string, unknown>;
   893→          if (typeof r.id !== "string" || r.id.trim() === "") return -1;
   894→          if (typeof r.title !== "string") return -1;
   895→          if (!r.personal || typeof r.personal !== "object" || Array.isArray(r.personal)) return -1;
   896→          const copy = structuredClone(entry) as ResumeData;
   897→          copy.template = typeof r.template === "string" && r.template in TEMPLATE_META ? (r.template as TemplateId) : blank.template;
   898→          copy.accent = typeof r.accent === "string" && r.accent.trim() !== "" ? r.accent : blank.accent;
   899→          copy.summary = typeof r.summary === "string" ? r.summary : blank.summary;
   900→          copy.personal = { ...blank.personal, ...(r.personal as Partial<PersonalInfo>) };
   901→          for (const key of ["experience", "education", "skills", "projects", "certifications", "languages"] as const) {
   902→            if (!Array.isArray(copy[key])) copy[key] = [];
   903→          }
   904→          copy.updatedAt = typeof r.updatedAt === "number" && Number.isFinite(r.updatedAt) ? r.updatedAt : Date.now();
   905→          resumes.push(copy);
   906→        }
   907→
   908→        /* Active resume falls back to the first import if the id is missing or unknown. */
   909→        const wantedActive = typeof raw.activeResumeId === "string" ? raw.activeResumeId : "";
   910→        const firstId = resumes[0]!.id;
   911→        const activeResumeId = resumes.some((r) => r.id === wantedActive) ? wantedActive : firstId;
   912→
   913→        const coverLetterCount =
   914→          typeof raw.coverLetterCount === "number" && Number.isFinite(raw.coverLetterCount)
   915→            ? Math.max(0, Math.round(raw.coverLetterCount))
   916→            : 0;
   917→
   918→        /* Applications and score history fall back to seeds when absent (mirrors resetAll). */
   919→        let applications = seedApplications();
   920→        if (Array.isArray(raw.applications)) {
   921→          const clean: JobApplication[] = [];
   922→          for (const entry of raw.applications) {
   923→            if (!entry || typeof entry !== "object" || Array.isArray(entry)) continue;
   924→            const a = entry as Record<string, unknown>;
   925→            clean.push({
   926→              id: typeof a.id === "string" && a.id !== "" ? a.id : uid(),
   927→              company: typeof a.company === "string" ? a.company : "",
   928→              role: typeof a.role === "string" ? a.role : "",
   929→              location: typeof a.location === "string" ? a.location : "",
   930→              salary: typeof a.salary === "string" ? a.salary : "",
   931→              url: typeof a.url === "string" ? a.url : "",
   932→              stage: typeof a.stage === "string" && a.stage in STAGE_META ? (a.stage as ApplicationStage) : "applied",
   933→              resumeId: typeof a.resumeId === "string" ? a.resumeId : "",
   934→              notes: typeof a.notes === "string" ? a.notes : "",
   935→              appliedAt: typeof a.appliedAt === "number" && Number.isFinite(a.appliedAt) ? a.appliedAt : Date.now(),
   936→              updatedAt: typeof a.updatedAt === "number" && Number.isFinite(a.updatedAt) ? a.updatedAt : Date.now(),
   937→            });
   938→          }
   939→          if (clean.length > 0) applications = clean;
   940→        }
   941→
   942→        let scoreHistory = seedScoreHistory();
   943→        if (Array.isArray(raw.scoreHistory)) {
   944→          const clean: ScoreEntry[] = [];
   945→          for (const entry of raw.scoreHistory) {
   946→            if (!entry || typeof entry !== "object" || Array.isArray(entry)) continue;
   947→            const e = entry as Record<string, unknown>;
   948→            if (typeof e.score !== "number" || !Number.isFinite(e.score)) continue;
   949→            clean.push({
   950→              id: typeof e.id === "string" && e.id !== "" ? e.id : uid(),
   951→              resumeId: typeof e.resumeId === "string" ? e.resumeId : "",
   952→              resumeTitle: typeof e.resumeTitle === "string" ? e.resumeTitle : "Untitled",
   953→              score: e.score,
   954→              at: typeof e.at === "number" && Number.isFinite(e.at) ? e.at : Date.now(),
   955→            });
   956→          }
   957→          if (clean.length > 0) scoreHistory = clean.slice(-60);
   958→        }
   959→
   960→        set({ resumes, activeResumeId, coverLetterCount, applications, scoreHistory });
   961→        return resumes.length;
   962→      },
   963→      incrementCoverLetters: () => set((s) => ({ coverLetterCount: s.coverLetterCount + 1 })),
   964→      addApplication: (app) => {
   965→        const id = uid();
   966→        const full: JobApplication = {
   967→          ...app,
   968→          id,
   969→          appliedAt: Date.now(),
   970→          updatedAt: Date.now(),
   971→        };
   972→        set((s) => ({ applications: [full, ...s.applications] }));
   973→        return id;
   974→      },
   975→      updateApplication: (id, patch) =>
   976→        set((s) => ({
   977→          applications: s.applications.map((a) =>
   978→            a.id === id ? { ...a, ...patch, updatedAt: Date.now() } : a
   979→          ),
   980→        })),
   981→      moveApplication: (id, stage) =>
   982→        set((s) => ({
   983→          applications: s.applications.map((a) =>
   984→            a.id === id ? { ...a, stage, updatedAt: Date.now() } : a
   985→          ),
   986→        })),
   987→      deleteApplication: (id) =>
   988→        set((s) => ({ applications: s.applications.filter((a) => a.id !== id) })),
   989→      logScore: (resumeId, resumeTitle, score, categories) =>
   990→        set((s) => ({
   991→          scoreHistory: [
   992→            ...s.scoreHistory,
   993→            { id: uid(), resumeId, resumeTitle, score, at: Date.now(), categories },
   994→          ].slice(-60),
   995→        })),
   996→      /* Per-tool data purge (Settings → Data & privacy) — one line each,
   997→         mirrors the one-liner style of incrementCoverLetters/deleteApplication. */
   998→      clearScoreHistory: () => set({ scoreHistory: [] }),
   999→      clearApplications: () => set({ applications: [] }),
  1000→      clearCoverLetters: () => set({ coverLetterCount: 0 }),
  1001→    }),
  1002→    {
  1003→      name: "resumeforge-store",
  1004→      storage: createJSONStorage(() => localStorage),
  1005→      // Rehydrate AFTER React hydration (see <StoreRehydrator /> in root layout).
  1006→      // Synchronous rehydration makes the first client render read localStorage
  1007→      // while the server rendered seed data — any divergence (extra resumes,
  1008→      // archived flags, renamed profile…) shifts the render tree and breaks
  1009→      // hydration (Radix useId mismatches on every dashboard route).
  1010→      skipHydration: true,
  1011→    }
  1012→  )
  1013→);
  1014→
  1015→/* Convenience selectors */
  1016→
  1017→/**
  1018→ * True once the persisted store has finished (post-hydration) rehydration —
  1019→ * see <StoreRehydrator />. Components whose DOM SHAPE depends on stored data
  1020→ * (e.g. the Compare category bars vs their muted "no scan" placeholders)
  1021→ * should read this and render their seed-neutral shape until it flips, so a
  1022→ * rehydrate landing inside a hydration pass can never change element types
  1023→ * or counts mid-hydration.
  1024→ */
  1025→export function useStoreHydrated(): boolean {
  1026→  return useSyncExternalStore(
  1027→    (onStoreChange) => useResumeStore.persist.onFinishHydration(onStoreChange),
  1028→    () => useResumeStore.persist.hasHydrated(),
  1029→    () => false
  1030→  );
  1031→}
