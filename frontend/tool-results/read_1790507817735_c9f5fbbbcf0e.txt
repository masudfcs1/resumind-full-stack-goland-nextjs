    98→    98→function initialsOf(name: string): string {
    99→    99→  const parts = name.trim().split(/\s+/).filter(Boolean);
   100→   100→  if (parts.length === 0) return "?";
   101→   101→  return parts
   102→   102→    .slice(0, 2)
   103→   103→    .map((p) => p[0]!.toUpperCase())
   104→   104→    .join("");
   105→   105→}

   117→   117→function parseBackupText(text: string): { ok: true; data: ParsedBackup } | { ok: false; reason: string } {
   118→   118→  let parsed: unknown;
   119→   119→  try {
   120→   120→    parsed = JSON.parse(text);
   121→   121→  } catch {
   122→   122→    return { ok: false, reason: "That file isn't valid JSON. Export a fresh backup from Settings and try again." };
   123→   123→  }
   124→   124→  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
   125→   125→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — expected a JSON object." };
   126→   126→  }
   127→   127→  const obj = parsed as Record<string, unknown>;
   128→   128→  const rawResumes = Array.isArray(obj.resumes) ? obj.resumes : null;
   129→   129→  if (obj.app !== "ResumeForge AI" && !rawResumes) {
   130→   130→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — the \"app\" marker and resumes array are missing." };
   131→   131→  }
   132→   132→  if (!rawResumes || rawResumes.length === 0) {
   133→   133→    return { ok: false, reason: "The backup contains no resumes, so there is nothing to restore." };
   134→   134→  }
   135→   135→  for (let i = 0; i < rawResumes.length; i++) {
   136→   136→    const entry: unknown = rawResumes[i];
   137→   137→    if (!entry || typeof entry !== "object" || Array.isArray(entry)) {
   138→   138→      return { ok: false, reason: `Resume #${i + 1} in the backup is malformed (expected an object).` };
   139→   139→    }
   140→   140→    const rec = entry as Record<string, unknown>;
   141→   141→    const titleOk = typeof rec.title === "string";
   142→   142→    if (
   143→   143→      typeof rec.id !== "string" ||
   144→   144→      rec.id.trim() === "" ||
   145→   145→      !titleOk ||
   146→   146→      !rec.personal ||
   147→   147→      typeof rec.personal !== "object" ||
   148→   148→      Array.isArray(rec.personal)
   149→   149→    ) {
   150→   150→      const label = titleOk ? `"${rec.title}"` : `#${i + 1}`;
   151→   151→      return {
   152→   152→        ok: false,
   153→   153→        reason: `Resume ${label} is missing an id, a title or personal details — the backup looks corrupted.`,
   154→   154→      };
   155→   155→    }
   156→   156→  }
   157→   157→  return {
   158→   158→    ok: true,
   159→   159→    data: {
   160→   160→      payload: obj,
   161→   161→      resumeCount: rawResumes.length,
   162→   162→      applicationCount: Array.isArray(obj.applications) ? obj.applications.length : null,
   163→   163→      scoreHistoryCount: Array.isArray(obj.scoreHistory) ? obj.scoreHistory.length : null,
   164→   164→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : null,
   165→   165→    },
   166→   166→  };
   167→   167→}

   169→   169→function formatBackupDate(iso: string | null): string {
   170→   170→  if (!iso) return "Not recorded";
   171→   171→  const date = new Date(iso);
   172→   172→  if (!isValidDate(date)) return "Not recorded";
   173→   173→  return format(date, "MMM d, yyyy 'at' h:mm a");
   174→   174→}
