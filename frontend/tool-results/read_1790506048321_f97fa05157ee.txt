   167→  onOpenChange,

   170→  onOpenChange: (open: boolean) => void;

   181→    setTitle("");

   189→        <DialogHeader>
   190→          <DialogTitle className="font-display">Create a new resume</DialogTitle>

   300→
   301→export default function MyResumesPage() {
   302→  const mounted = useMounted();
   303→  const resumes = useResumeStore((s) => s.resumes);
   304→  const activeResumeId = useResumeStore((s) => s.activeResumeId);
   305→  const active = useActiveResume();
   306→  const duplicateResume = useResumeStore((s) => s.duplicateResume);
   307→  const deleteResume = useResumeStore((s) => s.deleteResume);
   308→
   309→  const [query, setQuery] = React.useState("");
   310→  const [sort, setSort] = React.useState<SortKey>("recent");
   311→  const [view, setView] = React.useState<ViewMode>("grid");
   312→  const [createOpen, setCreateOpen] = React.useState(false);
   313→  const [selectMode, setSelectMode] = React.useState(false);
   314→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   315→  const [confirmDelete, setConfirmDelete] = React.useState(false);
   316→
   317→  const scoreMap = React.useMemo(() => {
   318→    const map = new Map<string, number>();
   319→    for (const r of resumes) map.set(r.id, atsAnalyze(r).score);
   320→    return map;
   321→  }, [resumes]);
   322→
   323→  const visible = React.useMemo(() => {
   324→    const q = query.trim().toLowerCase();
   325→    const filtered = resumes.filter(
   326→      (r) =>
   327→        !q ||
   328→        r.title.toLowerCase().includes(q) ||
   329→        r.personal.fullName.toLowerCase().includes(q) ||
   330→        r.personal.jobTitle.toLowerCase().includes(q)
   331→    );
   332→    const scoreOf = (id: string) => scoreMap.get(id) ?? 0;
   333→    switch (sort) {
   334→      case "name":
   335→        return [...filtered].sort((a, b) => a.title.localeCompare(b.title));
   336→      case "ats":
   337→        return [...filtered].sort((a, b) => scoreOf(b.id) - scoreOf(a.id));
   338→      default:
   339→        return [...filtered].sort((a, b) => b.updatedAt - a.updatedAt);
   340→    }
   341→  }, [resumes, query, sort, scoreMap]);
   342→
   343→  /* Keep the selection honest when resumes disappear (per-card deletes, resets) */
   344→  React.useEffect(() => {
   345→    setSelected((prev) => {
   346→      if (prev.size === 0) return prev;
   347→      const ids = new Set(resumes.map((r) => r.id));
   348→      const next = new Set([...prev].filter((id) => ids.has(id)));
   349→      return next.size === prev.size ? prev : next;
   350→    });
   351→  }, [resumes]);
   352→
   353→  const exitSelectMode = React.useCallback(() => {
   354→    setSelectMode(false);
   355→    setSelected(new Set());
   356→  }, []);
   357→
   358→  const toggleSelect = React.useCallback((id: string) => {
   359→    setSelected((prev) => {
   360→      const next = new Set(prev);
   361→      if (next.has(id)) next.delete(id);
   362→      else next.add(id);
   363→      return next;
   364→    });
   365→  }, []);
   366→
   367→  /* ESC exits selection mode — but never while an AlertDialog (bulk delete or
   368→     per-card delete) is open. The listener runs in the CAPTURE phase so it sees
   369→     the dialog before Radix's document-level handler closes it: Radix calls
   370→     onOpenChange(false) synchronously during the Escape dispatch (React 19
   371→     flush), so a bubble-phase DOM/state check would already observe the dialog
   372→     as gone and wrongly exit selection mode. */
   373→  React.useEffect(() => {
   374→    if (!selectMode) return;
   375→    const onKey = (e: KeyboardEvent) => {
   376→      if (e.key !== "Escape") return;
   377→      if (document.querySelector("[role=alertdialog]")) return;
   378→      exitSelectMode();
   379→    };
   380→    window.addEventListener("keydown", onKey, true);
   381→    return () => window.removeEventListener("keydown", onKey, true);
   382→  }, [selectMode, exitSelectMode]);
   383→
   384→  const handleBulkDuplicate = () => {
   385→    let ok = 0;
   386→    for (const id of selected) {
   387→      if (duplicateResume(id)) ok += 1;
   388→    }
   389→    if (ok > 0) {
   390→      toast.success(`Duplicated ${ok} ${ok === 1 ? "resume" : "resumes"}`, {
   391→        description: "Copies were added to the top of your list.",
   392→      });
   393→    }
   394→    /* Selection mode stays ON and the same set stays selected so users can continue. */
   395→  };
   396→
   397→  const handleBulkDelete = () => {
   398→    const ids = [...selected];
   399→    for (const id of ids) deleteResume(id);
   400→    if (ids.length > 0) {
   401→      toast.success(`Deleted ${ids.length} ${ids.length === 1 ? "resume" : "resumes"}`);
   402→    }
   403→    setConfirmDelete(false);
   404→    setSelected(new Set());
   405→    setSelectMode(false);
   406→  };
   407→
   408→  if (!mounted) {
   409→    return <ResumesSkeleton />;
   410→  }
   411→
   412→  const hasAny = resumes.length > 0;
   413→  const selectedCount = selected.size;
   414→  const allVisibleSelected =
   415→    visible.length > 0 && visible.every((r) => selected.has(r.id));
   416→  const toggleSelectAll = () => {
   417→    setSelected(allVisibleSelected ? new Set() : new Set(visible.map((r) => r.id)));
   418→  };
   419→
   420→  return (
   421→    <div className="mx-auto max-w-6xl">
   422→      {/* Header row */}
   423→      <header className="flex flex-wrap items-center gap-3">
   424→        <div className="min-w-0">
   425→          <h1 className="flex items-center gap-2 font-display text-2xl font-bold tracking-tight">
   426→            My Resumes
   427→            <Badge
   428→              variant="secondary"
   429→              className="rounded-full bg-emerald-500/10 text-emerald-700 dark:text-emerald-400"
   430→            >
   431→              {resumes.length}
   432→            </Badge>
   433→          </h1>
   434→          <p className="mt-0.5 text-[13px] text-muted-foreground">
   435→            Manage, duplicate and track every version of your resume.
   436→          </p>
   437→        </div>
   438→
   439→        <div className="ml-auto flex flex-wrap items-center gap-2">
   440→          <div className="relative w-full sm:w-52">
   441→            <Search
   442→              className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground"
   443→              aria-hidden="true"
   444→            />
   445→            <Input
   446→              value={query}
   447→              onChange={(e) => setQuery(e.target.value)}
   448→              placeholder="Search resumes…"
   449→              aria-label="Search resumes"
   450→              className="h-9 rounded-lg pl-8"
   451→            />
   452→          </div>
   453→
   454→          <Select value={sort} onValueChange={(v) => setSort(v as SortKey)}>
   455→            <SelectTrigger className="h-9 w-[164px] rounded-lg" aria-label="Sort resumes">
   456→              <SelectValue />
   457→            </SelectTrigger>
   458→            <SelectContent>
   459→              <SelectItem value="recent">Recently edited</SelectItem>
   460→              <SelectItem value="name">Name A-Z</SelectItem>
   461→              <SelectItem value="ats">ATS score</SelectItem>
   462→            </SelectContent>
   463→          </Select>
   464→
   465→          <div
   466→            className="flex items-center rounded-lg border p-0.5"
   467→            role="group"
   468→            aria-label="View mode"
   469→          >
   470→            <Button
   471→              variant={view === "grid" ? "secondary" : "ghost"}
   472→              size="icon"
   473→              className="h-8 w-8"
   474→              onClick={() => setView("grid")}
   475→              aria-pressed={view === "grid"}
   476→              aria-label="Grid view"
   477→            >
   478→              <LayoutGrid className="h-4 w-4" />
   479→            </Button>
   480→            <Button
   481→              variant={view === "list" ? "secondary" : "ghost"}
   482→              size="icon"
   483→              className="h-8 w-8"
   484→              onClick={() => setView("list")}
   485→              aria-pressed={view === "list"}
   486→              aria-label="List view"
   487→            >
   488→              <List className="h-4 w-4" />
   489→            </Button>
   490→          </div>
   491→
   492→          {selectMode && (
   493→            <Button
   494→              variant="outline"
   495→              size="sm"
   496→              className="h-9 rounded-lg"
   497→              onClick={toggleSelectAll}
   498→            >
   499→              {allVisibleSelected ? "Deselect all" : "Select all"}
   500→            </Button>
   501→          )}
   502→
   503→          <Button
   504→            variant="ghost"
   505→            onClick={() => (selectMode ? exitSelectMode() : setSelectMode(true))}
   506→            aria-pressed={selectMode}
   507→            className="h-9 gap-1.5 rounded-lg"
   508→          >
   509→            {selectMode ? (
   510→              <X className="h-4 w-4" aria-hidden="true" />
   511→            ) : (
   512→              <CheckSquare className="h-4 w-4" aria-hidden="true" />
   513→            )}
   514→            {selectMode ? "Cancel" : "Select"}
   515→          </Button>
   516→
   517→          <Button
   518→            onClick={() => setCreateOpen(true)}
   519→            className="h-9 gap-1.5 rounded-lg bg-gradient-to-r from-emerald-500 to-teal-600 text-white shadow-md shadow-emerald-500/20 hover:from-emerald-600 hover:to-teal-700"
   520→          >
   521→            <Plus className="h-4 w-4" aria-hidden="true" /> New Resume
   522→          </Button>
   523→        </div>
   524→      </header>
   525→
   526→      {/* Active resume hint */}
   527→      {active && (
   528→        <p className="mt-3 flex items-center gap-1.5 text-[12.5px] text-muted-foreground">
   529→          <Sparkles className="h-3.5 w-3.5 text-emerald-500" aria-hidden="true" />
   530→          Active resume: <span className="font-medium text-foreground">{active.title}</span>
   531→        </p>
   532→      )}
   533→
   534→      {/* Content */}
   535→      {!hasAny ? (
   536→        <div className="mt-6 flex flex-col items-center gap-3 rounded-2xl border border-dashed border-emerald-500/30 bg-emerald-500/[0.04] p-10 text-center">
   537→          <h3 className="font-display text-lg font-bold tracking-tight">No resumes yet</h3>
   538→          <p className="max-w-sm text-sm text-muted-foreground">
   539→            Create your first resume to unlock the builder, ATS scanner and cover letters.
   540→          </p>
   541→          <Button
   542→            onClick={() => setCreateOpen(true)}
   543→            className="gap-1.5 bg-gradient-to-r from-emerald-500 to-teal-600 text-white shadow-md shadow-emerald-500/20 hover:from-emerald-600 hover:to-teal-700"
   544→          >
   545→            <Plus className="h-4 w-4" aria-hidden="true" /> New Resume
   546→          </Button>
   547→        </div>
   548→      ) : visible.length === 0 ? (
   549→        <div className="mt-6 flex flex-col items-center gap-3 rounded-xl border border-dashed p-10 text-center">
   550→          <SearchX className="h-8 w-8 text-muted-foreground/60" aria-hidden="true" />
   551→          <p className="text-sm text-muted-foreground">
   552→            No resumes match “{query}”. Try a different search.
   553→          </p>
   554→          <Button variant="outline" size="sm" onClick={() => setQuery("")}>
   555→            Clear search
   556→          </Button>
   557→        </div>
   558→      ) : view === "grid" ? (
   559→        <div
   560→          className={cn(
   561→            "mt-6 grid min-h-[200px] max-h-[calc(100vh-250px)] gap-5 overflow-y-auto pr-1 scrollbar-thin sm:grid-cols-2 xl:grid-cols-3",
   562→            selectMode ? "pb-36" : "pb-4"
   563→          )}
   564→          role="list"
   565→          aria-label="Resumes grid"
   566→        >
   567→          {visible.map((r: ResumeData, i: number) => (
   568→            <ResumeCard
   569→              key={r.id}
   570→              resume={r}
   571→              isActive={r.id === activeResumeId}
   572→              index={i}
   573→              selectionMode={selectMode}
   574→              selected={selected.has(r.id)}
   575→              onToggleSelect={toggleSelect}
   576→            />
   577→          ))}
   578→        </div>
   579→      ) : (
   580→        <div
   581→          className={cn(
   582→            "mt-6 min-h-[200px] max-h-[calc(100vh-250px)] space-y-3 overflow-y-auto pr-1 scrollbar-thin",
   583→            selectMode ? "pb-36" : "pb-4"
   584→          )}
   585→          role="list"
   586→          aria-label="Resumes list"
   587→        >
   588→          {visible.map((r: ResumeData, i: number) => (
   589→            <ResumeListRow
   590→              key={r.id}
   591→              resume={r}
   592→              isActive={r.id === activeResumeId}
   593→              index={i}
   594→              selectionMode={selectMode}
   595→              selected={selected.has(r.id)}
   596→              onToggleSelect={toggleSelect}
   597→            />
   598→          ))}
   599→        </div>
   600→      )}
   601→
   602→      <NewResumeDialog open={createOpen} onOpenChange={setCreateOpen} />
   603→
   604→      {/* Floating bulk action bar (multi-select mode) */}
   605→      <AnimatePresence>
   606→        {selectMode && (
   607→          <motion.div
   608→            key="bulk-bar"
   609→            initial={{ y: 20, opacity: 0 }}
   610→            animate={{ y: 0, opacity: 1 }}
   611→            exit={{ y: 20, opacity: 0 }}
   612→            transition={{ type: "spring", stiffness: 400, damping: 32 }}
   613→            className="pointer-events-none fixed inset-x-0 bottom-20 z-50 flex justify-center px-4 pb-[env(safe-area-inset-bottom)]"
   614→          >
   615→            <div
   616→              role="toolbar"
   617→              aria-label="Bulk actions"
   618→              className="pointer-events-auto flex items-center gap-2 rounded-full border bg-background/95 px-4 py-2.5 shadow-xl backdrop-blur"
   619→            >
   620→              <span
   621→                className="whitespace-nowrap text-sm font-medium tabular-nums"
   622→                aria-live="polite"
   623→              >
   624→                {selectedCount} selected
   625→              </span>
   626→
   627→              <span className="h-5 w-px shrink-0 bg-border" aria-hidden="true" />
   628→
   629→              <Button
   630→                variant="outline"
   631→                size="sm"
   632→                className="h-8 gap-1.5 rounded-full"
   633→                disabled={selectedCount === 0}
   634→                onClick={handleBulkDuplicate}
   635→              >
   636→                <Copy className="h-3.5 w-3.5" aria-hidden="true" /> Duplicate
   637→              </Button>
   638→
   639→              <Button
   640→                variant="outline"
   641→                size="sm"
   642→                disabled={selectedCount === 0}
   643→                onClick={() => setConfirmDelete(true)}
   644→                className="h-8 gap-1.5 rounded-full border-rose-500/40 text-rose-600 hover:border-rose-500/60 hover:bg-rose-500/10 hover:text-rose-700 dark:text-rose-400 dark:hover:text-rose-300"
   645→              >
   646→                <Trash2 className="h-3.5 w-3.5" aria-hidden="true" /> Delete
   647→              </Button>
   648→
   649→              <Button
   650→                variant="ghost"
   651→                size="icon"
   652→                className="h-8 w-8 shrink-0 rounded-full text-muted-foreground hover:text-foreground"
   653→                onClick={exitSelectMode}
   654→                aria-label="Exit selection mode"
   655→              >
   656→                <X className="h-4 w-4" aria-hidden="true" />
   657→              </Button>
   658→            </div>
   659→          </motion.div>
   660→        )}
   661→      </AnimatePresence>
   662→
   663→      {/* Bulk delete confirmation */}
   664→      <AlertDialog open={confirmDelete} onOpenChange={setConfirmDelete}>
   665→        <AlertDialogContent>
   666→          <AlertDialogHeader>
   667→            <AlertDialogTitle>
   668→              Delete {selectedCount} {selectedCount === 1 ? "resume" : "resumes"}?
   669→            </AlertDialogTitle>
   670→            <AlertDialogDescription>
   671→              This permanently removes {selectedCount === 1 ? "it" : "them"} and all of their
   672→              content. This action cannot be undone.
   673→            </AlertDialogDescription>
   674→          </AlertDialogHeader>
   675→          <AlertDialogFooter>
   676→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   677→            <AlertDialogAction
   678→              className="bg-destructive text-white hover:bg-destructive/90"
   679→              onClick={handleBulkDelete}
   680→            >
   681→              Delete
   682→            </AlertDialogAction>
   683→          </AlertDialogFooter>
   684→        </AlertDialogContent>
   685→      </AlertDialog>
   686→    </div>
   687→  );
   688→}
