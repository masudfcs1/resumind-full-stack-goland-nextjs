   383→   383→
   384→   384→/**
   385→   385→ * Slim 3-segment difficulty ladder strip (Warm-up / Core / Stretch) shown
   386→   386→ * above the question list. Each segment carries its phase count and a
   387→   387→ * practiced-progress fill; the segment containing the current question is
   388→   388→ * highlighted in its tier tint, and a one-line coach hint sits underneath.
   389→   389→ */
   390→   390→function PhaseStrip({
   391→   391→  questions,
   392→   392→  practicedIds,
   393→   393→  currentIndex,
   394→   394→}: {
   395→   395→  questions: PracticeQuestion[];
   396→   396→  practicedIds: ReadonlySet<string>;
   397→   397→  currentIndex: number;
   398→   398→}) {
   399→   399→  const reducedMotion = useReducedMotion();
   400→   400→
   401→   401→  const segments = React.useMemo(
   402→   402→    () =>
   403→   403→      PHASE_SEGMENTS.map((seg) => {
   404→   404→        const idxs = questions
   405→   405→          .map((q, i) => (q.difficulty === seg.tier ? i : -1))
   406→   406→          .filter((i) => i >= 0);
   407→   407→        const practiced = idxs.filter((i) => {
   408→   408→          const q = questions[i];
   409→   409→          return q ? practicedIds.has(q.id) : false;
   410→   410→        }).length;
   411→   411→        return { ...seg, count: idxs.length, practiced };
   412→   412→      }),
   413→   413→    [questions, practicedIds]
   414→   414→  );
   415→   415→
   416→   416→  const currentTier = questions[currentIndex]?.difficulty ?? "easy";
   417→   417→  const currentSegment = PHASE_SEGMENTS.find((s) => s.tier === currentTier);
   418→   418→
   419→   419→  return (
   420→   420→    <motion.section
   421→   421→      variants={staggerItem}
   422→   422→      aria-label="Difficulty phases"
   423→   423→      data-phase-strip="true"
   424→   424→      data-current-phase={currentSegment?.id ?? "warm-up"}
   425→   425→    >
   426→   426→      <div className="flex gap-1.5 sm:gap-2">
   427→   427→        {segments.map((seg) => {
   428→   428→          const active = seg.tier === currentTier;
   429→   429→          const fillPct =
   430→   430→            seg.count > 0 ? Math.round((seg.practiced / seg.count) * 100) : 0;
   431→   431→          return (
   432→   432→            <div
   433→   433→              key={seg.tier}
   434→   434→              data-phase={seg.id}
   435→   435→              data-tier={seg.tier}
   436→   436→              data-count={seg.count}
   437→   437→              data-active={active ? "true" : "false"}
   438→   438→              className={cn(
   439→   439→                "relative min-w-0 flex-1 overflow-hidden rounded-lg border px-2.5 py-1.5 transition-colors sm:px-3",
   440→   440→                active ? seg.active : "border-border"
   441→   441→              )}
   442→   442→            >
   443→   443→              <motion.div
   444→   444→                aria-hidden="true"
   445→   445→                className={cn("absolute inset-y-0 left-0", seg.fill)}
   446→   446→                initial={false}
   447→   447→                animate={{ width: `${fillPct}%` }}
   448→   448→                transition={
   449→   449→                  reducedMotion
   450→   450→                    ? { duration: 0 }
   451→   451→                    : { type: "spring", stiffness: 140, damping: 22 }
   452→   452→                }
   453→   453→              />
   454→   454→              <div className="relative flex min-w-0 items-center justify-between gap-2">
   455→   455→                <span
   456→   456→                  className={cn(
   457→   457→                    "truncate text-[10px] font-semibold uppercase tracking-[0.12em]",
   458→   458→                    active ? seg.text : "text-muted-foreground"
   459→   459→                  )}
   460→   460→                >
   461→   461→                  {DIFFICULTY_META[seg.tier].phase}
   462→   462→                </span>
   463→   463→                <span
   464→   464→                  className={cn(
   465→   465→                    "shrink-0 text-[10px] font-semibold tabular-nums",
   466→   466→                    active ? seg.text : "text-muted-foreground/70"
   467→   467→                  )}
   468→   468→                >
   469→   469→                  {seg.practiced}/{seg.count}
   470→   470→                </span>
   471→   471→              </div>
   472→   472→            </div>
   473→   473→          );
   474→   474→        })}
   475→   475→      </div>
   476→   476→      <p
   477→   477→        className={cn(
   478→   478→          "mt-2 flex items-start gap-2 rounded-lg border px-3 py-2 text-xs leading-relaxed",
   479→   479→          PHASE_COACH_TINT[currentTier]
   480→   480→        )}
   481→   481→      >
   482→   482→        <Compass className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   483→   483→        <span>
   484→   484→          <span className="font-semibold">Coach:</span> {PHASE_COACH[currentTier]}
   485→   485→        </span>
   486→   486→      </p>
   487→   487→      <p className="sr-only">
   488→   488→        {`Phases: ${segments
   489→   489→          .map((s) => `${DIFFICULTY_META[s.tier].phase} ${s.count} questions`)
   490→   490→          .join(" · ")}. Currently in the ${DIFFICULTY_META[currentTier].phase} phase.`}
   491→   491→      </p>
   492→   492→    </motion.section>
   493→   493→  );
   494→   494→}
