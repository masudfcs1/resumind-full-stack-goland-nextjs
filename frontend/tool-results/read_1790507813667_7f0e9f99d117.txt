    98→function initialsOf(name: string): string {
    99→  const parts = name.trim().split(/\s+/).filter(Boolean);
   100→  if (parts.length === 0) return "?";
   101→  return parts
   102→    .slice(0, 2)
   103→    .map((p) => p[0]!.toUpperCase())
   104→    .join("");
   105→}

   109→interface ParsedBackup {
   110→  payload: Record<string, unknown>;
   111→  resumeCount: number;
   112→  applicationCount: number | null; // null = absent, store will fall back to seed applications
   113→  scoreHistoryCount: number | null; // null = absent, store will fall back to seed score history

   117→function parseBackupText(text: string): { ok: true; data: ParsedBackup } | { ok: false; reason: string } {
   118→  let parsed: unknown;
   119→  try {
   120→    parsed = JSON.parse(text);
   121→  } catch {
   122→    return { ok: false, reason: "That file isn't valid JSON. Export a fresh backup from Settings and try again." };
   123→  }
   124→  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
   125→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — expected a JSON object." };
   126→  }
   127→  const obj = parsed as Record<string, unknown>;
   128→  const rawResumes = Array.isArray(obj.resumes) ? obj.resumes : null;
   129→  if (obj.app !== "ResumeForge AI" && !rawResumes) {
   130→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — the \"app\" marker and resumes array are missing." };
   131→  }
   132→  if (!rawResumes || rawResumes.length === 0) {
   133→    return { ok: false, reason: "The backup contains no resumes, so there is nothing to restore." };
   134→  }
   135→  for (let i = 0; i < rawResumes.length; i++) {
   136→    const entry: unknown = rawResumes[i];
   137→    if (!entry || typeof entry !== "object" || Array.isArray(entry)) {
   138→      return { ok: false, reason: `Resume #${i + 1} in the backup is malformed (expected an object).` };
   139→    }
   140→    const rec = entry as Record<string, unknown>;
   141→    const titleOk = typeof rec.title === "string";
   142→    if (
   143→      typeof rec.id !== "string" ||
   144→      rec.id.trim() === "" ||
   145→      !titleOk ||
   146→      !rec.personal ||
   147→      typeof rec.personal !== "object" ||
   148→      Array.isArray(rec.personal)
   149→    ) {
   150→      const label = titleOk ? `"${rec.title}"` : `#${i + 1}`;
   151→      return {
   152→        ok: false,
   153→        reason: `Resume ${label} is missing an id, a title or personal details — the backup looks corrupted.`,
   154→      };
   155→    }
   156→  }
   157→  return {
   158→    ok: true,
   159→    data: {
   160→      payload: obj,
   161→      resumeCount: rawResumes.length,
   162→      applicationCount: Array.isArray(obj.applications) ? obj.applications.length : null,
   163→      scoreHistoryCount: Array.isArray(obj.scoreHistory) ? obj.scoreHistory.length : null,
   164→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : null,
   165→    },
   166→  };
   167→}

   169→function formatBackupDate(iso: string | null): string {
   170→  if (!iso) return "Not recorded";
   171→  const date = new Date(iso);
   172→  if (!isValidDate(date)) return "Not recorded";
   173→  return format(date, "MMM d, yyyy 'at' h:mm a");
   174→}

   192→/** Number of notification ids marked as read in the bell's read map. */
   193→function countReadNotificationIds(): number {
   194→  if (typeof window === "undefined") return 0;
   195→  try {
   196→    const raw = window.localStorage.getItem(NOTIFS_READ_KEY);
   197→    if (!raw) return 0;
   198→    const parsed: unknown = JSON.parse(raw);
   199→    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return 0;
   200→    return Object.keys(parsed as Record<string, unknown>).length;
   201→  } catch {
   202→    return 0;
   203→  }
