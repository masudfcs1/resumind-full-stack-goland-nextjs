   300→   300→
   301→   301→export default function MyResumesPage() {
   302→   302→  const mounted = useMounted();
   303→   303→  const resumes = useResumeStore((s) => s.resumes);
   304→   304→  const activeResumeId = useResumeStore((s) => s.activeResumeId);
   305→   305→  const active = useActiveResume();
   306→   306→  const duplicateResume = useResumeStore((s) => s.duplicateResume);
   307→   307→  const deleteResume = useResumeStore((s) => s.deleteResume);
   308→   308→
   309→   309→  const [query, setQuery] = React.useState("");
   310→   310→  const [sort, setSort] = React.useState<SortKey>("recent");
   311→   311→  const [view, setView] = React.useState<ViewMode>("grid");
   312→   312→  const [createOpen, setCreateOpen] = React.useState(false);
   313→   313→  const [selectMode, setSelectMode] = React.useState(false);
   314→   314→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   315→   315→  const [confirmDelete, setConfirmDelete] = React.useState(false);
   316→   316→
   317→   317→  const scoreMap = React.useMemo(() => {
   318→   318→    const map = new Map<string, number>();
   319→   319→    for (const r of resumes) map.set(r.id, atsAnalyze(r).score);
   320→   320→    return map;
   321→   321→  }, [resumes]);
   322→   322→
   323→   323→  const visible = React.useMemo(() => {
   324→   324→    const q = query.trim().toLowerCase();
   325→   325→    const filtered = resumes.filter(
   326→   326→      (r) =>
   327→   327→        !q ||
   328→   328→        r.title.toLowerCase().includes(q) ||
   329→   329→        r.personal.fullName.toLowerCase().includes(q) ||
   330→   330→        r.personal.jobTitle.toLowerCase().includes(q)
   331→   331→    );
   332→   332→    const scoreOf = (id: string) => scoreMap.get(id) ?? 0;
   333→   333→    switch (sort) {
   334→   334→      case "name":
   335→   335→        return [...filtered].sort((a, b) => a.title.localeCompare(b.title));
   336→   336→      case "ats":
   337→   337→        return [...filtered].sort((a, b) => scoreOf(b.id) - scoreOf(a.id));
   338→   338→      default:
   339→   339→        return [...filtered].sort((a, b) => b.updatedAt - a.updatedAt);
   340→   340→    }
   341→   341→  }, [resumes, query, sort, scoreMap]);
   342→   342→
   343→   343→  /* Keep the selection honest when resumes disappear (per-card deletes, resets) */
   344→   344→  React.useEffect(() => {
   345→   345→    setSelected((prev) => {
   346→   346→      if (prev.size === 0) return prev;
   347→   347→      const ids = new Set(resumes.map((r) => r.id));
   348→   348→      const next = new Set([...prev].filter((id) => ids.has(id)));
   349→   349→      return next.size === prev.size ? prev : next;
   350→   350→    });
   351→   351→  }, [resumes]);
   352→   352→
   353→   353→  const exitSelectMode = React.useCallback(() => {
   354→   354→    setSelectMode(false);
   355→   355→    setSelected(new Set());
   356→   356→  }, []);
   357→   357→
   358→   358→  const toggleSelect = React.useCallback((id: string) => {
   359→   359→    setSelected((prev) => {
   360→   360→      const next = new Set(prev);
   361→   361→      if (next.has(id)) next.delete(id);
   362→   362→      else next.add(id);
   363→   363→      return next;
   364→   364→    });
   365→   365→  }, []);
   366→   366→
   367→   367→  /* ESC exits selection mode — but never while an AlertDialog (bulk delete or
   368→   368→     per-card delete) is open. The listener runs in the CAPTURE phase so it sees
   369→   369→     the dialog before Radix's document-level handler closes it: Radix calls
   370→   370→     onOpenChange(false) synchronously during the Escape dispatch (React 19
   371→   371→     flush), so a bubble-phase DOM/state check would already observe the dialog
   372→   372→     as gone and wrongly exit selection mode. */
   373→   373→  React.useEffect(() => {
   374→   374→    if (!selectMode) return;
   375→   375→    const onKey = (e: KeyboardEvent) => {
   376→   376→      if (e.key !== "Escape") return;
   377→   377→      if (document.querySelector("[role=alertdialog]")) return;
   378→   378→      exitSelectMode();
   379→   379→    };
   380→   380→    window.addEventListener("keydown", onKey, true);
   381→   381→    return () => window.removeEventListener("keydown", onKey, true);
   382→   382→  }, [selectMode, exitSelectMode]);
   383→   383→
   384→   384→  const handleBulkDuplicate = () => {
   385→   385→    let ok = 0;
   386→   386→    for (const id of selected) {
   387→   387→      if (duplicateResume(id)) ok += 1;
   388→   388→    }
   389→   389→    if (ok > 0) {
   390→   390→      toast.success(`Duplicated ${ok} ${ok === 1 ? "resume" : "resumes"}`, {
   391→   391→        description: "Copies were added to the top of your list.",
   392→   392→      });
   393→   393→    }
   394→   394→    /* Selection mode stays ON and the same set stays selected so users can continue. */
   395→   395→  };
   396→   396→
   397→   397→  const handleBulkDelete = () => {
   398→   398→    const ids = [...selected];
   399→   399→    for (const id of ids) deleteResume(id);
   400→   400→    if (ids.length > 0) {
   401→   401→      toast.success(`Deleted ${ids.length} ${ids.length === 1 ? "resume" : "resumes"}`);
   402→   402→    }
   403→   403→    setConfirmDelete(false);
   404→   404→    setSelected(new Set());
   405→   405→    setSelectMode(false);
   406→   406→  };
   407→   407→
   408→   408→  if (!mounted) {
   409→   409→    return <ResumesSkeleton />;
   410→   410→  }
   411→   411→
   412→   412→  const hasAny = resumes.length > 0;
   413→   413→  const selectedCount = selected.size;
   414→   414→  const allVisibleSelected =
   415→   415→    visible.length > 0 && visible.every((r) => selected.has(r.id));
   416→   416→  const toggleSelectAll = () => {
   417→   417→    setSelected(allVisibleSelected ? new Set() : new Set(visible.map((r) => r.id)));
   418→   418→  };
   419→   419→
   420→   420→  return (
   421→   421→    <div className="mx-auto max-w-6xl">
   422→   422→      {/* Header row */}
   423→   423→      <header className="flex flex-wrap items-center gap-3">
   424→   424→        <div className="min-w-0">
   425→   425→          <h1 className="flex items-center gap-2 font-display text-2xl font-bold tracking-tight">
   426→   426→            My Resumes
   427→   427→            <Badge
   428→   428→              variant="secondary"
   429→   429→              className="rounded-full bg-emerald-500/10 text-emerald-700 dark:text-emerald-400"
   430→   430→            >
   431→   431→              {resumes.length}
   432→   432→            </Badge>
   433→   433→          </h1>
   434→   434→          <p className="mt-0.5 text-[13px] text-muted-foreground">
   435→   435→            Manage, duplicate and track every version of your resume.
   436→   436→          </p>
   437→   437→        </div>
   438→   438→
   439→   439→        <div className="ml-auto flex flex-wrap items-center gap-2">
   440→   440→          <div className="relative w-full sm:w-52">
   441→   441→            <Search
   442→   442→              className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground"
   443→   443→              aria-hidden="true"
   444→   444→            />
   445→   445→            <Input
   446→   446→              value={query}
   447→   447→              onChange={(e) => setQuery(e.target.value)}
   448→   448→              placeholder="Search resumes…"
   449→   449→              aria-label="Search resumes"
   450→   450→              className="h-9 rounded-lg pl-8"
   451→   451→            />
   452→   452→          </div>
   453→   453→
