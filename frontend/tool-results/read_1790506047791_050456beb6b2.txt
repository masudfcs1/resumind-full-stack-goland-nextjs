    91→    91→function initialsOf(name: string): string {
    92→    92→  const parts = name.trim().split(/\s+/).filter(Boolean);
    93→    93→  if (parts.length === 0) return "?";
    94→    94→  return parts
    95→    95→    .slice(0, 2)
    96→    96→    .map((p) => p[0]!.toUpperCase())
    97→    97→    .join("");
    98→    98→}

   102→   102→interface ParsedBackup {
   103→   103→  payload: Record<string, unknown>;
   104→   104→  resumeCount: number;
   105→   105→  applicationCount: number | null; // null = absent, store will fall back to seed applications

   109→   109→
   110→   110→function parseBackupText(text: string): { ok: true; data: ParsedBackup } | { ok: false; reason: string } {
   111→   111→  let parsed: unknown;
   112→   112→  try {
   113→   113→    parsed = JSON.parse(text);
   114→   114→  } catch {
   115→   115→    return { ok: false, reason: "That file isn't valid JSON. Export a fresh backup from Settings and try again." };
   116→   116→  }
   117→   117→  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
   118→   118→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — expected a JSON object." };
   119→   119→  }
   120→   120→  const obj = parsed as Record<string, unknown>;
   121→   121→  const rawResumes = Array.isArray(obj.resumes) ? obj.resumes : null;
   122→   122→  if (obj.app !== "ResumeForge AI" && !rawResumes) {
   123→   123→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — the \"app\" marker and resumes array are missing." };
   124→   124→  }
   125→   125→  if (!rawResumes || rawResumes.length === 0) {
   126→   126→    return { ok: false, reason: "The backup contains no resumes, so there is nothing to restore." };
   127→   127→  }
   128→   128→  for (let i = 0; i < rawResumes.length; i++) {
   129→   129→    const entry: unknown = rawResumes[i];
   130→   130→    if (!entry || typeof entry !== "object" || Array.isArray(entry)) {
   131→   131→      return { ok: false, reason: `Resume #${i + 1} in the backup is malformed (expected an object).` };
   132→   132→    }
   133→   133→    const rec = entry as Record<string, unknown>;
   134→   134→    const titleOk = typeof rec.title === "string";
   135→   135→    if (
   136→   136→      typeof rec.id !== "string" ||
   137→   137→      rec.id.trim() === "" ||
   138→   138→      !titleOk ||
   139→   139→      !rec.personal ||
   140→   140→      typeof rec.personal !== "object" ||
   141→   141→      Array.isArray(rec.personal)
   142→   142→    ) {
   143→   143→      const label = titleOk ? `"${rec.title}"` : `#${i + 1}`;
   144→   144→      return {
   145→   145→        ok: false,
   146→   146→        reason: `Resume ${label} is missing an id, a title or personal details — the backup looks corrupted.`,
   147→   147→      };
   148→   148→    }
   149→   149→  }
   150→   150→  return {
   151→   151→    ok: true,
   152→   152→    data: {
   153→   153→      payload: obj,
   154→   154→      resumeCount: rawResumes.length,
   155→   155→      applicationCount: Array.isArray(obj.applications) ? obj.applications.length : null,
   156→   156→      scoreHistoryCount: Array.isArray(obj.scoreHistory) ? obj.scoreHistory.length : null,
   157→   157→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : null,
   158→   158→    },
   159→   159→  };
   160→   160→}

   162→   162→function formatBackupDate(iso: string | null): string {
   163→   163→  if (!iso) return "Not recorded";
   164→   164→  const date = new Date(iso);
   165→   165→  if (!isValidDate(date)) return "Not recorded";
   166→   166→  return format(date, "MMM d, yyyy 'at' h:mm a");
   167→   167→}
