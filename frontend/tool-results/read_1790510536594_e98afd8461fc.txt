   108→/* ============================== Placeholder resolution ============================== */
   109→
   110→/** Replaces {company} / {skill} / {role} with data from the active resume. */
   111→export function resolvePlaceholders(text: string, resume: ResumeData): string {
   112→  const company = resume.experience[0]?.company || "your current company";
   113→  const skill = resume.skills[0]?.name || "your core skill";
   114→  const role = resume.personal.jobTitle || resume.title || "this role";
   115→  return text
   116→    .replace(/\{company\}/g, company)
   117→    .replace(/\{skill\}/g, skill)
   118→    .replace(/\{role\}/g, role);
   119→}
   120→
   121→function resolveStar(star: StarAnswer, resume: ResumeData): StarAnswer {
   122→  return {
   123→    situation: resolvePlaceholders(star.situation, resume),
   124→    task: resolvePlaceholders(star.task, resume),
   125→    action: resolvePlaceholders(star.action, resume),
   126→    result: resolvePlaceholders(star.result, resume),
   127→  };
   128→}

   946→function shuffle<T>(input: readonly T[]): T[] {
   947→  const arr = [...input];
   948→  for (let i = arr.length - 1; i > 0; i--) {
   949→    const j = Math.floor(Math.random() * (i + 1));
   950→    const a = arr[i];
   951→    const b = arr[j];
   952→    if (a !== undefined && b !== undefined) {
   953→      arr[i] = b;
   954→      arr[j] = a;
   955→    }
   956→  }
   957→  return arr;
   958→}

   960→function instantiate(question: Question, resume: ResumeData): PracticeQuestion {
   961→  const tailoredText = question.tailored?.(resume);
   962→  const useTailored = typeof tailoredText === "string" && tailoredText.length > 0;
   963→  return {
   964→    ...question,
   965→    text:
   966→      useTailored && tailoredText
   967→        ? tailoredText
   968→        : resolvePlaceholders(question.question, resume),
   969→    isTailored: useTailored,
   970→    resolvedStar: resolveStar(question.star, resume),
   971→  };
   972→}

   974→/**
   975→ * Builds a practice set: filters the bank by category + difficulty, shuffles,
   976→ * prefers resume-tailored variants, and round-robins across categories so
   977→ * every selected category is represented when possible.
   978→ */
   979→export function generateQuestions(
   980→  resume: ResumeData,
   981→  { categories, difficulty, count = 8 }: GenerateOptions
   982→): PracticeQuestion[] {
   983→  const allowed = categories.length > 0 ? categories : CATEGORY_ORDER;
   984→  const pool = QUESTION_BANK.filter(
   985→    (q) =>
   986→      allowed.includes(q.category) &&
   987→      (difficulty === "all" || q.difficulty === difficulty)
   988→  );
   989→
   990→  // One shuffled, tailored-first bucket per selected category.
   991→  const buckets = new Map<QuestionCategory, Question[]>();
   992→  for (const cat of allowed) {
   993→    const inCat = shuffle(pool.filter((q) => q.category === cat)).sort(
   994→      (a, b) => Number(Boolean(b.tailored)) - Number(Boolean(a.tailored))
   995→    );
   996→    if (inCat.length > 0) buckets.set(cat, inCat);
   997→  }
   998→
   999→  const picked: Question[] = [];
  1000→  while (picked.length < count && buckets.size > 0) {
  1001→    for (const [cat, queue] of buckets) {
  1002→      const next = queue.shift();
  1003→      if (next) picked.push(next);
  1004→      if (queue.length === 0) buckets.delete(cat);
  1005→      if (picked.length >= count) break;
  1006→    }
  1007→  }
  1008→
  1009→  return picked.map((q) => instantiate(q, resume));
  1010→}
