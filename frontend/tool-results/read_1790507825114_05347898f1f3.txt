    42→function CardShell({

    44→  className,
    45→  delay = 0,

    47→  cta,
    48→  label,
    49→}: {

    66→          linked &&
    67→            "cursor-pointer focus-within:border-emerald-500/70 focus-within:ring-2 focus-within:ring-emerald-500/30"
    68→        )}
    69→      >
    70→        {children}

    72→          <>
    73→            {/* CTA pill — space is reserved in-flow (no layout shift), revealed on hover / keyboard focus */}

    75→              aria-hidden
    76→              className="pointer-events-none mt-auto flex items-center pt-4"
    77→            >
    78→              <span className="inline-flex translate-y-1 items-center gap-1.5 rounded-full bg-emerald-500/10 px-3 py-1.5 text-[11px] font-semibold text-emerald-600 opacity-0 transition-all duration-300 group-focus-within:translate-y-0 group-focus-within:opacity-100 group-hover:translate-y-0 group-hover:opacity-100 dark:text-emerald-400">

   425→/* "+12 pts" pill that counts up when scrolled into view */
   426→function DeltaPill() {
   427→  const ref = React.useRef<HTMLSpanElement>(null);
   428→  const inView = useInView(ref, { once: true, margin: "-60px" });
   429→  const mv = useMotionValue(0);
   430→  const text = useTransform(mv, (v: number) => `+${Math.round(v)} pts`);
   431→
   432→  React.useEffect(() => {
   433→    if (!inView) return;
   434→    const controls = animate(mv, 12, { duration: 1.4, ease: "easeOut", delay: 0.5 });
   435→    return () => controls.stop();
   436→  }, [inView, mv]);
   437→
   438→  return (
   439→    <span
   440→      ref={ref}
   441→      className="absolute -bottom-3 left-1/2 z-10 inline-flex -translate-x-1/2 items-center gap-1 rounded-full border border-emerald-500/30 bg-background/90 px-2.5 py-1 text-[10px] font-extrabold text-emerald-600 shadow-sm backdrop-blur-sm dark:border-emerald-400/25 dark:text-emerald-400"
   442→    >
   443→      <TrendingUp className="size-3" aria-hidden />
   444→      <motion.span className="tabular-nums">{text}</motion.span>
   445→    </span>
   446→  );
   447→}
