   157→export function formatMonth(value: string): string {
   158→  if (!value) return "";
   159→  const [y, m] = value.split("-");
   160→  if (!y) return value;
   161→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   162→  if (!m) return y;
   163→  const idx = parseInt(m, 10) - 1;
   164→  return `${months[idx] ?? ""} ${y}`;
   165→}

   611→    at: now - day * daysAgo,
   612→  });
   613→  return [
   614→    mk("seed-alex", "Senior Frontend Engineer — Google", 71, 20),
   615→    mk("seed-alex", "Senior Frontend Engineer — Google", 78, 14),
   616→    mk("seed-maya", "Product Manager — Stripe", 74, 11),
   617→    mk("seed-alex", "Senior Frontend Engineer — Google", 82, 9),
   618→    mk("seed-david", "Data Scientist — OpenAI", 80, 7),
   619→    mk("seed-maya", "Product Manager — Stripe", 80, 5),
   620→    mk("seed-david", "Data Scientist — OpenAI", 86, 3),
   621→    mk("seed-alex", "Senior Frontend Engineer — Google", 87, 1),
   622→  ];

   625→/* ============================== Undo buffer (module-local, NOT persisted) ============================== */
   626→
   627→/* Captures the most recent delete action so the Undo button on the delete
   628→   toast can put the resumes back exactly where they were. Deliberately kept
   629→   OUTSIDE the persisted store: undo data is transient and must never reach
   630→   localStorage or widen the persisted schema. */
   631→let undoBuffer: DeletedResumeEntry[] = [];
   632→let undoBatchIds: string[] = [];
   633→let undoBatchSealed = false;
   634→
   635→/* deleteResume may fire several times for ONE user action (multi-select
   636→   delete loops over the selected ids). Calls made within the same synchronous
   637→   task share one batch: the first call snapshots the pre-delete id order so
   638→   every captured entry keeps its ORIGINAL position even though the array
   639→   shrinks between calls; the next task starts a fresh batch. */
   640→function openUndoBatch(): void {
   641→  if (undoBatchSealed) return;
   642→  undoBuffer = [];
   643→  undoBatchIds = useResumeStore.getState().resumes.map((r) => r.id);
   644→  undoBatchSealed = true;
   645→  queueMicrotask(() => {
   646→    undoBatchSealed = false;
   647→  });
   648→}
   649→
   650→/** Returns (and clears) the snapshots captured for the most recent delete
   651→    action. Feed the result straight into showDeleteUndoToast. */
   652→export function takeUndoEntries(): DeletedResumeEntry[] {
   653→  const entries = undoBuffer;
   654→  undoBuffer = [];
   655→  return entries;
   656→}
   657→
   658→/** Standard "deleted" toast with a 7s Undo action that restores the given
   659→    entries at their original positions (restoreResumes). */
   660→export function showDeleteUndoToast(entries: DeletedResumeEntry[]): void {
   661→  const n = entries.length;
   662→  if (n === 0) return;
   663→  toast.success(n === 1 ? "Resume deleted" : `Deleted ${n} resumes`, {
   664→    description:
   665→      n === 1
   666→        ? `“${entries[0]!.resume.title}” was removed from your workspace.`
   667→        : `${n} resumes were removed from your workspace.`,
   668→    duration: 7000,
   669→    action: {
   670→      label: "Undo",
   671→      onClick: () => {
   672→        useResumeStore.getState().restoreResumes(entries);
   673→        toast.success(n === 1 ? "Resume restored" : `Restored ${n} resumes`, {
   674→          description: n === 1 ? "Back at its original position." : "Back at their original positions.",
   675→        });
   676→      },
   677→    },
   678→  });
   679→}

   681→/* ============================== Store ============================== */
   682→
   683→interface ResumeState {
   684→  resumes: ResumeData[];
   685→  activeResumeId: string;
   686→  coverLetterCount: number;
   687→  applications: JobApplication[];
   688→  scoreHistory: ScoreEntry[];
   689→  createResume: (title?: string, template?: TemplateId, accent?: string) => string;
   690→  duplicateResume: (id: string) => string | undefined;
   691→  deleteResume: (id: string) => void;
   692→  /** Soft-hide: keeps the resume in the store but flags it archived. */
   693→  archiveResume: (id: string) => void;
   694→  /** Reverse of archiveResume — clears the archived flag. */
   695→  unarchiveResume: (id: string) => void;
   696→  /** Batch archive/unarchive for the resume select-mode bulk toolbar.
   697→      Idempotent per resume: flags already in the target state are left so
   698→      their updatedAt is untouched. */
   699→  bulkSetArchived: (ids: string[], archived: boolean) => void;
   700→  /** Splices previously deleted resumes back at their original positions
   701→      (used by the Undo action on delete toasts). Idempotent: entries whose id
   702→      is already present are skipped. */
   703→  restoreResumes: (entries: DeletedResumeEntry[]) => void;
   704→  setActive: (id: string) => void;
   705→  updateResume: (id: string, patch: Partial<ResumeData>) => void;
   706→  resetAll: () => void;
   707→  /** Validates a parsed backup payload and replaces the workspace with it. Returns the number of resumes imported, or -1 if the payload is invalid. */
   708→  importData: (payload: unknown) => number;
   709→  incrementCoverLetters: () => void;
   710→  addApplication: (app: Omit<JobApplication, "id" | "appliedAt" | "updatedAt">) => string;
   711→  updateApplication: (id: string, patch: Partial<JobApplication>) => void;
   712→  moveApplication: (id: string, stage: ApplicationStage) => void;
   713→  deleteApplication: (id: string) => void;
   714→  logScore: (
   715→    resumeId: string,
   716→    resumeTitle: string,
   717→    score: number,
   718→    categories?: Record<string, number>
   719→  ) => void;
   720→}
   721→
   722→export const useResumeStore = create<ResumeState>()(
   723→  persist(
   724→    (set, get) => ({
   725→      resumes: [seedResume1(), seedResume2(), seedResume3()],
   726→      activeResumeId: "seed-alex",
   727→      coverLetterCount: 2,
   728→      applications: seedApplications(),
   729→      scoreHistory: seedScoreHistory(),
   730→      createResume: (title, template, accent) => {
   731→        const id = uid();
   732→        const base = emptyResume(title ?? "Untitled Resume", template ?? "modern", accent ?? "#10b981");
   733→        const resume: ResumeData = { ...base, id, updatedAt: Date.now() };
   734→        set((s) => ({ resumes: [resume, ...s.resumes], activeResumeId: id }));
   735→        return id;
   736→      },
   737→      duplicateResume: (id) => {
   738→        const src = get().resumes.find((r) => r.id === id);
   739→        if (!src) return undefined;
   740→        const newId = uid();
   741→        const copy: ResumeData = {
   742→          ...structuredClone(src),
   743→          id: newId,
   744→          title: `${src.title} (Copy)`,
   745→          updatedAt: Date.now(),
   746→          /* Copies always start life in the active list, even if the source was archived. */
   747→          archived: false,
   748→        };
   749→        set((s) => ({ resumes: [copy, ...s.resumes] }));
   750→        return newId;
   751→      },
   752→      deleteResume: (id) => {
   753→        const s = get();
   754→        const index = s.resumes.findIndex((r) => r.id === id);
   755→        if (index === -1) return;
   756→        /* Capture BEFORE removal (deep copy + original position, anchored to
   757→           the batch's pre-delete order) so the Undo toast can restore it. */
   758→        openUndoBatch();
   759→        const batchIndex = undoBatchIds.indexOf(id);
   760→        undoBuffer.push({
   761→          resume: structuredClone(s.resumes[index]!),
   762→          index: batchIndex !== -1 ? batchIndex : index,
   763→        });
   764→        const resumes = s.resumes.filter((r) => r.id !== id);
   765→        set({
   766→          resumes,
   767→          activeResumeId:
   768→            s.activeResumeId === id ? (resumes[0]?.id ?? "") : s.activeResumeId,
   769→        });
   770→      },
   771→      /* Archive bumps updatedAt like every other mutation (store convention).
   772→         The active pointer is intentionally left untouched when archiving the
   773→         active resume — it still resolves (the resume stays in the store), so
   774→         the builder/ATS binding never breaks. */
   775→      archiveResume: (id) =>
   776→        set((s) => ({
   777→          resumes: s.resumes.map((r) =>
   778→            r.id === id ? { ...r, archived: true, updatedAt: Date.now() } : r
   779→          ),
   780→        })),
   781→      unarchiveResume: (id) =>
   782→        set((s) => ({
   783→          resumes: s.resumes.map((r) =>
   784→            r.id === id ? { ...r, archived: false, updatedAt: Date.now() } : r
   785→          ),
   786→        })),
   787→      bulkSetArchived: (ids, archived) =>
   788→        set((s) => {
   789→          const wanted = new Set(ids);
   790→          return {
   791→            resumes: s.resumes.map((r) =>
   792→              wanted.has(r.id) && r.archived !== archived
   793→                ? { ...r, archived, updatedAt: Date.now() }
   794→                : r
   795→            ),
   796→          };
   797→        }),
   798→      restoreResumes: (entries) =>
   799→        set((s) => {
   800→          const resumes = [...s.resumes];
   801→          const present = new Set(resumes.map((r) => r.id));
   802→          /* Splice in descending index order so every entry lands exactly
   803→             where it came from, regardless of how many were removed. */
   804→          const ordered = [...entries].sort((a, b) => b.index - a.index);
   805→          for (const entry of ordered) {
   806→            if (present.has(entry.resume.id)) continue;
   807→            const at = Math.min(Math.max(entry.index, 0), resumes.length);
   808→            resumes.splice(at, 0, structuredClone(entry.resume));
   809→            present.add(entry.resume.id);
   810→          }
   811→          return { resumes };
   812→        }),
   813→      setActive: (id) => set({ activeResumeId: id }),
   814→      updateResume: (id, patch) =>
   815→        set((s) => ({
   816→          resumes: s.resumes.map((r) =>
   817→            r.id === id ? { ...r, ...patch, updatedAt: Date.now() } : r
   818→          ),
   819→        })),
   820→      resetAll: () =>
   821→        set({
   822→          resumes: [seedResume1(), seedResume2(), seedResume3()],
   823→          activeResumeId: "seed-alex",
   824→          coverLetterCount: 2,
   825→          applications: seedApplications(),
   826→          scoreHistory: seedScoreHistory(),
   827→        }),
   828→      importData: (payload) => {
   829→        if (!payload || typeof payload !== "object" || Array.isArray(payload)) return -1;
   830→        const raw = payload as Record<string, unknown>;
   831→
   832→        /* Resumes are required — every entry needs an id, a title and a personal object. */
   833→        const rawResumes = raw.resumes;
   834→        if (!Array.isArray(rawResumes) || rawResumes.length === 0) return -1;
   835→        const blank = emptyResume();
   836→        const resumes: ResumeData[] = [];
   837→        for (const entry of rawResumes) {
   838→          if (!entry || typeof entry !== "object" || Array.isArray(entry)) return -1;
   839→          const r = entry as Record<string, unknown>;
   840→          if (typeof r.id !== "string" || r.id.trim() === "") return -1;
   841→          if (typeof r.title !== "string") return -1;
   842→          if (!r.personal || typeof r.personal !== "object" || Array.isArray(r.personal)) return -1;
   843→          const copy = structuredClone(entry) as ResumeData;
   844→          copy.template = typeof r.template === "string" && r.template in TEMPLATE_META ? (r.template as TemplateId) : blank.template;
   845→          copy.accent = typeof r.accent === "string" && r.accent.trim() !== "" ? r.accent : blank.accent;
   846→          copy.summary = typeof r.summary === "string" ? r.summary : blank.summary;
   847→          copy.personal = { ...blank.personal, ...(r.personal as Partial<PersonalInfo>) };
   848→          for (const key of ["experience", "education", "skills", "projects", "certifications", "languages"] as const) {
   849→            if (!Array.isArray(copy[key])) copy[key] = [];
   850→          }
   851→          copy.updatedAt = typeof r.updatedAt === "number" && Number.isFinite(r.updatedAt) ? r.updatedAt : Date.now();
   852→          resumes.push(copy);
   853→        }
   854→
   855→        /* Active resume falls back to the first import if the id is missing or unknown. */
   856→        const wantedActive = typeof raw.activeResumeId === "string" ? raw.activeResumeId : "";
   857→        const firstId = resumes[0]!.id;
   858→        const activeResumeId = resumes.some((r) => r.id === wantedActive) ? wantedActive : firstId;
   859→
   860→        const coverLetterCount =
   861→          typeof raw.coverLetterCount === "number" && Number.isFinite(raw.coverLetterCount)
   862→            ? Math.max(0, Math.round(raw.coverLetterCount))
   863→            : 0;
   864→
   865→        /* Applications and score history fall back to seeds when absent (mirrors resetAll). */
   866→        let applications = seedApplications();
   867→        if (Array.isArray(raw.applications)) {
   868→          const clean: JobApplication[] = [];
   869→          for (const entry of raw.applications) {
   870→            if (!entry || typeof entry !== "object" || Array.isArray(entry)) continue;
   871→            const a = entry as Record<string, unknown>;
   872→            clean.push({
   873→              id: typeof a.id === "string" && a.id !== "" ? a.id : uid(),
   874→              company: typeof a.company === "string" ? a.company : "",
   875→              role: typeof a.role === "string" ? a.role : "",
   876→              location: typeof a.location === "string" ? a.location : "",
   877→              salary: typeof a.salary === "string" ? a.salary : "",
   878→              url: typeof a.url === "string" ? a.url : "",
   879→              stage: typeof a.stage === "string" && a.stage in STAGE_META ? (a.stage as ApplicationStage) : "applied",
   880→              resumeId: typeof a.resumeId === "string" ? a.resumeId : "",
   881→              notes: typeof a.notes === "string" ? a.notes : "",
   882→              appliedAt: typeof a.appliedAt === "number" && Number.isFinite(a.appliedAt) ? a.appliedAt : Date.now(),
   883→              updatedAt: typeof a.updatedAt === "number" && Number.isFinite(a.updatedAt) ? a.updatedAt : Date.now(),
   884→            });
   885→          }
   886→          if (clean.length > 0) applications = clean;
   887→        }
   888→
   889→        let scoreHistory = seedScoreHistory();
   890→        if (Array.isArray(raw.scoreHistory)) {
   891→          const clean: ScoreEntry[] = [];
   892→          for (const entry of raw.scoreHistory) {
   893→            if (!entry || typeof entry !== "object" || Array.isArray(entry)) continue;
   894→            const e = entry as Record<string, unknown>;
   895→            if (typeof e.score !== "number" || !Number.isFinite(e.score)) continue;
   896→            clean.push({
   897→              id: typeof e.id === "string" && e.id !== "" ? e.id : uid(),
   898→              resumeId: typeof e.resumeId === "string" ? e.resumeId : "",
   899→              resumeTitle: typeof e.resumeTitle === "string" ? e.resumeTitle : "Untitled",
   900→              score: e.score,
   901→              at: typeof e.at === "number" && Number.isFinite(e.at) ? e.at : Date.now(),
   902→            });
   903→          }
   904→          if (clean.length > 0) scoreHistory = clean.slice(-60);
   905→        }
   906→
   907→        set({ resumes, activeResumeId, coverLetterCount, applications, scoreHistory });
   908→        return resumes.length;
   909→      },
   910→      incrementCoverLetters: () => set((s) => ({ coverLetterCount: s.coverLetterCount + 1 })),
   911→      addApplication: (app) => {
   912→        const id = uid();
   913→        const full: JobApplication = {
   914→          ...app,
   915→          id,
   916→          appliedAt: Date.now(),
   917→          updatedAt: Date.now(),
   918→        };
   919→        set((s) => ({ applications: [full, ...s.applications] }));
   920→        return id;
   921→      },
   922→      updateApplication: (id, patch) =>
   923→        set((s) => ({
   924→          applications: s.applications.map((a) =>
   925→            a.id === id ? { ...a, ...patch, updatedAt: Date.now() } : a
   926→          ),
   927→        })),
   928→      moveApplication: (id, stage) =>
   929→        set((s) => ({
   930→          applications: s.applications.map((a) =>
   931→            a.id === id ? { ...a, stage, updatedAt: Date.now() } : a
   932→          ),
   933→        })),
   934→      deleteApplication: (id) =>
   935→        set((s) => ({ applications: s.applications.filter((a) => a.id !== id) })),
   936→      logScore: (resumeId, resumeTitle, score, categories) =>
   937→        set((s) => ({
   938→          scoreHistory: [
   939→            ...s.scoreHistory,
   940→            { id: uid(), resumeId, resumeTitle, score, at: Date.now(), categories },
   941→          ].slice(-60),
   942→        })),
   943→    }),
   944→    {
   945→      name: "resumeforge-store",
   946→      storage: createJSONStorage(() => localStorage),
   947→      // Rehydrate AFTER React hydration (see <StoreRehydrator /> in root layout).
   948→      // Synchronous rehydration makes the first client render read localStorage
   949→      // while the server rendered seed data — any divergence (extra resumes,
   950→      // archived flags, renamed profile…) shifts the render tree and breaks
   951→      // hydration (Radix useId mismatches on every dashboard route).
   952→      skipHydration: true,
   953→    }
   954→  )
   955→);

   960→}
