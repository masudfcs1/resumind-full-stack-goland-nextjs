   172→   172→
   173→   173→function AtsScanner() {
   174→   174→  const mounted = useMounted();
   175→   175→  const searchParams = useSearchParams();
   176→   176→  const router = useRouter();
   177→   177→  const resumes = useResumeStore((s) => s.resumes);
   178→   178→  const scoreHistory = useResumeStore((s) => s.scoreHistory);
   179→   179→  const activeResumeId = useResumeStore((s) => s.activeResumeId);
   180→   180→  const setActive = useResumeStore((s) => s.setActive);
   181→   181→  const logScore = useResumeStore((s) => s.logScore);
   182→   182→
   183→   183→  const resume =
   184→   184→    resumes.find((r) => r.id === activeResumeId) ?? resumes[0];
   185→   185→
   186→   186→  const [jd, setJd] = React.useState("");
   187→   187→  const [jdOpen, setJdOpen] = React.useState(false);
   188→   188→  const [scanning, setScanning] = React.useState(false);
   189→   189→  const [result, setResult] = React.useState<AtsResult | null>(null);
   190→   190→  const [scanId, setScanId] = React.useState(0);
   191→   191→  /** Last history entry that existed BEFORE the live scan (the "old" side). */
   192→   192→  const [baseline, setBaseline] = React.useState<ScoreEntry | null>(null);
   193→   193→
   194→   194→  const usedJobDescription = jd.trim().length > 40;
   195→   195→
   196→   196→  /* --- Shareable ?resume=<id> param --- */
   197→   197→
   198→   198→  const paramResumeId = searchParams.get("resume");
   199→   199→
   200→   200→  // URL → store: a valid ?resume= preselects that resume (falls back to the
   201→   201→  // store's active resume, then the first resume, when missing/invalid).
   202→   202→  React.useEffect(() => {
   203→   203→    if (!mounted) return;
   204→   204→    // window.location fallback: on a fresh load the router's searchParams
   205→   205→    // (hook) settle a beat later than the address bar — without this, a
   206→   206→    // deep link would be missed entirely.
   207→   207→    const urlId =
   208→   208→      paramResumeId ?? new URLSearchParams(window.location.search).get("resume");
   209→   209→    if (!urlId) return;
   210→   210→    if (!resumes.some((r) => r.id === urlId)) return;
   211→   211→    if (urlId !== activeResumeId) setActive(urlId);
   212→   212→  }, [mounted, paramResumeId, activeResumeId, resumes, setActive]);
   213→   213→
   214→   214→  // Store → URL: keep ?resume= in sync with the selection so any scan state is
   215→   215→  // shareable (replace — the back button stays clean). Also heals an invalid
   216→   216→  // or missing param to the resume actually shown.
   217→   217→  React.useEffect(() => {
   218→   218→    if (!mounted || !activeResumeId) return;
   219→   219→    const locParams = new URLSearchParams(window.location.search);
   220→   220→    // Router state not ready yet (hook still empty while the address bar
   221→   221→    // carries a ?resume=) — never "heal" a deep link before the hook has had
   222→   222→    // the chance to see it.
   223→   223→    if (searchParams.get("resume") === null && locParams.get("resume")) return;
   224→   224→    if (locParams.get("resume") === activeResumeId) return;
   225→   225→    locParams.set("resume", activeResumeId);
   226→   226→    router.replace(`/dashboard/ats?${locParams.toString()}`, { scroll: false });
   227→   227→  }, [mounted, activeResumeId, router, searchParams]);
   228→   228→
   229→   229→  /* --- Scan state --- */
   230→   230→
   231→   231→  const resumeId = resume?.id ?? "";
   232→   232→
   233→   233→  // A finished scan always describes the currently selected resume: switching
   234→   234→  // resumes clears the previous result so the delta panel starts fresh.
   235→   235→  React.useEffect(() => {
   236→   236→    if (!mounted) return;
   237→   237→    setResult(null);
   238→   238→    setBaseline(null);
   239→   239→  }, [mounted, resumeId]);
   240→   240→
   241→   241→  const handleAnalyze = async () => {
   242→   242→    if (!resume || scanning) return;
   243→   243→    const target = resume;
   244→   244→    setScanning(true);
   245→   245→    try {
   246→   246→      await sleep(1200);
   247→   247→      // Ignore stale completions if the user switched resumes mid-scan.
   248→   248→      if (useResumeStore.getState().activeResumeId !== target.id) return;
   249→   249→      const res = atsAnalyze(target, jd.trim() || undefined);
   250→   250→      // Capture the pre-scan "old" side before this scan is logged.
   251→   251→      setBaseline(
   252→   252→        lastScoreEntryFor(
   253→   253→          useResumeStore.getState().scoreHistory,
   254→   254→          target.id
   255→   255→        )
   256→   256→      );
   257→   257→      setResult(res);
   258→   258→      setScanId((n) => n + 1);
   259→   259→      logScore(target.id, target.title, res.score);
   260→   260→      toast.success(`Scan complete — ATS score ${res.score}/100`, {
   261→   261→        description: "Score logged to your dashboard history.",
   262→   262→      });
   263→   263→    } catch {
   264→   264→      toast.error("Something went wrong while scanning. Try again.");
   265→   265→    } finally {
   266→   266→      setScanning(false);
   267→   267→    }
   268→   268→  };
   269→   269→
   270→   270→  /* --- vs last scan (delta) --- */
   271→   271→
   272→   272→  // Before any scan: the resume's latest history entry, if one exists.
   273→   273→  const loadBaseline = React.useMemo(
   274→   274→    () =>
   275→   275→      mounted && resume && !result
   276→   276→        ? lastScoreEntryFor(scoreHistory, resume.id)
   277→   277→        : null,
   278→   278→    [mounted, resume, result, scoreHistory]
   279→   279→  );
   280→   280→
   281→   281→  const panelEntry = result ? baseline : loadBaseline;
   282→   282→  const showDeltaPanel = Boolean(mounted && resume && panelEntry);
   283→   283→  const deltaDiff = result && panelEntry ? result.score - panelEntry.score : 0;
   284→   284→  const deltaKindValue = deltaKind(deltaDiff);
   285→   285→
   286→   286→  return (
   287→   287→    <motion.div
   288→   288→      variants={staggerContainer}
   289→   289→      initial="hidden"
   290→   290→      animate="show"
   291→   291→      className="mx-auto w-full max-w-6xl space-y-6"
   292→   292→    >
   293→   293→      <ToolPageHeader
   294→   294→        icon={Target}
   295→   295→        title="ATS Score Scanner"
   296→   296→        description="Run your resume through our simulated applicant tracking system. Get an instant score, a category breakdown, and concrete fixes."
   297→   297→        actions={
   298→   298→          <>
   299→   299→            <ResumeSelect
   300→   300→              value={resume?.id ?? ""}
   301→   301→              onValueChange={setActive}
   302→   302→              ariaLabel="Resume to analyze"
   303→   303→              className="w-full sm:w-64"
   304→   304→            />
   305→   305→            <Button
   306→   306→              onClick={handleAnalyze}
   307→   307→              disabled={scanning || !resume || !mounted}
   308→   308→              className="bg-gradient-to-r from-emerald-500 to-teal-600 text-white shadow-lg shadow-emerald-500/25 hover:from-emerald-600 hover:to-teal-700"
   309→   309→              aria-label="Analyze resume with ATS scanner"
   310→   310→            >
   311→   311→              {scanning ? (
   312→   312→                <Loader2 className="size-4 animate-spin" aria-hidden="true" />
   313→   313→              ) : (
   314→   314→                <ScanSearch className="size-4" aria-hidden="true" />
   315→   315→              )}
   316→   316→              {scanning ? "Scanning…" : "Analyze"}
   317→   317→            </Button>
   318→   318→          </>
   319→   319→        }
   320→   320→      />
   321→   321→
   322→   322→      {/* Job description (optional) */}
   323→   323→      <motion.section variants={staggerItem} aria-label="Job description input">
   324→   324→        <Card>
   325→   325→          <CardContent className="p-4 sm:p-5">
   326→   326→            <Collapsible open={jdOpen} onOpenChange={setJdOpen}>
   327→   327→              <div className="flex flex-wrap items-center justify-between gap-3">
   328→   328→                <CollapsibleTrigger asChild>
   329→   329→                  <Button
   330→   330→                    variant="ghost"
   331→   331→                    size="sm"
   332→   332→                    className="text-muted-foreground"
   333→   333→                    aria-expanded={jdOpen}
   334→   334→                  >
   335→   335→                    <FileText className="size-4" aria-hidden="true" />
   336→   336→                    Job description{" "}
   337→   337→                    <span className="font-normal">
   338→   338→                      {jd.trim() ? "(added)" : "(optional)"}
   339→   339→                    </span>
   340→   340→                    <ChevronDown
   341→   341→                      className={cn(
   342→   342→                        "size-4 transition-transform duration-200",
   343→   343→                        jdOpen && "rotate-180"
   344→   344→                      )}
   345→   345→                      aria-hidden="true"
   346→   346→                    />
   347→   347→                  </Button>
   348→   348→                </CollapsibleTrigger>
   349→   349→                <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   350→   350→                  <Sparkles className="size-3.5 text-emerald-500" aria-hidden="true" />
   351→   351→                  Matching against a real job description boosts accuracy
   352→   352→                </p>
   353→   353→              </div>
   354→   354→              <CollapsibleContent className="pt-4">
   355→   355→                <div className="space-y-2">
   356→   356→                  <Textarea
   357→   357→                    value={jd}
   358→   358→                    onChange={(e) => setJd(e.target.value)}
   359→   359→                    placeholder="Paste the full job description here — responsibilities, requirements, the works…"
   360→   360→                    aria-label="Job description"
   361→   361→                    className="min-h-[140px] resize-y"
   362→   362→                  />
   363→   363→                  <div className="flex items-center gap-3">
   364→   364→                    <button
   365→   365→                      type="button"
   366→   366→                      onClick={() => setJd(SAMPLE_JD)}
   367→   367→                      className="text-xs font-medium text-emerald-600 underline-offset-4 hover:underline dark:text-emerald-400"
   368→   368→                    >
   369→   369→                      Paste sample JD
   370→   370→                    </button>
   371→   371→                    {jd ? (
   372→   372→                      <button
   373→   373→                        type="button"
   374→   374→                        onClick={() => setJd("")}
   375→   375→                        className="text-xs font-medium text-muted-foreground underline-offset-4 hover:underline"
   376→   376→                      >
   377→   377→                        Clear
   378→   378→                      </button>
   379→   379→                    ) : null}
   380→   380→                  </div>
   381→   381→                </div>
   382→   382→              </CollapsibleContent>
   383→   383→            </Collapsible>
   384→   384→          </CardContent>
   385→   385→        </Card>
   386→   386→      </motion.section>
   387→   387→
   388→   388→      {/* vs last scan (delta panel) */}
   389→   389→      {showDeltaPanel && panelEntry ? (
   390→   390→        <motion.section
   391→   391→          key={result ? `delta-full-${scanId}` : "delta-load"}
   392→   392→          variants={staggerItem}
   393→   393→          aria-label="Comparison with last scan"
   394→   394→          aria-live="polite"
   395→   395→          data-delta={result ? deltaKindValue : "none"}
   396→   396→        >
   397→   397→          <div className="rounded-xl border p-4 sm:p-5">
   398→   398→            <div className="flex flex-wrap items-center gap-x-3 gap-y-2">
   399→   399→              <span
   400→   400→                aria-hidden="true"
   401→   401→                className="flex size-8 shrink-0 items-center justify-center rounded-lg bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   402→   402→              >
   403→   403→                <History className="size-4" />
   404→   404→              </span>
   405→   405→              <div className="min-w-0 flex-1">
   406→   406→                <p className="text-sm font-medium">vs last scan</p>
   407→   407→                <p className="mt-0.5 text-sm tabular-nums text-muted-foreground">
   408→   408→                  Last scan{" "}
   409→   409→                  {formatDistanceToNow(new Date(panelEntry.at), {
   410→   410→                    addSuffix: true,
   411→   411→                  })}
   412→   412→                  :{" "}
   413→   413→                  <span className="font-semibold text-foreground">
   414→   414→                    {panelEntry.score}
   415→   415→                  </span>
   416→   416→                  {result ? (
   417→   417→                    <>
   418→   418→                      {" \u2192 "}
   419→   419→                      Now:{" "}
   420→   420→                      <span className="font-semibold text-foreground">
   421→   421→                        {result.score}
   422→   422→                      </span>
   423→   423→                    </>
   424→   424→                  ) : (
   425→   425→                    <span className="font-normal">
   426→   426→                      {" \u00b7 "} run a scan to compare
   427→   427→                    </span>
   428→   428→                  )}
   429→   429→                </p>
   430→   430→              </div>
   431→   431→              {result ? (
   432→   432→                <span
   433→   433→                  data-delta={deltaKindValue}
   434→   434→                  data-delta-value={deltaDiff}
   435→   435→                  aria-label={deltaAriaLabel(deltaDiff)}
   436→   436→                  className={cn(
   437→   437→                    "inline-flex shrink-0 items-center gap-1 rounded-full px-2.5 py-1 text-xs font-semibold tabular-nums ring-1",
   438→   438→                    DELTA_CHIP_CLASS[deltaKindValue]
   439→   439→                  )}
   440→   440→                >
   441→   441→                  <DeltaChipIcon kind={deltaKindValue} />
   442→   442→                  {deltaKindValue === "flat" ? "No change" : formatDelta(deltaDiff)}
   443→   443→                </span>
   444→   444→              ) : null}
   445→   445→            </div>
   446→   446→            {result && deltaInsight(result.breakdown) ? (
   447→   447→              <p className="mt-3 flex items-center gap-1.5 border-t pt-3 text-xs text-muted-foreground">
   448→   448→                <Sparkles
   449→   449→                  className="size-3.5 shrink-0 text-emerald-500"
   450→   450→                  aria-hidden="true"
   451→   451→                />
   452→   452→                {deltaInsight(result.breakdown)}
   453→   453→              </p>
   454→   454→            ) : null}
   455→   455→          </div>
   456→   456→        </motion.section>
   457→   457→      ) : null}
   458→   458→
   459→   459→      {/* Scanning animation */}
   460→   460→      {scanning ? (
   461→   461→        <motion.section
   462→   462→          variants={staggerItem}
   463→   463→          aria-busy="true"
