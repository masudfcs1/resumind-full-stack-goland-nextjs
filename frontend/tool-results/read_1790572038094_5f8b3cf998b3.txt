   430→/* "+12 pts" pill that counts up when scrolled into view */
   431→function DeltaPill() {
   432→  const ref = React.useRef<HTMLSpanElement>(null);
   433→  const inView = useInView(ref, { once: true, margin: "-60px" });
   434→  const mv = useMotionValue(0);
   435→  const text = useTransform(mv, (v: number) => `+${Math.round(v)} pts`);
   436→
   437→  React.useEffect(() => {
   438→    if (!inView) return;
   439→    const controls = animate(mv, 12, { duration: 1.4, ease: "easeOut", delay: 0.5 });
   440→    return () => controls.stop();
   441→  }, [inView, mv]);
   442→
   443→  return (
   444→    <span
   445→      ref={ref}
   446→      className="absolute -bottom-3 left-1/2 z-10 inline-flex -translate-x-1/2 items-center gap-1 rounded-full border border-emerald-500/30 bg-background/90 px-2.5 py-1 text-[10px] font-extrabold text-emerald-600 shadow-sm backdrop-blur-sm dark:border-emerald-400/25 dark:text-emerald-400"
   447→    >
   448→      <TrendingUp className="size-3" aria-hidden />
   449→      <motion.span className="tabular-nums">{text}</motion.span>
   450→    </span>
   451→  );
   452→}
