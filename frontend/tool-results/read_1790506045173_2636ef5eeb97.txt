    91→function initialsOf(name: string): string {
    92→  const parts = name.trim().split(/\s+/).filter(Boolean);
    93→  if (parts.length === 0) return "?";
    94→  return parts
    95→    .slice(0, 2)
    96→    .map((p) => p[0]!.toUpperCase())
    97→    .join("");
    98→}

   102→interface ParsedBackup {
   103→  payload: Record<string, unknown>;
   104→  resumeCount: number;
   105→  applicationCount: number | null; // null = absent, store will fall back to seed applications
   106→  scoreHistoryCount: number | null; // null = absent, store will fall back to seed score history
   107→  exportedAt: string | null;
   108→}
   109→
   110→function parseBackupText(text: string): { ok: true; data: ParsedBackup } | { ok: false; reason: string } {
   111→  let parsed: unknown;
   112→  try {
   113→    parsed = JSON.parse(text);
   114→  } catch {
   115→    return { ok: false, reason: "That file isn't valid JSON. Export a fresh backup from Settings and try again." };
   116→  }
   117→  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
   118→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — expected a JSON object." };
   119→  }
   120→  const obj = parsed as Record<string, unknown>;
   121→  const rawResumes = Array.isArray(obj.resumes) ? obj.resumes : null;
   122→  if (obj.app !== "ResumeForge AI" && !rawResumes) {
   123→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — the \"app\" marker and resumes array are missing." };
   124→  }
   125→  if (!rawResumes || rawResumes.length === 0) {
   126→    return { ok: false, reason: "The backup contains no resumes, so there is nothing to restore." };
   127→  }
   128→  for (let i = 0; i < rawResumes.length; i++) {
   129→    const entry: unknown = rawResumes[i];
   130→    if (!entry || typeof entry !== "object" || Array.isArray(entry)) {
   131→      return { ok: false, reason: `Resume #${i + 1} in the backup is malformed (expected an object).` };
   132→    }
   133→    const rec = entry as Record<string, unknown>;
   134→    const titleOk = typeof rec.title === "string";
   135→    if (
   136→      typeof rec.id !== "string" ||
   137→      rec.id.trim() === "" ||
   138→      !titleOk ||
   139→      !rec.personal ||
   140→      typeof rec.personal !== "object" ||
   141→      Array.isArray(rec.personal)
   142→    ) {
   143→      const label = titleOk ? `"${rec.title}"` : `#${i + 1}`;
   144→      return {
   145→        ok: false,
   146→        reason: `Resume ${label} is missing an id, a title or personal details — the backup looks corrupted.`,
   147→      };
   148→    }
   149→  }
   150→  return {
   151→    ok: true,
   152→    data: {
   153→      payload: obj,
   154→      resumeCount: rawResumes.length,
   155→      applicationCount: Array.isArray(obj.applications) ? obj.applications.length : null,
   156→      scoreHistoryCount: Array.isArray(obj.scoreHistory) ? obj.scoreHistory.length : null,
   157→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : null,
   158→    },
   159→  };
   160→}
   161→
   162→function formatBackupDate(iso: string | null): string {
   163→  if (!iso) return "Not recorded";
   164→  const date = new Date(iso);
   165→  if (!isValidDate(date)) return "Not recorded";
   166→  return format(date, "MMM d, yyyy 'at' h:mm a");
   167→}
