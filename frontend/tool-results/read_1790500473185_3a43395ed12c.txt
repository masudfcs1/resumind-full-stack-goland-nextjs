   381→/* "+12 pts" pill that counts up when scrolled into view */
   382→function DeltaPill() {
   383→  const ref = React.useRef<HTMLSpanElement>(null);
   384→  const inView = useInView(ref, { once: true, margin: "-60px" });
   385→  const mv = useMotionValue(0);
   386→  const text = useTransform(mv, (v: number) => `+${Math.round(v)} pts`);
   387→
   388→  React.useEffect(() => {
   389→    if (!inView) return;
   390→    const controls = animate(mv, 12, { duration: 1.4, ease: "easeOut", delay: 0.5 });
   391→    return () => controls.stop();
   392→  }, [inView, mv]);
   393→
   394→  return (
   395→    <span
   396→      ref={ref}
   397→      className="absolute -bottom-3 left-1/2 z-10 inline-flex -translate-x-1/2 items-center gap-1 rounded-full border border-emerald-500/30 bg-background/90 px-2.5 py-1 text-[10px] font-extrabold text-emerald-600 shadow-sm backdrop-blur-sm dark:border-emerald-400/25 dark:text-emerald-400"
   398→    >
   399→      <TrendingUp className="size-3" aria-hidden />
   400→      <motion.span className="tabular-nums">{text}</motion.span>
   401→    </span>
   402→  );
   403→}
