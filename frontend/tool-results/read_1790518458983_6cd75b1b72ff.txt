   452→/* ============================== Phase strip ============================== */
   453→
   454→/**
   455→ * Slim 3-segment difficulty ladder strip (Warm-up / Core / Stretch) shown
   456→ * above the question list. Each segment carries its phase count and a
   457→ * practiced-progress fill; the active segment glows in its tier tint, and a
   458→ * one-line coach hint sits underneath. While ladder-aware Focus Mode is on,
   459→ * the focus tier drives the active segment and fully-practiced segments
   460→ * show a small check.
   461→ */
   462→function PhaseStrip({
   463→  questions,
   464→  practicedIds,
   465→  activeTier,
   466→  focusActive,
   467→  focusTier,
   468→  focusTierCleared,
   469→}: {
   470→  questions: PracticeQuestion[];
   471→  practicedIds: ReadonlySet<string>;
   472→  /** Tier shown as the active segment (frontier, focus tier, or dialog card). */
   473→  activeTier: Difficulty;
   474→  /** Focus Mode on? Gates the cleared checks + the focus coach note. */
   475→  focusActive: boolean;
   476→  /** Tier Focus Mode filters to (null while off). */
   477→  focusTier: Difficulty | null;
   478→  /** Every card of the focus tier is practiced — clearance CTA is showing. */
   479→  focusTierCleared: boolean;
   480→}) {
   481→  const reducedMotion = useReducedMotion();
   482→
   483→  const segments = React.useMemo(
   484→    () =>
   485→      PHASE_SEGMENTS.map((seg) => {
   486→        const idxs = questions
   487→          .map((q, i) => (q.difficulty === seg.tier ? i : -1))
   488→          .filter((i) => i >= 0);
   489→        const practiced = idxs.filter((i) => {
   490→          const q = questions[i];
   491→          return q ? practicedIds.has(q.id) : false;
   492→        }).length;
   493→        return { ...seg, count: idxs.length, practiced };
   494→      }),
   495→    [questions, practicedIds]
   496→  );
   497→
   498→  const currentSegment = PHASE_SEGMENTS.find((s) => s.tier === activeTier);
   499→
   500→  return (
   501→    <motion.section
   502→      variants={staggerItem}
   503→      aria-label="Difficulty phases"
   504→      data-phase-strip="true"
   505→      data-current-phase={currentSegment?.id ?? "warm-up"}
   506→      data-focus-active={focusActive ? "true" : "false"}
   507→      data-focus-tier={focusActive && focusTier ? focusTier : ""}
   508→    >
   509→      <div className="flex gap-1.5 sm:gap-2">
   510→        {segments.map((seg) => {
   511→          const active = seg.tier === activeTier;
   512→          const cleared = seg.count > 0 && seg.practiced === seg.count;
   513→          const fillPct =
   514→            seg.count > 0 ? Math.round((seg.practiced / seg.count) * 100) : 0;
   515→          return (
   516→            <div
   517→              key={seg.tier}
   518→              data-phase={seg.id}
   519→              data-tier={seg.tier}
   520→              data-count={seg.count}
   521→              data-active={active ? "true" : "false"}
   522→              data-cleared={cleared ? "true" : "false"}
   523→              className={cn(
   524→                "relative min-w-0 flex-1 overflow-hidden rounded-lg border px-2.5 py-1.5 transition-all sm:px-3",
   525→                active ? cn(seg.active, seg.glow) : "border-border"
   526→              )}
   527→            >
   528→              <motion.div
   529→                aria-hidden="true"
   530→                className={cn("absolute inset-y-0 left-0", seg.fill)}
   531→                initial={false}
   532→                animate={{ width: `${fillPct}%` }}
   533→                transition={
   534→                  reducedMotion
   535→                    ? { duration: 0 }
   536→                    : { type: "spring", stiffness: 140, damping: 22 }
   537→                }
   538→              />
   539→              <div className="relative flex min-w-0 items-center justify-between gap-2">
   540→                <span
   541→                  className={cn(
   542→                    "truncate text-[10px] font-semibold uppercase tracking-[0.12em]",
   543→                    active ? seg.text : "text-muted-foreground"
   544→                  )}
   545→                >
   546→                  {DIFFICULTY_META[seg.tier].phase}
   547→                </span>
   548→                <span
   549→                  className={cn(
   550→                    "flex shrink-0 items-center gap-1 text-[10px] font-semibold tabular-nums",
   551→                    active ? seg.text : "text-muted-foreground/70"
   552→                  )}
   553→                >
   554→                  {focusActive && cleared ? (
   555→                    <Check
   556→                      className="size-3 shrink-0 text-emerald-600 dark:text-emerald-400"
   557→                      aria-hidden="true"
   558→                    />
   559→                  ) : null}
   560→                  {seg.practiced}/{seg.count}
   561→                </span>
   562→              </div>
   563→            </div>
   564→          );
   565→        })}
   566→      </div>
   567→      <p
   568→        className={cn(
   569→          "mt-2 flex items-start gap-2 rounded-lg border px-3 py-2 text-xs leading-relaxed",
   570→          PHASE_COACH_TINT[activeTier]
   571→        )}
   572→      >
   573→        <Compass className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   574→        <span>
   575→          <span className="font-semibold">Coach:</span> {PHASE_COACH[activeTier]}
   576→          {focusActive ? (
   577→            <span className="font-medium">
   578→              {focusTierCleared
