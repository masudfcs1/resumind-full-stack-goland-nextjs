   425→   425→/* "+12 pts" pill that counts up when scrolled into view */
   426→   426→function DeltaPill() {
   427→   427→  const ref = React.useRef<HTMLSpanElement>(null);
   428→   428→  const inView = useInView(ref, { once: true, margin: "-60px" });
   429→   429→  const mv = useMotionValue(0);
   430→   430→  const text = useTransform(mv, (v: number) => `+${Math.round(v)} pts`);
   431→   431→
   432→   432→  React.useEffect(() => {
   433→   433→    if (!inView) return;
   434→   434→    const controls = animate(mv, 12, { duration: 1.4, ease: "easeOut", delay: 0.5 });
   435→   435→    return () => controls.stop();
   436→   436→  }, [inView, mv]);
   437→   437→
   438→   438→  return (
   439→   439→    <span
   440→   440→      ref={ref}
