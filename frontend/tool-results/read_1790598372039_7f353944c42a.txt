   192→   192→export function formatMonth(value: string): string {
   193→   193→  if (!value) return "";
   194→   194→  const [y, m] = value.split("-");
   195→   195→  if (!y) return value;
   196→   196→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   197→   197→  if (!m) return y;
   198→   198→  const idx = parseInt(m, 10) - 1;
   199→   199→  return `${months[idx] ?? ""} ${y}`;
   200→   200→}

   660→   660→/* ============================== Undo buffer (module-local, NOT persisted) ============================== */
   661→   661→
   662→   662→/* Captures the most recent delete action so the Undo button on the delete
   663→   663→   toast can put the resumes back exactly where they were. Deliberately kept
   664→   664→   OUTSIDE the persisted store: undo data is transient and must never reach
   665→   665→   localStorage or widen the persisted schema. */
   666→   666→let undoBuffer: DeletedResumeEntry[] = [];
   667→   667→let undoBatchIds: string[] = [];
   668→   668→let undoBatchSealed = false;
   669→   669→
   670→   670→/* deleteResume may fire several times for ONE user action (multi-select
   671→   671→   delete loops over the selected ids). Calls made within the same synchronous
   672→   672→   task share one batch: the first call snapshots the pre-delete id order so
   673→   673→   every captured entry keeps its ORIGINAL position even though the array
   674→   674→   shrinks between calls; the next task starts a fresh batch. */
   675→   675→function openUndoBatch(): void {
   676→   676→  if (undoBatchSealed) return;
   677→   677→  undoBuffer = [];
   678→   678→  undoBatchIds = useResumeStore.getState().resumes.map((r) => r.id);
   679→   679→  undoBatchSealed = true;
   680→   680→  queueMicrotask(() => {
   681→   681→    undoBatchSealed = false;
   682→   682→  });
   683→   683→}
   684→   684→
   685→   685→/** Returns (and clears) the snapshots captured for the most recent delete
   686→   686→    action. Feed the result straight into showDeleteUndoToast. */
   687→   687→export function takeUndoEntries(): DeletedResumeEntry[] {
   688→   688→  const entries = undoBuffer;
   689→   689→  undoBuffer = [];
   690→   690→  return entries;
   691→   691→}

   694→   694→    entries at their original positions (restoreResumes). */
   695→   695→export function showDeleteUndoToast(entries: DeletedResumeEntry[]): void {
   696→   696→  const n = entries.length;
   697→   697→  if (n === 0) return;
   698→   698→  toast.success(n === 1 ? "Resume deleted" : `Deleted ${n} resumes`, {
   699→   699→    description:

   702→   702→        : `${n} resumes were removed from your workspace.`,
   703→   703→    duration: 7000,
   704→   704→    action: {
   705→   705→      label: "Undo",
   706→   706→      onClick: () => {
   707→   707→        useResumeStore.getState().restoreResumes(entries);
   708→   708→        toast.success(n === 1 ? "Resume restored" : `Restored ${n} resumes`, {
   709→   709→          description: n === 1 ? "Back at its original position." : "Back at their original positions.",
   710→   710→        });

   712→   712→    },
   713→   713→  });
   714→   714→}

   718→   718→interface ResumeState {
   719→   719→  resumes: ResumeData[];
   720→   720→  activeResumeId: string;
   721→   721→  coverLetterCount: number;
   722→   722→  applications: JobApplication[];
   723→   723→  scoreHistory: ScoreEntry[];
   724→   724→  createResume: (title?: string, template?: TemplateId, accent?: string) => string;
   725→   725→  duplicateResume: (id: string) => string | undefined;
   726→   726→  deleteResume: (id: string) => void;
   727→   727→  /** Soft-hide: keeps the resume in the store but flags it archived. */
   728→   728→  archiveResume: (id: string) => void;
   729→   729→  /** Reverse of archiveResume — clears the archived flag. */
   730→   730→  unarchiveResume: (id: string) => void;
   731→   731→  /** Batch archive/unarchive for the resume select-mode bulk toolbar.
   732→   732→      Idempotent per resume: flags already in the target state are left so
   733→   733→      their updatedAt is untouched. */
   734→   734→  bulkSetArchived: (ids: string[], archived: boolean) => void;
   735→   735→  /** Splices previously deleted resumes back at their original positions
   736→   736→      (used by the Undo action on delete toasts). Idempotent: entries whose id
   737→   737→      is already present are skipped. */
   738→   738→  restoreResumes: (entries: DeletedResumeEntry[]) => void;
   739→   739→  setActive: (id: string) => void;
   740→   740→  updateResume: (id: string, patch: Partial<ResumeData>) => void;
   741→   741→  resetAll: () => void;
   742→   742→  /** Validates a parsed backup payload and replaces the workspace with it. Returns the number of resumes imported, or -1 if the payload is invalid. */
   743→   743→  importData: (payload: unknown) => number;
   744→   744→  incrementCoverLetters: () => void;
   745→   745→  addApplication: (app: Omit<JobApplication, "id" | "appliedAt" | "updatedAt">) => string;
   746→   746→  updateApplication: (id: string, patch: Partial<JobApplication>) => void;
   747→   747→  moveApplication: (id: string, stage: ApplicationStage) => void;
   748→   748→  deleteApplication: (id: string) => void;
   749→   749→  logScore: (
   750→   750→    resumeId: string,
   751→   751→    resumeTitle: string,
   752→   752→    score: number,
   753→   753→    categories?: Record<string, number>
   754→   754→  ) => void;
   755→   755→  /* --- Per-tool data purge (Settings → Data & privacy). Each action clears
   756→   756→         exactly ONE domain to its empty state and deliberately leaves
   757→   757→         `resumes` (and `activeResumeId`) untouched. Additive: older persisted
   758→   758→         state simply never had these keys, and clearing only writes slices the
   759→   759→         persist config already serializes. --- */
   760→   760→  /** Empties the ATS score-history log (all entries). Resumes are untouched,
   761→   761→      but anything derived from this log resets with it: resume-health
   762→   762→      freshness falls back to "never scanned" and Compare loses its baseline
   763→   763→      entries. Not undoable — callers confirm first. */
   764→   764→  clearScoreHistory: () => void;
   765→   765→  /** Empties the job-application tracker. Resumes are untouched; the dangling
   766→   766→      `resumeId` references simply have no applications left to point at. */
   767→   767→  clearApplications: () => void;
   768→   768→  /** Resets the generated-cover-letter counter to 0. Cover letters are not
   769→   769→      stored as documents anywhere — this counter is the only persisted trace
   770→   770→      of them — so resetting it IS the full cover-letter purge. Resumes are
   771→   771→      untouched. */
   772→   772→  clearCoverLetters: () => void;

   777→   777→    (set, get) => ({

   779→   779→      activeResumeId: "seed-alex",
   780→   780→      coverLetterCount: 2,
   781→   781→      applications: seedApplications(),
   782→   782→      scoreHistory: seedScoreHistory(),
   783→   783→      createResume: (title, template, accent) => {
   784→   784→        const id = uid();
   785→   785→        const base = emptyResume(title ?? "Untitled Resume", template ?? "modern", accent ?? "#10b981");
   786→   786→        const resume: ResumeData = { ...base, id, updatedAt: Date.now() };
   787→   787→        set((s) => ({ resumes: [resume, ...s.resumes], activeResumeId: id }));
   788→   788→        return id;
   789→   789→      },
   790→   790→      duplicateResume: (id) => {
   791→   791→        const src = get().resumes.find((r) => r.id === id);
   792→   792→        if (!src) return undefined;
   793→   793→        const newId = uid();
   794→   794→        const copy: ResumeData = {
   795→   795→          ...structuredClone(src),
   796→   796→          id: newId,
   797→   797→          title: `${src.title} (Copy)`,
   798→   798→          updatedAt: Date.now(),
   799→   799→          /* Copies always start life in the active list, even if the source was archived. */
   800→   800→          archived: false,
   801→   801→        };
   802→   802→        set((s) => ({ resumes: [copy, ...s.resumes] }));
   803→   803→        return newId;
   804→   804→      },
   805→   805→      deleteResume: (id) => {
   806→   806→        const s = get();
   807→   807→        const index = s.resumes.findIndex((r) => r.id === id);
   808→   808→        if (index === -1) return;
   809→   809→        /* Capture BEFORE removal (deep copy + original position, anchored to
   810→   810→           the batch's pre-delete order) so the Undo toast can restore it. */
   811→   811→        openUndoBatch();
   812→   812→        const batchIndex = undoBatchIds.indexOf(id);
   813→   813→        undoBuffer.push({
   814→   814→          resume: structuredClone(s.resumes[index]!),
   815→   815→          index: batchIndex !== -1 ? batchIndex : index,
   816→   816→        });
   817→   817→        const resumes = s.resumes.filter((r) => r.id !== id);
   818→   818→        set({
   819→   819→          resumes,
   820→   820→          activeResumeId:
   821→   821→            s.activeResumeId === id ? (resumes[0]?.id ?? "") : s.activeResumeId,
   822→   822→        });
   823→   823→      },
   824→   824→      /* Archive bumps updatedAt like every other mutation (store convention).
   825→   825→         The active pointer is intentionally left untouched when archiving the
   826→   826→         active resume — it still resolves (the resume stays in the store), so
   827→   827→         the builder/ATS binding never breaks. */
   828→   828→      archiveResume: (id) =>
   829→   829→        set((s) => ({
   830→   830→          resumes: s.resumes.map((r) =>
   831→   831→            r.id === id ? { ...r, archived: true, updatedAt: Date.now() } : r
   832→   832→          ),
   833→   833→        })),
   834→   834→      unarchiveResume: (id) =>
   835→   835→        set((s) => ({
   836→   836→          resumes: s.resumes.map((r) =>
   837→   837→            r.id === id ? { ...r, archived: false, updatedAt: Date.now() } : r
   838→   838→          ),
   839→   839→        })),
   840→   840→      bulkSetArchived: (ids, archived) =>
   841→   841→        set((s) => {
   842→   842→          const wanted = new Set(ids);
   843→   843→          return {
   844→   844→            resumes: s.resumes.map((r) =>
   845→   845→              wanted.has(r.id) && r.archived !== archived
   846→   846→                ? { ...r, archived, updatedAt: Date.now() }
   847→   847→                : r
   848→   848→            ),
   849→   849→          };
   850→   850→        }),
   851→   851→      restoreResumes: (entries) =>
   852→   852→        set((s) => {
   853→   853→          const resumes = [...s.resumes];
   854→   854→          const present = new Set(resumes.map((r) => r.id));
   855→   855→          /* Splice in descending index order so every entry lands exactly
   856→   856→             where it came from, regardless of how many were removed. */
   857→   857→          const ordered = [...entries].sort((a, b) => b.index - a.index);
   858→   858→          for (const entry of ordered) {
   859→   859→            if (present.has(entry.resume.id)) continue;
   860→   860→            const at = Math.min(Math.max(entry.index, 0), resumes.length);
   861→   861→            resumes.splice(at, 0, structuredClone(entry.resume));
   862→   862→            present.add(entry.resume.id);
   863→   863→          }
   864→   864→          return { resumes };
   865→   865→        }),
   866→   866→      setActive: (id) => set({ activeResumeId: id }),
   867→   867→      updateResume: (id, patch) =>
   868→   868→        set((s) => ({
   869→   869→          resumes: s.resumes.map((r) =>
   870→   870→            r.id === id ? { ...r, ...patch, updatedAt: Date.now() } : r
   871→   871→          ),
   872→   872→        })),
   873→   873→      resetAll: () =>
   874→   874→        set({
   875→   875→          resumes: [seedResume1(), seedResume2(), seedResume3()],
   876→   876→          activeResumeId: "seed-alex",
   877→   877→          coverLetterCount: 2,
   878→   878→          applications: seedApplications(),
   879→   879→          scoreHistory: seedScoreHistory(),
   880→   880→        }),
   881→   881→      importData: (payload) => {
   882→   882→        if (!payload || typeof payload !== "object" || Array.isArray(payload)) return -1;
   883→   883→        const raw = payload as Record<string, unknown>;
   884→   884→
   885→   885→        /* Resumes are required — every entry needs an id, a title and a personal object. */
   886→   886→        const rawResumes = raw.resumes;
   887→   887→        if (!Array.isArray(rawResumes) || rawResumes.length === 0) return -1;
   888→   888→        const blank = emptyResume();
   889→   889→        const resumes: ResumeData[] = [];
   890→   890→        for (const entry of rawResumes) {
   891→   891→          if (!entry || typeof entry !== "object" || Array.isArray(entry)) return -1;
   892→   892→          const r = entry as Record<string, unknown>;
   893→   893→          if (typeof r.id !== "string" || r.id.trim() === "") return -1;
   894→   894→          if (typeof r.title !== "string") return -1;
   895→   895→          if (!r.personal || typeof r.personal !== "object" || Array.isArray(r.personal)) return -1;
   896→   896→          const copy = structuredClone(entry) as ResumeData;
   897→   897→          copy.template = typeof r.template === "string" && r.template in TEMPLATE_META ? (r.template as TemplateId) : blank.template;
   898→   898→          copy.accent = typeof r.accent === "string" && r.accent.trim() !== "" ? r.accent : blank.accent;
   899→   899→          copy.summary = typeof r.summary === "string" ? r.summary : blank.summary;
   900→   900→          copy.personal = { ...blank.personal, ...(r.personal as Partial<PersonalInfo>) };
   901→   901→          for (const key of ["experience", "education", "skills", "projects", "certifications", "languages"] as const) {
   902→   902→            if (!Array.isArray(copy[key])) copy[key] = [];
   903→   903→          }
   904→   904→          copy.updatedAt = typeof r.updatedAt === "number" && Number.isFinite(r.updatedAt) ? r.updatedAt : Date.now();
   905→   905→          resumes.push(copy);
   906→   906→        }
   907→   907→
   908→   908→        /* Active resume falls back to the first import if the id is missing or unknown. */
   909→   909→        const wantedActive = typeof raw.activeResumeId === "string" ? raw.activeResumeId : "";
   910→   910→        const firstId = resumes[0]!.id;
   911→   911→        const activeResumeId = resumes.some((r) => r.id === wantedActive) ? wantedActive : firstId;
   912→   912→
   913→   913→        const coverLetterCount =
   914→   914→          typeof raw.coverLetterCount === "number" && Number.isFinite(raw.coverLetterCount)
   915→   915→            ? Math.max(0, Math.round(raw.coverLetterCount))
   916→   916→            : 0;
   917→   917→
   918→   918→        /* Applications and score history fall back to seeds when absent (mirrors resetAll). */
   919→   919→        let applications = seedApplications();
   920→   920→        if (Array.isArray(raw.applications)) {
   921→   921→          const clean: JobApplication[] = [];
   922→   922→          for (const entry of raw.applications) {
   923→   923→            if (!entry || typeof entry !== "object" || Array.isArray(entry)) continue;
   924→   924→            const a = entry as Record<string, unknown>;
