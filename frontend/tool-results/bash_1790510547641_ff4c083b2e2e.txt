   144→export function formatMonth(value: string): string {
   145→  if (!value) return "";
   146→  const [y, m] = value.split("-");
   147→  if (!y) return value;
   148→  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
   149→  if (!m) return y;
   150→  const idx = parseInt(m, 10) - 1;
   151→  return `${months[idx] ?? ""} ${y}`;
   152→}

   612→  resumes: ResumeData[];
   613→  activeResumeId: string;
   614→  coverLetterCount: number;
   615→  applications: JobApplication[];
   616→  scoreHistory: ScoreEntry[];
   617→  createResume: (title?: string, template?: TemplateId, accent?: string) => string;
   618→  duplicateResume: (id: string) => string | undefined;
   619→  deleteResume: (id: string) => void;
   620→  setActive: (id: string) => void;
   621→  updateResume: (id: string, patch: Partial<ResumeData>) => void;
   622→  resetAll: () => void;
   623→  /** Validates a parsed backup payload and replaces the workspace with it. Returns the number of resumes imported, or -1 if the payload is invalid. */
   624→  importData: (payload: unknown) => number;
   625→  incrementCoverLetters: () => void;
   626→  addApplication: (app: Omit<JobApplication, "id" | "appliedAt" | "updatedAt">) => string;
   627→  updateApplication: (id: string, patch: Partial<JobApplication>) => void;
   628→  moveApplication: (id: string, stage: ApplicationStage) => void;
   629→  deleteApplication: (id: string) => void;
   630→  logScore: (resumeId: string, resumeTitle: string, score: number) => void;
   631→}
   632→
   633→export const useResumeStore = create<ResumeState>()(
   634→  persist(
   635→    (set, get) => ({
   636→      resumes: [seedResume1(), seedResume2(), seedResume3()],
   637→      activeResumeId: "seed-alex",
   638→      coverLetterCount: 2,
   639→      applications: seedApplications(),
   640→      scoreHistory: seedScoreHistory(),
   641→      createResume: (title, template, accent) => {
   642→        const id = uid();
   643→        const base = emptyResume(title ?? "Untitled Resume", template ?? "modern", accent ?? "#10b981");
   644→        const resume: ResumeData = { ...base, id, updatedAt: Date.now() };
   645→        set((s) => ({ resumes: [resume, ...s.resumes], activeResumeId: id }));
   646→        return id;
   647→      },
   648→      duplicateResume: (id) => {
   649→        const src = get().resumes.find((r) => r.id === id);
   650→        if (!src) return undefined;
   651→        const newId = uid();
   652→        const copy: ResumeData = {
   653→          ...structuredClone(src),
   654→          id: newId,
   655→          title: `${src.title} (Copy)`,
   656→          updatedAt: Date.now(),
   657→        };
   658→        set((s) => ({ resumes: [copy, ...s.resumes] }));
   659→        return newId;
   660→      },
   661→      deleteResume: (id) =>
   662→        set((s) => {
   663→          const resumes = s.resumes.filter((r) => r.id !== id);
   664→          const activeResumeId =
   665→            s.activeResumeId === id ? (resumes[0]?.id ?? "") : s.activeResumeId;
   666→          return { resumes, activeResumeId };
   667→        }),
   668→      setActive: (id) => set({ activeResumeId: id }),
   669→      updateResume: (id, patch) =>
   670→        set((s) => ({
   671→          resumes: s.resumes.map((r) =>
   672→            r.id === id ? { ...r, ...patch, updatedAt: Date.now() } : r
   673→          ),
   674→        })),
   675→      resetAll: () =>
   676→        set({
   677→          resumes: [seedResume1(), seedResume2(), seedResume3()],
   678→          activeResumeId: "seed-alex",
   679→          coverLetterCount: 2,
   680→          applications: seedApplications(),
   681→          scoreHistory: seedScoreHistory(),
   682→        }),
   683→      importData: (payload) => {
   684→        if (!payload || typeof payload !== "object" || Array.isArray(payload)) return -1;
   685→        const raw = payload as Record<string, unknown>;
   686→
   687→        /* Resumes are required — every entry needs an id, a title and a personal object. */
   688→        const rawResumes = raw.resumes;
   689→        if (!Array.isArray(rawResumes) || rawResumes.length === 0) return -1;
   690→        const blank = emptyResume();
   691→        const resumes: ResumeData[] = [];
   692→        for (const entry of rawResumes) {
   693→          if (!entry || typeof entry !== "object" || Array.isArray(entry)) return -1;
   694→          const r = entry as Record<string, unknown>;
   695→          if (typeof r.id !== "string" || r.id.trim() === "") return -1;
   696→          if (typeof r.title !== "string") return -1;
   697→          if (!r.personal || typeof r.personal !== "object" || Array.isArray(r.personal)) return -1;
   698→          const copy = structuredClone(entry) as ResumeData;
   699→          copy.template = typeof r.template === "string" && r.template in TEMPLATE_META ? (r.template as TemplateId) : blank.template;
   700→          copy.accent = typeof r.accent === "string" && r.accent.trim() !== "" ? r.accent : blank.accent;
   701→          copy.summary = typeof r.summary === "string" ? r.summary : blank.summary;
   702→          copy.personal = { ...blank.personal, ...(r.personal as Partial<PersonalInfo>) };
   703→          for (const key of ["experience", "education", "skills", "projects", "certifications", "languages"] as const) {
   704→            if (!Array.isArray(copy[key])) copy[key] = [];
   705→          }
   706→          copy.updatedAt = typeof r.updatedAt === "number" && Number.isFinite(r.updatedAt) ? r.updatedAt : Date.now();
   707→          resumes.push(copy);
   708→        }
   709→
   710→        /* Active resume falls back to the first import if the id is missing or unknown. */
   711→        const wantedActive = typeof raw.activeResumeId === "string" ? raw.activeResumeId : "";
   712→        const firstId = resumes[0]!.id;
   713→        const activeResumeId = resumes.some((r) => r.id === wantedActive) ? wantedActive : firstId;
   714→
   715→        const coverLetterCount =
   716→          typeof raw.coverLetterCount === "number" && Number.isFinite(raw.coverLetterCount)
   717→            ? Math.max(0, Math.round(raw.coverLetterCount))
   718→            : 0;
   719→
   720→        /* Applications and score history fall back to seeds when absent (mirrors resetAll). */
   721→        let applications = seedApplications();
   722→        if (Array.isArray(raw.applications)) {
   723→          const clean: JobApplication[] = [];
   724→          for (const entry of raw.applications) {
   725→            if (!entry || typeof entry !== "object" || Array.isArray(entry)) continue;
   726→            const a = entry as Record<string, unknown>;
   727→            clean.push({
   728→              id: typeof a.id === "string" && a.id !== "" ? a.id : uid(),
   729→              company: typeof a.company === "string" ? a.company : "",
   730→              role: typeof a.role === "string" ? a.role : "",
   731→              location: typeof a.location === "string" ? a.location : "",
   732→              salary: typeof a.salary === "string" ? a.salary : "",
   733→              url: typeof a.url === "string" ? a.url : "",
   734→              stage: typeof a.stage === "string" && a.stage in STAGE_META ? (a.stage as ApplicationStage) : "applied",
   735→              resumeId: typeof a.resumeId === "string" ? a.resumeId : "",
   736→              notes: typeof a.notes === "string" ? a.notes : "",
   737→              appliedAt: typeof a.appliedAt === "number" && Number.isFinite(a.appliedAt) ? a.appliedAt : Date.now(),
   738→              updatedAt: typeof a.updatedAt === "number" && Number.isFinite(a.updatedAt) ? a.updatedAt : Date.now(),
   739→            });
   740→          }
   741→          if (clean.length > 0) applications = clean;
   742→        }
   743→
   744→        let scoreHistory = seedScoreHistory();
   745→        if (Array.isArray(raw.scoreHistory)) {
   746→          const clean: ScoreEntry[] = [];
   747→          for (const entry of raw.scoreHistory) {
   748→            if (!entry || typeof entry !== "object" || Array.isArray(entry)) continue;
   749→            const e = entry as Record<string, unknown>;
   750→            if (typeof e.score !== "number" || !Number.isFinite(e.score)) continue;
   751→            clean.push({
   752→              id: typeof e.id === "string" && e.id !== "" ? e.id : uid(),
   753→              resumeId: typeof e.resumeId === "string" ? e.resumeId : "",
   754→              resumeTitle: typeof e.resumeTitle === "string" ? e.resumeTitle : "Untitled",
   755→              score: e.score,
   756→              at: typeof e.at === "number" && Number.isFinite(e.at) ? e.at : Date.now(),
   757→            });
   758→          }
   759→          if (clean.length > 0) scoreHistory = clean.slice(-60);
   760→        }
   761→
   762→        set({ resumes, activeResumeId, coverLetterCount, applications, scoreHistory });
   763→        return resumes.length;
   764→      },
   765→      incrementCoverLetters: () => set((s) => ({ coverLetterCount: s.coverLetterCount + 1 })),
   766→      addApplication: (app) => {
   767→        const id = uid();
   768→        const full: JobApplication = {
   769→          ...app,
   770→          id,
   771→          appliedAt: Date.now(),
   772→          updatedAt: Date.now(),
   773→        };
   774→        set((s) => ({ applications: [full, ...s.applications] }));
   775→        return id;
   776→      },
   777→      updateApplication: (id, patch) =>
   778→        set((s) => ({
   779→          applications: s.applications.map((a) =>
   780→            a.id === id ? { ...a, ...patch, updatedAt: Date.now() } : a
   781→          ),
   782→        })),
   783→      moveApplication: (id, stage) =>
   784→        set((s) => ({
   785→          applications: s.applications.map((a) =>
   786→            a.id === id ? { ...a, stage, updatedAt: Date.now() } : a
   787→          ),
   788→        })),
   789→      deleteApplication: (id) =>
   790→        set((s) => ({ applications: s.applications.filter((a) => a.id !== id) })),
   791→      logScore: (resumeId, resumeTitle, score) =>
   792→        set((s) => ({
   793→          scoreHistory: [
   794→            ...s.scoreHistory,
   795→            { id: uid(), resumeId, resumeTitle, score, at: Date.now() },
   796→          ].slice(-60),
   797→        })),
   798→    }),
   799→    {
   800→      name: "resumeforge-store",
   801→      storage: createJSONStorage(() => localStorage),
   802→    }
   803→  )
   804→);
   805→
   806→/* Convenience selectors */
   807→export function useActiveResume(): ResumeData | undefined {
   808→  return useResumeStore((s) => s.resumes.find((r) => r.id === s.activeResumeId));
   809→}
   810→
   811→export function useGetResume() {
   812→  return (id: string) => useResumeStore.getState().resumes.find((r) => r.id === id);
   813→}
