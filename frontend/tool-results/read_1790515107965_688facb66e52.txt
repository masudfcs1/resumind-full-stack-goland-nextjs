   382→/* ============================== Phase strip ============================== */
   383→
   384→/**
   385→ * Slim 3-segment difficulty ladder strip (Warm-up / Core / Stretch) shown
   386→ * above the question list. Each segment carries its phase count and a
   387→ * practiced-progress fill; the segment containing the current question is
   388→ * highlighted in its tier tint, and a one-line coach hint sits underneath.
   389→ */
   390→function PhaseStrip({
   391→  questions,
   392→  practicedIds,
   393→  currentIndex,
   394→}: {
   395→  questions: PracticeQuestion[];
   396→  practicedIds: ReadonlySet<string>;
   397→  currentIndex: number;
   398→}) {
   399→  const reducedMotion = useReducedMotion();
   400→
   401→  const segments = React.useMemo(
   402→    () =>
   403→      PHASE_SEGMENTS.map((seg) => {
   404→        const idxs = questions
   405→          .map((q, i) => (q.difficulty === seg.tier ? i : -1))
   406→          .filter((i) => i >= 0);
   407→        const practiced = idxs.filter((i) => {
   408→          const q = questions[i];
   409→          return q ? practicedIds.has(q.id) : false;
   410→        }).length;
   411→        return { ...seg, count: idxs.length, practiced };
   412→      }),
   413→    [questions, practicedIds]
   414→  );
   415→
   416→  const currentTier = questions[currentIndex]?.difficulty ?? "easy";
   417→  const currentSegment = PHASE_SEGMENTS.find((s) => s.tier === currentTier);
   418→
   419→  return (
   420→    <motion.section
   421→      variants={staggerItem}
   422→      aria-label="Difficulty phases"
   423→      data-phase-strip="true"
   424→      data-current-phase={currentSegment?.id ?? "warm-up"}
   425→    >
   426→      <div className="flex gap-1.5 sm:gap-2">
   427→        {segments.map((seg) => {
   428→          const active = seg.tier === currentTier;
   429→          const fillPct =
   430→            seg.count > 0 ? Math.round((seg.practiced / seg.count) * 100) : 0;
   431→          return (
   432→            <div
   433→              key={seg.tier}
   434→              data-phase={seg.id}
   435→              data-tier={seg.tier}
   436→              data-count={seg.count}
   437→              data-active={active ? "true" : "false"}
   438→              className={cn(
   439→                "relative min-w-0 flex-1 overflow-hidden rounded-lg border px-2.5 py-1.5 transition-colors sm:px-3",
   440→                active ? seg.active : "border-border"
   441→              )}
   442→            >
   443→              <motion.div
   444→                aria-hidden="true"
   445→                className={cn("absolute inset-y-0 left-0", seg.fill)}
   446→                initial={false}
   447→                animate={{ width: `${fillPct}%` }}
   448→                transition={
   449→                  reducedMotion
   450→                    ? { duration: 0 }
   451→                    : { type: "spring", stiffness: 140, damping: 22 }
   452→                }
   453→              />
   454→              <div className="relative flex min-w-0 items-center justify-between gap-2">
   455→                <span
   456→                  className={cn(
   457→                    "truncate text-[10px] font-semibold uppercase tracking-[0.12em]",
   458→                    active ? seg.text : "text-muted-foreground"
   459→                  )}
   460→                >
   461→                  {DIFFICULTY_META[seg.tier].phase}
   462→                </span>
   463→                <span
   464→                  className={cn(
   465→                    "shrink-0 text-[10px] font-semibold tabular-nums",
   466→                    active ? seg.text : "text-muted-foreground/70"
   467→                  )}
   468→                >
   469→                  {seg.practiced}/{seg.count}
   470→                </span>
   471→              </div>
   472→            </div>
   473→          );
   474→        })}
   475→      </div>
   476→      <p
   477→        className={cn(
   478→          "mt-2 flex items-start gap-2 rounded-lg border px-3 py-2 text-xs leading-relaxed",
   479→          PHASE_COACH_TINT[currentTier]
   480→        )}
   481→      >
   482→        <Compass className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   483→        <span>
   484→          <span className="font-semibold">Coach:</span> {PHASE_COACH[currentTier]}
   485→        </span>
   486→      </p>
   487→      <p className="sr-only">
   488→        {`Phases: ${segments
   489→          .map((s) => `${DIFFICULTY_META[s.tier].phase} ${s.count} questions`)
   490→          .join(" · ")}. Currently in the ${DIFFICULTY_META[currentTier].phase} phase.`}
   491→      </p>
   492→    </motion.section>
   493→  );
   494→}

   512→  const [generation, setGeneration] = React.useState(0);
   513→  const [practicedIds, setPracticedIds] = React.useState<Set<string>>(new Set());
   514→  const [focusOpen, setFocusOpen] = React.useState(false);
   515→  const [focusIndex, setFocusIndex] = React.useState(0);
   516→  const [starNotes, setStarNotes] = React.useState<Record<string, StarAnswer>>({});
   517→  const [sessionOrigin, setSessionOrigin] = React.useState<SessionOrigin | null>(null);
   518→  const [companyVariant, setCompanyVariant] = React.useState(0);
   519→  /** Starred companies (normalized keys) — lazy-read after mount, SSR-safe. */
   520→  const [favorites, setFavorites] = React.useState<string[]>(() => readFavorites());
   521→  const [favFilter, setFavFilter] = React.useState<"all" | "starred">("all");
   522→  const timerRef = React.useRef<number | null>(null);
   523→
   524→  /** Sync mirror of favorites so rapid toggles never race the persisted copy. */
   525→  const favoritesRef = React.useRef<string[]>(favorites);
   526→
   527→  /** Adds/removes a company from favorites and persists best-effort. */
   528→  const toggleFavorite = React.useCallback(
   529→    (companyName: string) => {
   530→      const key = favoriteKey(companyName);
   531→      const cur = favoritesRef.current;
   532→      const wasFavorite = cur.includes(key);
   533→      const next = wasFavorite
   534→        ? cur.filter((f) => f !== key)
   535→        : [...cur, key];
   536→      favoritesRef.current = next;
   537→      setFavorites(next);
   538→      if (!writeFavorites(next)) {
   539→        toast.error("Couldn't save favorites", {
   540→          description: "Browser storage is unavailable in this session.",
   541→        });
   542→      }
   543→      if (wasFavorite) {
   544→        toast(`${companyName} removed from favorites`);
   545→      } else {
   546→        toast.success(`${companyName} added to favorites`);
   547→      }
   548→    },
   549→    []
   550→  );
   551→
   552→  React.useEffect(() => {
   553→    // If the last favorite is un-starred while the Starred filter is active,
   554→    // fall back to All so the list never renders as an empty mystery.
   555→    if (favFilter === "starred" && favorites.length === 0) setFavFilter("all");
   556→  }, [favFilter, favorites.length]);
   557→
   558→  React.useEffect(() => {
   559→    return () => {
   560→      if (timerRef.current !== null) window.clearTimeout(timerRef.current);
   561→    };
   562→  }, []);
   563→
   564→  const generate = React.useCallback(() => {
   565→    if (!resume || generating) return;
   566→    setGenerating(true);
   567→    const next = generateQuestions(resume, {
   568→      categories: selectedCategories,
   569→      difficulty,
   570→      count: QUESTIONS_PER_SET,
   571→    });
   572→    timerRef.current = window.setTimeout(() => {
   573→      if (next.length === 0) {
   574→        toast.warning("No questions match those filters", {
   575→          description: "Try selecting more categories or a different difficulty.",
   576→        });
   577→        setGenerating(false);
   578→        return;
   579→      }
   580→      setSession(next);
   581→      setPracticedIds(new Set());
   582→      setFocusIndex(0);
   583→      setGeneration((g) => g + 1);
   584→      setSessionOrigin({ kind: "bank" });
   585→      setCompanyVariant(0);
