    99→function initialsOf(name: string): string {
   100→  const parts = name.trim().split(/\s+/).filter(Boolean);
   101→  if (parts.length === 0) return "?";
   102→  return parts
   103→    .slice(0, 2)
   104→    .map((p) => p[0]!.toUpperCase())
   105→    .join("");
   106→}

   110→interface ParsedBackup {
   111→  payload: Record<string, unknown>;
   112→  resumeCount: number;
   113→  applicationCount: number | null; // null = absent, store will fall back to seed applications
   114→  scoreHistoryCount: number | null; // null = absent, store will fall back to seed score history
   115→  exportedAt: string | null;
   116→  /** "v2" when the backup carries a 2.x version string or a meta object, "v1" otherwise. */
   117→  formatVersion: "v1" | "v2";
   118→  /** Interview favorites from meta.favorites.interview — null when absent or malformed (v1 backups, bare v2). */
   119→  interviewFavorites: string[] | null;
   120→}
   121→
   122→function parseBackupText(text: string): { ok: true; data: ParsedBackup } | { ok: false; reason: string } {
   123→  let parsed: unknown;
   124→  try {
   125→    parsed = JSON.parse(text);
   126→  } catch {
   127→    return { ok: false, reason: "That file isn't valid JSON. Export a fresh backup from Settings and try again." };
   128→  }
   129→  if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) {
   130→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — expected a JSON object." };
   131→  }
   132→  const obj = parsed as Record<string, unknown>;
   133→  const rawResumes = Array.isArray(obj.resumes) ? obj.resumes : null;
   134→  if (obj.app !== "ResumeForge AI" && !rawResumes) {
   135→    return { ok: false, reason: "This doesn't look like a ResumeForge backup — the \"app\" marker and resumes array are missing." };
   136→  }
   137→  if (!rawResumes || rawResumes.length === 0) {
   138→    return { ok: false, reason: "The backup contains no resumes, so there is nothing to restore." };
   139→  }
   140→  for (let i = 0; i < rawResumes.length; i++) {
   141→    const entry: unknown = rawResumes[i];
   142→    if (!entry || typeof entry !== "object" || Array.isArray(entry)) {
   143→      return { ok: false, reason: `Resume #${i + 1} in the backup is malformed (expected an object).` };
   144→    }
   145→    const rec = entry as Record<string, unknown>;
   146→    const titleOk = typeof rec.title === "string";
   147→    if (
   148→      typeof rec.id !== "string" ||
   149→      rec.id.trim() === "" ||
   150→      !titleOk ||
   151→      !rec.personal ||
   152→      typeof rec.personal !== "object" ||
   153→      Array.isArray(rec.personal)
   154→    ) {
   155→      const label = titleOk ? `"${rec.title}"` : `#${i + 1}`;
   156→      return {
   157→        ok: false,
   158→        reason: `Resume ${label} is missing an id, a title or personal details — the backup looks corrupted.`,
   159→      };
   160→    }
   161→  }
   162→  /* Format detection (v1 vs v2): a 2.x version string or a meta object marks a v2 backup.
   163→     Everything else — including versionless legacy exports — is treated as v1. */
   164→  const rawMeta: unknown = obj.meta;
   165→  const hasMeta = !!rawMeta && typeof rawMeta === "object" && !Array.isArray(rawMeta);
   166→  const versionIs2 = typeof obj.version === "string" && obj.version.trim().startsWith("2");
   167→  const formatVersion: "v1" | "v2" = versionIs2 || hasMeta ? "v2" : "v1";
   168→
   169→  /* v2 extras: meta.favorites.interview is an optional array of company keys.
   170→     Normalized the same way readFavorites() stores them (trim/lowercase/dedupe). */
   171→  let interviewFavorites: string[] | null = null;
   172→  if (formatVersion === "v2" && hasMeta) {
   173→    const favorites: unknown = (rawMeta as Record<string, unknown>).favorites;
   174→    if (favorites && typeof favorites === "object" && !Array.isArray(favorites)) {
   175→      const interview: unknown = (favorites as Record<string, unknown>).interview;
   176→      if (Array.isArray(interview)) {
   177→        interviewFavorites = interview
   178→          .filter((v): v is string => typeof v === "string")
   179→          .map((v) => v.trim().toLowerCase())
   180→          .filter((v) => v.length > 0)
   181→          .filter((v, i, arr) => arr.indexOf(v) === i); // dedupe, keep first
   182→      }
   183→    }
   184→  }
   185→
   186→  return {
   187→    ok: true,
   188→    data: {
   189→      payload: obj,
   190→      resumeCount: rawResumes.length,
   191→      applicationCount: Array.isArray(obj.applications) ? obj.applications.length : null,
   192→      scoreHistoryCount: Array.isArray(obj.scoreHistory) ? obj.scoreHistory.length : null,
   193→      exportedAt: typeof obj.exportedAt === "string" ? obj.exportedAt : null,
   194→      formatVersion,
   195→      interviewFavorites,
   196→    },
   197→  };
   198→}

   200→function formatBackupDate(iso: string | null): string {
   201→  if (!iso) return "Not recorded";
   202→  const date = new Date(iso);
   203→  if (!isValidDate(date)) return "Not recorded";
   204→  return format(date, "MMM d, yyyy 'at' h:mm a");
   205→}

   223→/** Number of notification ids marked as read in the bell's read map. */
   224→function countReadNotificationIds(): number {
   225→  if (typeof window === "undefined") return 0;
   226→  try {
   227→    const raw = window.localStorage.getItem(NOTIFS_READ_KEY);
   228→    if (!raw) return 0;
   229→    const parsed: unknown = JSON.parse(raw);
   230→    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return 0;
   231→    return Object.keys(parsed as Record<string, unknown>).length;
   232→  } catch {
   233→    return 0;
   234→  }
